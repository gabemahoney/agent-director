# agent-10: a Codex launch writes nothing into any Codex folder

Maps to SRD SR-5.2, SR-9 and SR-14.5 of Epic 13 (`b.ssz`): everything a
Codex launch needs from agent-director (hook entries, their trust, the
typing setting and the folder trust) is on the launch's command line, so a
`spawn --agent codex` reads and writes no file of Codex's. With
`CODEX_HOME` given in `--extra-env`, that folder's tree (files, modes and
content hashes) is the same before and after the launch; with `CODEX_HOME`
unset, `~/.codex` is not created. The launch is not a Claude Code launch
either: it writes no `~/.claude.json` and no `~/.claude`.

Agent and hooks (collector "Driver mode"): `[launch] codex_path` names the
Codex stand-in `/opt/driver/stand-in-codex/codex`, which records its
launch in `/tmp/stand-in-codex/<instance id>/` and writes nothing else.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path`, and makes a Codex folder of its own
(`config.toml`, `auth.json` mode 600, `sessions/` with one dated
subdirectory and a file in it). It removes the config, the folder, the
cwds it made and the stand-in's record root on exit.

## Steps

1. Write the config, make the folder and a cwd, and take
   `/opt/driver/tree-snapshot.sh <folder>`; note whether `~/.codex`,
   `~/.claude.json` and `~/.claude` exist.
2. `agent-director spawn --agent codex --cwd <cwd> --extra-env
   CODEX_HOME=<folder> --instance-id <id1> --tmux-session-name cd-ag-cf1`
   → exit 0, `pre_trust` `ok`.
3. Wait for the stand-in's record of id1; take the folder's snapshot again.
4. `agent-director spawn --agent codex --cwd <cwd> --instance-id <id2>
   --tmux-session-name cd-ag-cf2` (no `CODEX_HOME` anywhere) → exit 0,
   `pre_trust` `ok`. Wait for the stand-in's record of id2.
5. Check `~/.codex`, `~/.claude.json` and `~/.claude`.

## Pass criteria

- Step 2 and 4: exit 0, the `instance_id` is the id sent, `pre_trust` is
  `ok`, and the row is `codex` and `pending`.
- Step 3: the stand-in's environment for id1 has `CODEX_HOME=<folder>`; the
  folder's snapshot is identical to the one of step 1 (no new file, no
  changed mode or content, no `hooks.json`).
- Step 4: the stand-in's environment for id2 has no `CODEX_HOME`.
- Step 5: `~/.codex` does not exist; `~/.claude.json` and `~/.claude` exist
  exactly as they did in step 1 (a `~/.claude.json` that existed has the
  same content hash).

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call. CODEX_HOME is
# unset here: the case gives it to the first spawn as --extra-env.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
id1="ag-cf1-$uuid"
id2="ag-cf2-$uuid"
codex_home=""
cwd=""
snap_before=""
snap_after=""
fail() { echo "FAIL: $*"; exit 1; }

cleanup() {
    local id row sock pane sid
    for id in "$id1" "$id2"; do
        row="$(/opt/driver/sql.sh -readonly "$db" "SELECT IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'') FROM spawns WHERE instance_id = '$id';" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        [ -n "$pane" ] || continue
        sid="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
        [ -z "$sid" ] || tmux ${sock:+-S "$sock"} kill-session -t "$sid" 2>/dev/null || true
    done
    rm -f "$config" ${snap_before:+"$snap_before"} ${snap_after:+"$snap_after"}
    rm -rf "$rec_root/$id1" "$rec_root/$id2" ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# Step 1: setup and the before state.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
codex_home="$(mktemp -d)"
printf 'model = "gpt-5"\n' > "$codex_home/config.toml"
printf '{"stand-in":true}\n' > "$codex_home/auth.json"
chmod 600 "$codex_home/auth.json"
mkdir -p "$codex_home/sessions/2026/10/11"
printf '{"type":"session_meta"}\n' > "$codex_home/sessions/2026/10/11/rollout-stand-in.jsonl"
cwd="$(mktemp -d /tmp/ag-cf-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
snap_before="$(mktemp)"
snap_after="$(mktemp)"
/opt/driver/tree-snapshot.sh "$codex_home" > "$snap_before"
[ ! -e "$HOME/.codex" ] || fail "$HOME/.codex exists before the case"
claude_json_before="absent"
[ ! -e "$HOME/.claude.json" ] || claude_json_before="$(sha256sum < "$HOME/.claude.json")"
claude_dir_before="absent"
[ ! -e "$HOME/.claude" ] || claude_dir_before="$(/opt/driver/tree-snapshot.sh "$HOME/.claude" | sha256sum)"

# spawn_codex ID NAME [EXTRA_ARGS...]: a codex spawn that must succeed with pre_trust ok;
# waits for the stand-in's record of ID.
spawn_codex() {
    local id="$1" name="$2" out rc err rec end
    shift 2
    err="$(mktemp)"
    set +e
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$id" --tmux-session-name "$name" "$@" 2>"$err")"
    rc=$?
    set -e
    [ "$rc" -eq 0 ] || fail "spawn $id exited $rc; want 0: $(cat "$err")"
    rm -f "$err"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$id" ] || fail "spawn echoed [$out]; want id $id"
    [ "$(jq -r '.pre_trust' <<<"$out")" = "ok" ] || fail "spawn $id: pre_trust is [$(jq -r '.pre_trust' <<<"$out")]; want ok: $out"
    [ "$(agent-director get --instance-id "$id" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] || fail "row $id is not codex|pending"
    rec="$rec_root/$id"
    end=$(( $(date +%s%N) + 10000000000 ))
    until [ -s "$rec/pid" ]; do
        [ "$(date +%s%N)" -lt "$end" ] || fail "no stand-in record at $rec within 10 s: the program named by codex_path did not run"
        sleep 0.1
    done
}

# env_has ID ENTRY: the stand-in's recorded environment of ID holds the exact entry ENTRY.
env_has() {
    local e
    while IFS= read -r -d '' e; do
        [ "$e" = "$2" ] && return 0
    done < "$rec_root/$1/env"
    return 1
}
# env_has_name ID NAME: the recorded environment holds an entry named NAME.
env_has_name() {
    local e
    while IFS= read -r -d '' e; do
        case "$e" in "$2="*) return 0 ;; esac
    done < "$rec_root/$1/env"
    return 1
}

# Steps 2-3: with CODEX_HOME given, the folder is the same afterwards.
spawn_codex "$id1" cd-ag-cf1 --extra-env "CODEX_HOME=$codex_home"
env_has "$id1" "CODEX_HOME=$codex_home" || fail "the launch of $id1 does not carry CODEX_HOME=$codex_home"
/opt/driver/tree-snapshot.sh "$codex_home" > "$snap_after"
diff -u "$snap_before" "$snap_after" || fail "the Codex folder $codex_home changed across the spawn"

# Step 4: with no CODEX_HOME.
spawn_codex "$id2" cd-ag-cf2
if env_has_name "$id2" CODEX_HOME; then fail "the launch of $id2 carries a CODEX_HOME"; fi

# Step 5: nothing created at the default locations.
[ ! -e "$HOME/.codex" ] || fail "$HOME/.codex was created"
if [ "$claude_json_before" = "absent" ]; then
    [ ! -e "$HOME/.claude.json" ] || fail "$HOME/.claude.json was created by a codex spawn"
else
    [ "$(sha256sum < "$HOME/.claude.json")" = "$claude_json_before" ] || fail "$HOME/.claude.json was changed by a codex spawn"
fi
if [ "$claude_dir_before" = "absent" ]; then
    [ ! -e "$HOME/.claude" ] || fail "$HOME/.claude was created by a codex spawn"
else
    [ "$(/opt/driver/tree-snapshot.sh "$HOME/.claude" | sha256sum)" = "$claude_dir_before" ] || fail "$HOME/.claude was changed by a codex spawn"
fi
# The first spawn's folder is still as it was after the second one.
/opt/driver/tree-snapshot.sh "$codex_home" | diff -u "$snap_before" - || fail "the Codex folder $codex_home changed across the second spawn"

echo "agent-10 ok: two codex spawns (CODEX_HOME given, and unset) -> pre_trust ok; the given folder's tree is unchanged, no ~/.codex, no ~/.claude.json or ~/.claude written"
```
