# Native journal durability across CLI processes

Maps to t3.7tz.tb.ib.yc, audit 21/25/26/33 and SR-2. A synthetic Claude
2.1.280 composer supplies the existing diagnostic/proof format through a real
raw terminal; it is not evidence qualifying a new provider/version/login.
Real SQLite, real process identities, tmux transport and independent CLI
processes exercise the integration boundary. Native API/provider matrices
remain owned by w3; Cursor retains the existing separate collectors.

## Setup and steps

Create a real previous-release store and migrate it with the administrator's
stopped-writer backup. Spawn a fresh supported launch. Inject a real SQLite INSERT failure: no input
may reach the pane. Use the same-source `/work/source/bin/ts-helper` built by
the harness to install and remove the canonical failure by its handle.
Observe `synchronous=FULL` inside the
actual connection at attempt insertion and each irreversible text/Enter intent
update using triggers, retaining phase, attempted bytes and Enter intent. Send a payload spanning
multiple chunks, then stop the calling CLI after Enter and before proof is
published. A new request and the unresolved original must refuse without sending.
Preserve the later DB/provider evidence, stop the agent, and restore the genuine
earlier store: it lacks the later request and proof. Reconcile the original from
an inspection copy of the preserved DB, then deliberately run the matching old
binary only after exact proof resolves the uncertainty. Repeated original requests
return identical proof without replay. FIFOs synchronize boundaries;
there are no wall-clock sleeps or screen-text readiness decisions.

## Pass criteria

Failed intent commit writes zero terminal bytes. The successful intent records
FULL for both chunks and Enter, one Enter and the exact payload. Caller death leaves unresolved authority.
Late proof belongs to the same request/launch; different text cannot steal it.
Repeated original calls return equal JSON and keep proof BLOB and input counts.
Restore loses the later request, but preserved evidence still records its exact
launch/request and delivered bytes. The unresolved checkpoint is observed before
reconciliation and older-runtime continuation; recovery never sends more text or Enter.

set -euo pipefail

test -x "$helper"

  result=$?

    for name in terminal-in-pane terminal-probe migrate-out migrate-err fail-out fail-err crash-out crash-err other-err unresolved-err proof-1 proof-1-err proof-2 proof-2-err got-intents want-intents; do

    fi
  fi

  test -z "$sid" || tmux kill-session -t "$sid" 2>/dev/null || true
  rm -rf "$work"
}
trap cleanup EXIT

  --backup-dir "$work/pre-upgrade" >"$work/migrate-out" 2>"$work/migrate-err"

cat >"$work/bin/claude" <<'JS'
#!/usr/bin/env node
const fs=require('fs'), path=require('path'), crypto=require('crypto');
if(process.argv.includes('--version')) { console.log('2.1.280 (Claude Code)'); process.exit(0); }
const arg=k=>process.argv[process.argv.indexOf(k)+1];
const session=arg('--session-id'), debug=arg('--debug-file');
const transcript=path.join(process.env.CLAUDE_CONFIG_DIR,'projects',process.cwd().replace(/[^A-Za-z0-9-]/g,'-'),session+'.jsonl');
fs.mkdirSync(path.dirname(transcript),{recursive:true});
fs.writeFileSync(transcript,'');
fs.writeFileSync(process.env.NATIVE_CAPTURE,'');
process.stdin.setRawMode(true); process.stdin.resume();
const terminal=require('child_process').spawnSync('python3',['-c',
  'import json,os; print(json.dumps({"owner":os.getppid(),"group":os.getpgrp(),"foreground":os.tcgetpgrp(0)}))'],
  {stdio:['inherit','pipe','inherit']});
if(terminal.status!==0) process.exit(1);
fs.writeFileSync(process.env.NATIVE_TERMINAL,terminal.stdout);
fs.writeFileSync(debug,'Terminal capabilities:\n');
fs.writeFileSync(process.env.NATIVE_READY,'ready\n');
let text='';
process.stdin.on('data',data=>{
  fs.appendFileSync(process.env.NATIVE_CAPTURE,data);
  text+=data.toString('utf8');
  const enter=text.indexOf('\r');
  if(enter<0) return;
  const payload=text.slice(0,enter).replace(/\x1b\[20[01]~/g,'');
  text=text.slice(enter+1);
  fs.writeFileSync(process.env.NATIVE_ENTERED,'entered\n');
  fs.readFileSync(process.env.NATIVE_RELEASE);
  const record={type:'user',uuid:crypto.randomUUID(),sessionId:session,cwd:process.cwd(),version:'2.1.280',message:{role:'user',content:payload}};
  const raw=JSON.stringify(record);
  fs.writeFileSync(process.env.NATIVE_PROOF,raw);
  fs.appendFileSync(transcript,raw+'\n');
  fs.writeFileSync(process.env.NATIVE_PUBLISHED,'published\n');
});
JS

PATH="$work/bin:$PATH" agent-director --store-path "$db" spawn --agent claude --instance-id "$id" --cwd /tmp --relay-mode off --no-pre-trust \

import array, fcntl, os, sys, termios
pid = int(sys.argv[1])
tty = os.readlink(f'/proc/{pid}/fd/0')
print('pid', pid, 'tty', tty, 'pgid', os.getpgid(pid), flush=True)
stat = open(f'/proc/{pid}/stat').read().rsplit(')', 1)[1].split()
print('proc session', stat[3], 'tty_nr', stat[4], 'tpgid', stat[5], flush=True)
fd = os.open(tty, os.O_RDONLY | os.O_NOCTTY | os.O_NONBLOCK)
try:
    attrs = termios.tcgetattr(fd)
    print('canonical', bool(attrs[3] & termios.ICANON), 'echo', bool(attrs[3] & termios.ECHO), flush=True)
    for name, code in [('foreground', termios.TIOCGPGRP), ('queue', termios.TIOCINQ)]:
        result = array.array('i', [0])
        try:
            fcntl.ioctl(fd, code, result)
            print(name, result[0], flush=True)

finally:

installed=$("$helper" install-write-failure --store "$db" --kind input-attempt --id "$id")
failure_handle=$(jq -er '.handle' <<<"$installed")
rc=0
timeout 20 agent-director --store-path "$db" send-keys --instance-id "$id" --allow-pending --delivery-request-id failed --text no-send >"$work/fail-out" 2>"$work/fail-err" || rc=$?
test "$rc" -ne 0

jq -e '.err_name=="ErrSubmissionUnconfirmed" and .err_details.outcome=="not_started" and .err_details.phase=="pre_input" and .err_details.reason=="cannot persist attempt before input"' "$work/fail-err" >/dev/null

test ! -s "$work/input"
"$helper" remove-write-failure --store "$db" --handle "$failure_handle" >/dev/null
"$sql" "$db" <<'SQL'
CREATE TABLE journal_full(request_id TEXT, revision INTEGER, phase TEXT, bytes INTEGER, enter_intent INTEGER, value INTEGER);
CREATE TRIGGER journal_durability_insert AFTER INSERT ON input_attempts
BEGIN
  INSERT INTO journal_full SELECT NEW.request_id, NEW.revision,
    json_extract(NEW.record,'$.phase'), json_extract(NEW.record,'$.bytes_attempted'),
    json_extract(NEW.record,'$.enter_attempted'), synchronous FROM pragma_synchronous;
END;
CREATE TRIGGER journal_durability_update AFTER UPDATE ON input_attempts
WHEN json_extract(NEW.record,'$.bytes_attempted') != json_extract(OLD.record,'$.bytes_attempted')
  OR json_extract(NEW.record,'$.enter_attempted') != json_extract(OLD.record,'$.enter_attempted')
BEGIN
  INSERT INTO journal_full SELECT NEW.request_id, NEW.revision,
    json_extract(NEW.record,'$.phase'), json_extract(NEW.record,'$.bytes_attempted'),
    json_extract(NEW.record,'$.enter_attempted'), synchronous FROM pragma_synchronous;
END;

sender=''

import os, select, subprocess, sys
