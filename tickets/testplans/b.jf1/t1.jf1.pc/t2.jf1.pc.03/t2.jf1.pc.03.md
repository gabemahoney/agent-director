# Test case: Confirmed clear and new current-life public history

Maps E3 AC4/5; SR-0.2, SR-3.3, SR-4.1–4.2, SR-6.2.

## Setup

Exact image CLI/MCP, isolated helper and synthetic runtime producing new session
IDs. Derive confirmed identity from actual protocol replies, never a desired DB row.

## Steps

Send /clear through CLI and /new through terminal-shaped input. For each confirmed
session/new reply compare current public session, actual protocol session,
current-life SQL history and stable launch/runtime/file identity. Finish with /exit.

## Pass criteria

Both real replies promote exactly once and archive the outgoing session. No model
prompt for native commands, no life/token/path/inode change, exact owned death on exit.

```bash
set -euo pipefail
source /work/source/test/driver/cursor-pane-controls.sh
pc_start
inode=$(stat -c '%d:%i' "$path")
n=0
for control in /clear /new; do
 n=$((n+1)); old=$(sql "SELECT session_id FROM spawns WHERE instance_id='$id';")
 if [[ $control == /clear ]]; then ad send-keys --instance-id "$id" --text "$control" >/dev/null; else pc_terminal "$control"; fi
 rotated() { [[ $(sql "SELECT session_id FROM spawns WHERE instance_id='$id';") != "$old" && $(row_state) == waiting ]] && pc_rotation_published "$n"; }
 check rotation.confirmed wait_until rotated
 current=$(sql "SELECT session_id FROM spawns WHERE instance_id='$id';")
 wire=$(jq -sr '[.[]|select(.result.sessionId!=null)]|last.result.sessionId' "$case_dir/sent.jsonl")
 check rotation.actual-peer-ID test "$current" = "$wire"
 check rotation.history-count test "$(sql "SELECT count(*) FROM session_history WHERE instance_id='$id' AND life_number=(SELECT life_number FROM spawns WHERE instance_id='$id');")" = "$n"
 check rotation.outgoing-history test "$(sql "SELECT count(*) FROM session_history WHERE instance_id='$id' AND session_id='$old';")" = 1
 pc_public MCP get "$(jq -cn --arg id "$id" '{instance_id:$id}')" > "$case_dir/public.json"
 jq -e --arg old "$old" --arg current "$current" '.session_id==$current and any(.prior_sessions[];.session_id==$old)' "$case_dir/public.json" >/dev/null
 check rotation.inode test "$(stat -c '%d:%i' "$path")" = "$inode"
 jq -e --argjson original "$initial" '.[0] as $now | $original[0] as $before | all(["life_number","launch_token","pane_pid","pane_starttime","jsonl_path"][]; $now[.]==$before[.])' <<<"$(sql -json "SELECT life_number,launch_token,pane_pid,pane_starttime,jsonl_path FROM spawns WHERE instance_id='$id';")" >/dev/null
done
check rotation.no-model-prompts pc_prompts_is 0
pc_terminal /exit
check owned.runtime-gone wait_gone "$runtime"
check owned.gateway-gone wait_gone "$gateway"
check owned.ended wait_state ended
pc_finish t2.jf1.pc.03 '["rotation.clear-new-confirmed","rotation.current-life-history","rotation.actual-peer-ID","rotation.stable-launch-file","control.exit-owned-death"]'
```
