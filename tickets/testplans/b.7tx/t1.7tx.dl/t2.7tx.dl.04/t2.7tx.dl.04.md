# Test case: built Guard in a Droid launch

Maps to b.s8e SR-3 and SR-8.1 Integration; Subtask t3.jo7.ob.bd.bb.

## Setup

Use the fresh harness database, built CLI and recording stand-in, with no
Guard selector override. Reuse `toolguardfix.New` unchanged through a temporary
Go caller in this case. Its decoy root contains quotes, spaces and shell
metacharacters; only its invented `home/token` path goes in the Path List.
No real Droid, account, credentials, network or billed turn is needed.

## Steps

1. Create the shared decoys and snapshot their contents and Factory settings.
2. Configure the Path List and launch Droid. The stand-in reads its actual
   `--settings` file and runs its exact Guard command through POSIX `sh` on
   a Read of the listed decoy and a Read of an ordinary file outside the list.
3. Wait for a FIFO receipt from the interactive shell, then check the recorded
   command outcomes, overlay, argv and private file modes.
4. Compare decoys and Factory settings to their snapshots, then kill the
   case's launch and remove its temporary state.

## Pass criteria

The protected Read exits 2 with the exact protected-path diagnostic; the
ordinary Read exits 0 with no stderr. Neither writes a stdout byte. The
overlay is exactly model plus the required hooks, in that order; the pane
receives only `--settings` and its private path. The pane consumed those same
bytes, stayed usable, and changed no decoy or Factory setting.

```bash
set -euo pipefail
case_dir=$(mktemp -d /tmp/droid-guard.XXXXXX)
export TMUX_TMPDIR="$case_dir/tmux"
export AD_CODEX_RECORD_DIR="$case_dir/panes"
export AD_DROID_GUARD_RECORD_DIR="$case_dir/guard"
unset AD_CODEX_HOOK AD_DROID_RULES_RECORD_DIR AD_DROID_RULES_CHECK_EXIT
mkdir -p "$TMUX_TMPDIR" "$AD_CODEX_RECORD_DIR" "$AD_DROID_GUARD_RECORD_DIR" "$case_dir/seed"
cfg="$HOME/.config/agent-director/config.toml"
db="$HOME/.config/agent-director/state.db"
launch="$HOME/.config/agent-director/launch"
factory="$HOME/.factory"
test ! -e "$cfg"
test ! -e "$factory"
test ! -e "$launch"
cleanup() {
    tmux kill-server 2>/dev/null || true
    rm -f "$cfg"
    rm -rf "$launch" "$factory" "$case_dir"
}
trap cleanup EXIT
mkdir -p "$factory"
printf '{"fixture":"unchanged"}\n' >"$factory/settings.json"
/opt/driver/tree-snapshot.sh "$factory" >"$case_dir/factory-before"

# The existing fixture is Go-only. This temporary caller imports it directly;
# it creates no second layout and needs no change to the shared smoke helper.
cat >"$case_dir/seed/go.mod" <<'EOF'
module github.com/gabemahoney/agent-director/guard-integration-fixture

go 1.27.3

require github.com/gabemahoney/agent-director v0.0.0
replace github.com/gabemahoney/agent-director => /work/source
EOF
cat >"$case_dir/seed/main.go" <<'EOF'
package main

import (
    "encoding/json"
    "os"

    "github.com/gabemahoney/agent-director/internal/testsupport/toolguardfix"
)

func main() {
    layout, err := toolguardfix.New(os.Args[1])
    if err != nil { panic(err) }
    if err := json.NewEncoder(os.Stdout).Encode(layout); err != nil { panic(err) }
}
EOF
decoys="$case_dir/decoys 'quoted' \$cash; [x]"
mkdir "$decoys"
(cd "$case_dir/seed" && go run . "$decoys") >"$case_dir/layout.json"
secret=$(jq -er .SecretFile "$case_dir/layout.json")
ordinary=$(jq -er .OrdinaryFile "$case_dir/layout.json")
project=$(jq -er .Work "$case_dir/layout.json")
printf '%s\n' "$secret" >"$case_dir/secret-paths"
/opt/driver/tree-snapshot.sh "$decoys" >"$case_dir/decoys-before"
for label in blocked allowed; do
    target="$secret"
    [[ "$label" == blocked ]] || target="$ordinary"
    jq -cn --arg cwd "$project" --arg target "$target" \
        '{hook_event_name:"PreToolUse",cwd:$cwd,tool_name:"Read",tool_input:{file_path:$target}}' \
        >"$AD_DROID_GUARD_RECORD_DIR/$label.json"
done
cat >"$cfg" <<EOF
[experimental]
droid = true
[launch]
droid_path = "/opt/driver/stand-in-codex/codex"
droid_secret_paths_file = "$case_dir/secret-paths"
EOF
mkfifo "$case_dir/ready"
exec 7<>"$case_dir/ready"
agent-director spawn --agent droid --cwd "$project" -- --model guard-model >"$case_dir/spawn"
jq -e '.pre_trust == "skipped"' "$case_dir/spawn" >/dev/null
id=$(jq -er .instance_id "$case_dir/spawn")
pane=$(/opt/driver/sql.sh -readonly "$db" "SELECT pane_id FROM spawns WHERE instance_id='$id';")
# This known shell starts only after both Guard invocations and the recorder.
tmux send-keys -t "$pane" "printf 'ready\\n' > '$case_dir/ready'" Enter
IFS= read -r -t 15 reply <&7
test "$reply" = ready
record="$AD_CODEX_RECORD_DIR/$id"
mapfile -d '' -t argv <"$record/argv"
test "${#argv[@]}" = 2
test "${argv[0]}" = --settings
settings="${argv[1]}"
test "$(stat -c %a "$settings")" = 600
test "$(stat -c %a "$(dirname "$settings")")" = 700
cmp "$settings" "$AD_DROID_GUARD_RECORD_DIR/pane-settings"
jq -e '
    keys_unsorted == ["model","hooks"] and .model == "guard-model" and
    (.hooks.PreToolUse[0].hooks[0].command | type == "string" and length > 0) and
    .hooks == {hooksDisabled:false,PreToolUse:[{matcher:"*",hooks:[{
        type:"command",command:.hooks.PreToolUse[0].hooks[0].command,timeout:60
    }]}]}' "$settings" >/dev/null
jq -r '.hooks.PreToolUse[0].hooks[0].command' "$settings" >"$case_dir/command"
cmp "$case_dir/command" "$AD_DROID_GUARD_RECORD_DIR/command"
test "$(cat "$AD_DROID_GUARD_RECORD_DIR/blocked.exit")" = 2
test "$(cat "$AD_DROID_GUARD_RECORD_DIR/allowed.exit")" = 0
test ! -s "$AD_DROID_GUARD_RECORD_DIR/blocked.out"
test ! -s "$AD_DROID_GUARD_RECORD_DIR/allowed.out"
test ! -s "$AD_DROID_GUARD_RECORD_DIR/allowed.err"
printf 'agent-director blocked this call: this call accesses a protected path\n' >"$case_dir/expected-error"
cmp "$case_dir/expected-error" "$AD_DROID_GUARD_RECORD_DIR/blocked.err"
/opt/driver/tree-snapshot.sh "$factory" >"$case_dir/factory-after"
/opt/driver/tree-snapshot.sh "$decoys" >"$case_dir/decoys-after"
cmp "$case_dir/factory-before" "$case_dir/factory-after"
cmp "$case_dir/decoys-before" "$case_dir/decoys-after"
agent-director kill --instance-id "$id" | jq -e '.kill_sent == true' >/dev/null
printf 'Real Droid Guard blocked/allowed overlay calls passed\n'
```
