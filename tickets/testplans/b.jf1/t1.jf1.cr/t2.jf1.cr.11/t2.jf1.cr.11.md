# Test case: Cursor origin, claim, issue and durable nonreplay

Maps to t3.7tz.6o.5h.ov and SR-5.5. Reuse the private owned-launch driver and synthetic ACP peer with captured tool/plan shapes. No authenticated-provider or power-loss qualification is claimed.

## Steps

Admit tool and plan requests through the shipped gateway in relay and native modes. Observe committed origin, issue and acknowledgement independently before the peer parses any response. Include closed-row fault injection and a real runtime crash after raw receipt but before parsing; reopen through public readers and attempt no replay. SQL is observation or explicitly labelled fault injection. Real SQLite before-COMMIT/lost-result fault injection and process crash/reopen are paired store tests TestPermissionClaimSQLiteCommitBoundary, TestPermissionClaimCrashReopen and TestCursorClaimAndIssueSQLiteBoundary, not claims that an injected error establishes filesystem durability.

## Pass criteria

Every selected response has the exact expected result, origin and issue state; native output has no relay acknowledgement. Closed requests produce zero additional raw bytes after a deterministic refusal-processing barrier while the helper lives. The entire wire after explicit shutdown is exactly that unchanged prefix plus either no bytes (runtime already gone) or one complete session/cancel for the same session; no reply, partial frame or arbitrary suffix is permitted. Crash results retain issued state and no execution proof through independent reopen, and retries emit no response. Positive controls ensure the peer and observers are live.

```bash
set -euo pipefail
receipts=$(mktemp /tmp/cr-origin-receipts-XXXXXX)
for kind in tool plan; do
 for relay in on off; do
  for cut in positive closed crash; do
   (
    source /work/source/test/driver/cursor-pane-controls.sh
    relay_scenario plan
    node - "$case_dir/scenario" "$kind" <<'SCENARIO'
const fs=require('fs'),p=process.argv[2],s=JSON.parse(fs.readFileSync(p));
s.requests=s.requests.filter(r=>(r.method==='cursor/create_plan')===(process.argv[3]==='plan'));
fs.writeFileSync(p,JSON.stringify(s));
SCENARIO
    if [[ $cut == closed && $relay == off ]]; then relay_gate_after_ack native-choice; fi
    relay_start "$relay"; wait_until request_count_is 1
    token=$(sql "SELECT request_token FROM permission_requests WHERE instance_id='$id';")
    [[ $(sql 'PRAGMA journal_mode;') == wal ]]
    relay_observe
    path=$(sql "SELECT jsonl_path FROM spawns WHERE instance_id='$id';")
    before=$(raw_size)
    cp "$case_dir/raw" "$case_dir/raw-before-closed"
    if [[ $cut == closed ]]; then
     # Explicit observational closeout fault; admission still used public writers.
     sql "UPDATE permission_requests SET closed_at=CAST(strftime('%s','now') AS INTEGER)*1000,decision='deny',decision_reason='ended',decision_origin='observation' WHERE request_token='$token';"
     if [[ $relay == on ]]; then
      if relay_decide "$token" allow >"$case_dir/out" 2>"$case_dir/error"; then exit 1; fi
      jq -e '.err_name=="ErrNoOpenPermissionRequest"' "$case_dir/error" >/dev/null
     else
      # Hold the native claim's independent check inside its writer transaction.
      touch "$case_dir/gate-armed"
      tmux -S "$socket" send-keys -t "$pane" -l -- 1
      wait_until test -f "$case_dir/gate-ready"
      touch "$case_dir/gate-release"
      # This lock can be acquired only after the consumed choice's claim attempt
      # finishes. No retirement can make an unprocessed key look like rejection.
      sql 'BEGIN IMMEDIATE; ROLLBACK;'
     fi
     # The native branch has consumed its choice and released its writer.
     # The relay branch has returned its explicit caller refusal. Two ordered
     # observations now prove the gateway advanced past the first frame before
     # comparing the complete wire. These events grant no answer authority.
     node - "$session" "$token" "$case_dir/closed-barrier.raw" <<'BARRIER'
const fs=require('fs'),[session,token,path]=process.argv.slice(2);
const frames=['observed','processed'].map(stage=>({jsonrpc:'2.0',method:'session/update',params:{sessionId:session,update:{sessionUpdate:'agent_message_chunk',content:{type:'text',text:'closed-refusal:'+token+':'+stage}}}}));
fs.writeFileSync(path,frames.map(f=>JSON.stringify(f)+'\n').join(''));
BARRIER
     jq -cn --arg id "closed-$token" --rawfile raw "$case_dir/closed-barrier.raw" '{id:$id,raw:$raw}' > "$case_dir/events.tmp"
     mv "$case_dir/events.tmp" "$case_dir/events.json"
     closed_processed() {
      node - "$path" "$case_dir/closed-barrier.raw" "$case_dir/closed-processing.json" <<'PROCESSED'
const fs=require('fs'),assert=require('assert'),[path,wire,receipt]=process.argv.slice(2);
const bytes=fs.readFileSync(path,'utf8'),records=bytes.slice(0,bytes.lastIndexOf('\n')).split('\n').filter(Boolean).map(JSON.parse);
const want=fs.readFileSync(wire,'utf8').trim().split('\n').map(line=>JSON.parse(line).params);
const found=want.map(p=>records.filter(r=>r.kind==='update'&&r.data.update?.update?.content?.text===p.update.content.text));
if(found.some(xs=>xs.length===0)) process.exit(1);
for(let n=0;n<2;n++) { assert.strictEqual(found[n].length,1); assert.deepStrictEqual(found[n][0].data.update,want[n]); }
assert(found[0][0].sequence<found[1][0].sequence);
assert.strictEqual(found[0][0].connection_id,found[1][0].connection_id);
fs.writeFileSync(receipt,JSON.stringify({observed:found[0][0],processed:found[1][0]})+'\n');
PROCESSED
     }
     wait_until closed_processed
     [[ $(raw_size) == "$before" ]]
     cmp "$case_dir/raw-before-closed" "$case_dir/raw"
     reply_count_is 0
     [[ $(sql "SELECT delivered_at IS NULL AND response_state IS NULL FROM permission_requests WHERE request_token='$token';") == 1 ]]
     sql -readonly -json "SELECT * FROM permission_requests WHERE request_token='$token';" > "$case_dir/closed-before.json"
     cp "$case_dir/raw" "$case_dir/raw-before-kill"
     # Cleanup may deliver the gateway's intentional session/cancel before the
     # runtime dies. Compare the ENTIRE suffix: never filter control frames.
     ad kill --instance-id "$id" >/dev/null
     wait_gone "$gateway"; wait_gone "$runtime"
     node - "$case_dir/raw-before-kill" "$case_dir/raw" "$session" "$case_dir/shutdown-wire.json" <<'SHUTDOWN'
const fs=require('fs'),assert=require('assert'),[beforePath,afterPath,session,receipt]=process.argv.slice(2);
const before=fs.readFileSync(beforePath),after=fs.readFileSync(afterPath);
assert.deepStrictEqual(after.subarray(0,before.length),before);
const suffix=after.subarray(before.length);
const cancel=Buffer.from(JSON.stringify({jsonrpc:'2.0',method:'session/cancel',params:{sessionId:session}})+'\n');
assert(suffix.length===0||suffix.equals(cancel),'unexpected complete or partial shutdown wire: '+suffix.toString('hex'));
fs.writeFileSync(receipt,JSON.stringify({before_bytes:before.length,after_bytes:after.length,suffix_hex:suffix.toString('hex'),expected_cancel_hex:cancel.toString('hex')})+'\n');
SHUTDOWN
     reply_count_is 0
     [[ $(sql "SELECT delivered_at IS NULL AND response_state IS NULL FROM permission_requests WHERE request_token='$token';") == 1 ]]
     sql -readonly -json "SELECT * FROM permission_requests WHERE request_token='$token';" > "$case_dir/closed-after.json"
    else
     if [[ $cut == crash ]]; then printf '\nkill -KILL "$PPID"\n' >> "$case_dir/observer"; fi
     if [[ $relay == on ]]; then relay_decide "$token" allow >/dev/null; else tmux -S "$socket" send-keys -t "$pane" -l -- 1; fi
     if [[ $cut == positive ]]; then
      wait_until reply_count_is 1
      expected='{"outcome":{"optionId":"allow-once","outcome":"selected"}}'
      [[ $kind != plan ]] || expected='{"outcome":{"outcome":"accepted"}}'
      [[ $(frames | jq -cS 'select(has("method")|not)|.result') == "$expected" ]]
     else
      wait_gone "$runtime"; wait_gone "$gateway"
      [[ $(raw_size) -gt "$before" ]]
      reply_count_is 0
     fi
     origin=caller; issued=relay_issued
     if [[ $relay == off ]]; then origin=pane; issued=native_issued; fi
     jq -se --arg token "$token" --arg origin "$origin" --arg issued "$issued" --arg relay "$relay" 'def issued: .decision=="allow" and .response_state==$issued and .decision_origin==$origin and .proven_gone_at==null and ((.delivered_at!=null)==($relay=="on")); [.[][]|select(.request_token==$token)]|length>0 and (.[0]|issued) and all(.[]; issued)' "$case_dir/commits.jsonl" >/dev/null
     [[ $(sql "SELECT decision_origin='$origin' AND response_state='$issued' AND proven_gone_at IS NULL FROM permission_requests WHERE request_token='$token';") == 1 ]]
     ad get-permission --request-token "$token" | jq -e '.decision=="allow" and .proven_gone_at==null' >/dev/null
     raw_after=$(raw_size)
     if relay_decide "$token" allow >"$case_dir/retry" 2>"$case_dir/retry-error"; then exit 1; fi
     jq -e '.err_name=="ErrAlreadyDecided" or .err_name=="ErrRelayModeOff" or .err_name=="ErrNoOpenPermissionRequest"' "$case_dir/retry-error" >/dev/null
     [[ $(raw_size) == "$raw_after" ]]
    fi
    relay_receipt t2.jf1.cr.11 '["origin-claim-issue","tool-plan-relay-native","first-byte-independent-commit","closed-no-authority","closed-processing-barrier","exact-shutdown-suffix","raw-crash-reopen-nonreplay","positive-control"]' >> "$receipts"
   )
  done
 done
done
jq -s '{case:"t2.jf1.cr.11",status:"pass",checks:["origin-claim-issue","tool-plan-relay-native","first-byte-independent-commit","closed-no-authority","closed-processing-barrier","exact-shutdown-suffix","raw-crash-reopen-nonreplay","positive-control"],authenticated_cursor:false,facts:{runs:map(.facts)}}' "$receipts"
```
