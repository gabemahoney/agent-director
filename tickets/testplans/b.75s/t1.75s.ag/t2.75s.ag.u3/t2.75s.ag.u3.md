# agent-23: Claude to Codex reuse hides history and records while preserving its AD child

Maps to SRD SR-6.8.9 Docker agent-23 and Subtask `t3.9r1.dn.ro.o8`.

## Setup

The driver runs `db-reset.sh`. Set both `[launch] claude_path` and
`codex_path` to existing stand-ins. Use unique parent/child ids, isolated
cwd, `CLAUDE_CONFIG_DIR`, transcript and per-case `CODEX_HOME`. Save and
restore config; cleanup ends recorded sessions and removes case files.

## Steps

1. Spawn Claude, seed and report S1 through `pane-hook.sh`, report an
   unnamed `PreToolUse(Agent)` launch request followed by SubagentStart
   with the same type, and verify its one confirmed visible record.
2. Spawn an AD-agent child with `AGENT_DIRECTOR_INSTANCE_ID` set to the
   parent. Verify `list --parent` and save the child's complete store row.
3. Report Claude SessionEnd, kill its recorded tmux session and wait at
   most 5 s for recorded processes to exit, as resume-9 does. Save its life
   and complete old-life subagent entry after the end.
4. Reuse the parent as Codex with its per-case `CODEX_HOME`, then report
   Codex SessionStart through `pane-hook.sh --agent codex`.
5. Compare get, list-subagents, child listing/store row and old-life entry.

## Pass criteria

All launches/hooks exit 0. The parent life advances exactly once; get
reports `agent` `codex`, `waiting`, the new Codex session and
`prior_sessions` `[]`. `list-subagents` is exactly
`{"tracked":false,"subagents":[]}` (key order ignored). `list --parent`
still lists exactly the child, whose complete store row is unchanged.
The old entry remains byte-for-byte equal as SQL JSON under the previous
life; S1's transcript remains present. No real agents are run.

## Script

```bash
set -euo pipefail
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY CODEX_HOME CLAUDE_CONFIG_DIR AD_CODEX_HOOK AD_CODEX_RECORD_DIR AGENT_DIRECTOR_INSTANCE_ID
for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
    tmux set-environment -g -u "$v" 2>/dev/null || true
done
db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
work="$(mktemp -d /tmp/agent-23-XXXXXX)"
ids=()
had_config=false
fail() { echo "FAIL: $*"; exit 1; }
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }
cleanup() {
    local x sock pane target
    for x in "${ids[@]}"; do
        sock="$(col "$x" "IFNULL(tmux_socket,'')" 2>/dev/null || true)"
        pane="$(col "$x" "IFNULL(pane_id,'')" 2>/dev/null || true)"
        target="$(timeout 5s tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
        [ -z "$target" ] || timeout 5s tmux ${sock:+-S "$sock"} kill-session -t "$target" 2>/dev/null || true
        rm -rf "/tmp/stand-in-codex/$x"
    done
    if $had_config; then cp -p "$work/config.before" "$config"; else rm -f "$config"; fi
    rm -rf "$work"
}
trap cleanup EXIT
if [ -e "$config" ]; then had_config=true; cp -p "$config" "$work/config.before"; fi
mkdir -p "$HOME/.config/agent-director" "$work/cwd" "$work/claude"
printf '[launch]\nclaude_path = "/opt/driver/stand-in-claude/claude"\ncodex_path = "/opt/driver/stand-in-codex/codex"\n' > "$config"
uuid="$(cat /proc/sys/kernel/random/uuid)"
# All hook payloads are typed into the row's recorded pane, never this shell.
fire() {
    local x="$1" agent="$2" event="$3" session="$4" transcript="$5" extra='{}' body out
    [ $# -lt 6 ] || extra="$6"
    body="$(jq -cn --arg e "$event" --arg s "$session" --arg t "$transcript" --arg c "$work/cwd" --argjson x "$extra" \
        '{hook_event_name:$e,session_id:$s,transcript_path:$t,cwd:$c} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent "$agent" "$x" "$body")" || fail "$event of $x failed"
    [ -z "$out" ] || fail "$event of $x printed [$out]; want empty stdout"
}
spawn_row() {
    local x="$1" agent="$2" name="$3" out
    shift 3
    ids+=("$x")
    out="$(agent-director spawn --agent "$agent" --instance-id "$x" --cwd "$work/cwd" --tmux-session-name "$name" \
        --extra-env "CODEX_HOME=$codex_home" --extra-env "CLAUDE_CONFIG_DIR=$work/claude" "$@")" || fail "spawn $agent $x failed"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed wrong id: $out"
    spawn_out="$out"
    local pane_pid deadline
    pane_pid="$(col "$x" pane_pid)"
    deadline=$(( $(date +%s) + 10 ))
    until [ "$(cat /proc/"$pane_pid"/comm 2>/dev/null || true)" = bash ]; do
        [ "$(date +%s)" -lt "$deadline" ] || fail "$x stand-in did not become interactive bash within 10 s"
        sleep 0.1
    done
    for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
        if grep -qz "^$v=" "/proc/$pane_pid/environ"; then fail "credentials reached $x pane"; fi
    done
    if [ "$agent" = codex ]; then
        until [ -s "/tmp/stand-in-codex/$x/pid" ]; do
            [ "$(date +%s)" -lt "$deadline" ] || fail "no Codex record for $x within 10 s"
            sleep 0.1
        done
        [ "$(cat /tmp/stand-in-codex/"$x"/pid)" = "$pane_pid" ] || fail "Codex record differs from pane pid"
    fi
}
end_and_wait() {
    local id="$1" target="$2" rec rpid rpane rserver rsock n left p end
    rec="$(/opt/driver/sql.sh -readonly "$db" "SELECT IFNULL(pid,'') || '|' || IFNULL(pane_pid,'') || '|' || IFNULL(tmux_server_pid,'') || '|' || IFNULL(tmux_socket,'') FROM spawns WHERE instance_id = '$id';")"
    IFS='|' read -r rpid rpane rserver rsock <<<"$rec"
    [[ "$rpid|$rpane|$rserver" =~ ^[0-9]+\|[0-9]+\|[0-9]+$ ]] || fail "$id records pid|pane_pid|tmux_server_pid [$rpid|$rpane|$rserver]; want all set"
    n="$(tmux ${rsock:+-S "$rsock"} list-sessions 2>/dev/null | wc -l)"
    timeout 5s tmux ${rsock:+-S "$rsock"} kill-session -t "$target" || fail "kill-session -t $target of $id failed"
    set -- "$rpid" "$rpane"
    [ "$n" -gt 1 ] || set -- "$@" "$rserver"
    end=$(( $(date +%s%N) + 5000000000 ))
    while :; do
        left=""
        for p in "$@"; do [ ! -e "/proc/$p" ] || left+=" $p"; done
        [ -n "$left" ] || break
        [ "$(date +%s%N)" -lt "$end" ] || fail "$id's recorded process(es)$left still running 5 s after its session $target ended"
        sleep 0.1
    done
    [ "$n" -le 1 ] || [ -e "/proc/$rserver" ] || fail "tmux server $rserver of $id ended although it held $((n - 1)) other session(s)"
}
finish_claude() {
    local x="$1" session="$2" transcript="$3" sock pane target
    fire "$x" claude SessionEnd "$session" "$transcript" '{"reason":"logout"}'
    [ "$(col "$x" state)" = ended ] || fail "Claude SessionEnd did not end $x"
    sock="$(col "$x" "IFNULL(tmux_socket,'')")"
    pane="$(col "$x" pane_id)"
    target="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}')"
    end_and_wait "$x" "$target"
}

id="ag23-$uuid"
child="ag23-child-$uuid"
codex_home="$work/codex"
mkdir -p "$codex_home"
s1="ag23-s1-$uuid"
s1_file="$work/claude/projects/case/$s1.jsonl"
mkdir -p "$(dirname "$s1_file")"
printf '{}\n' > "$s1_file"
spawn_row "$id" claude cd-ag23-parent
fire "$id" claude SessionStart "$s1" "$s1_file" '{"source":"startup"}'
[ "$(col "$id" state)" = waiting ] || fail "Claude did not report waiting"
# A start alone is an unconfirmed observation, excluded by list-subagents.
# Its matching unnamed Agent launch request lets the start confirm the record.
fire "$id" claude PreToolUse "$s1" "$s1_file" '{"tool_name":"Agent","tool_use_id":"retained-launch","tool_input":{"subagent_type":"worker"}}'
fire "$id" claude SubagentStart "$s1" "$s1_file" '{"agent_id":"retained-subagent","agent_type":"worker"}'
result="$(agent-director list-subagents --instance-id "$id")" || fail "setup list-subagents failed"
jq -e '.tracked == true and (.subagents | length) == 1 and .subagents[0].subagent_id == "retained-subagent"' <<<"$result" >/dev/null \
    || fail "setup did not make one visible Claude record: $result"
# AD's child relation is independent of subagent records and lives.
AGENT_DIRECTOR_INSTANCE_ID="$id" spawn_row "$child" claude cd-ag23-child
[ "$(col "$child" parent_id)" = "$id" ] || fail "AD child did not inherit parent id"
child_before="$(/opt/driver/sql.sh -readonly -json "$db" "SELECT * FROM spawns WHERE instance_id = '$child';" | jq -cS .)"
listed_before="$(agent-director list --parent "$id" | jq -c '.spawns | map(.instance_id)')"
[ "$listed_before" = "[\"$child\"]" ] || fail "setup list --parent: $listed_before"
finish_claude "$id" "$s1" "$s1_file"
life0="$(col "$id" life_number)"
old_entry="$(/opt/driver/sql.sh -readonly -json "$db" "SELECT * FROM subagents WHERE instance_id = '$id' AND life_number = $life0 AND subagent_id = 'retained-subagent';" | jq -cS .)"
[ "$(jq length <<<"$old_entry")" = 1 ] || fail "setup needs one stored entry under old life"
spawn_row "$id" codex cd-ag23-parent --reuse-finished
[ "$(col "$id" life_number)" -eq "$((life0 + 1))" ] || fail "Claude to Codex switch did not advance life once"
cs="$(cat /proc/sys/kernel/random/uuid)"
rollout="$codex_home/sessions/2026/10/13/rollout-2026-10-13T10-00-00-$cs.jsonl"
mkdir -p "$(dirname "$rollout")"
printf '{"type":"session_meta"}\n' > "$rollout"
fire "$id" codex SessionStart "$cs" "$rollout" '{"source":"startup"}'
row="$(agent-director get --instance-id "$id")"
jq -e --arg sid "$cs" '.agent == "codex" and .state == "waiting" and .session_id == $sid and .prior_sessions == []' <<<"$row" >/dev/null \
    || fail "Codex get did not hide prior life: $row"
result="$(agent-director list-subagents --instance-id "$id" | jq -cS .)"
[ "$result" = '{"subagents":[],"tracked":false}' ] || fail "Codex list-subagents: $result"
listed_after="$(agent-director list --parent "$id" | jq -c '.spawns | map(.instance_id)')"
[ "$listed_after" = "$listed_before" ] || fail "switch hid AD child: $listed_after"
child_after="$(/opt/driver/sql.sh -readonly -json "$db" "SELECT * FROM spawns WHERE instance_id = '$child';" | jq -cS .)"
[ "$child_after" = "$child_before" ] || fail "switch changed AD child row"
retained="$(/opt/driver/sql.sh -readonly -json "$db" "SELECT * FROM subagents WHERE instance_id = '$id' AND life_number = $life0 AND subagent_id = 'retained-subagent';" | jq -cS .)"
[ "$retained" = "$old_entry" ] || fail "switch changed or removed old-life entry"
[ -f "$s1_file" ] || fail "switch removed original Claude transcript"
echo "agent-23 ok: Claude session/record/child -> Codex report-in; life +1, get agent codex and empty history, exact untracked empty, child row unchanged/listed, old-life entry retained"
```
