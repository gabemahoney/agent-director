# agent-14: pause on a codex row types /exit; the end report and the gone process move it to ended

Maps to SRD SR-6.5, SR-7.2, SR-14.5 and SR-22.9 of Epic 13 (`b.ssz`): a
Codex agent that exits (`/exit`) sends `SessionEnd` and the process ends. The
`SessionEnd` only records the end report (`end_reported_at`); the row keeps
its state. `pause` types `/exit` into the pane, and once the agent process is
gone and the end report is in, it moves the row to `ended` itself
(`find-missing` does the same for a row nobody paused). That move is
written as one `ad.find_missing.tick` with `reconciliation_reason`
`proc_absent` and `new_state` `ended`.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex`, named by `[launch] codex_path`.
Typing `/exit` at its pane makes the stand-in fire a Codex `SessionEnd`
(reason `other`) through `agent-director hook --agent codex`, then exit; it
sends nothing on a signal.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"`, makes a
cwd and a temporary Codex folder holding the rollout the payloads name, and
removes them, the config and the stand-in's record directory on exit.

## Steps

1. Write the config; make the cwd and the Codex folder with its rollout.
2. Spawn a codex row (`--extra-env CODEX_HOME=<folder>`), type a
   `SessionStart` with `pane-hook.sh --agent codex`; the row is `waiting`.
3. `agent-director pause --instance-id <id>`.
4. `get` the row; read its `end_reported_at` from the store and the
   stand-in's `exit-hook.exit`.
5. Read the trail's `ad.find_missing.tick` records for the row.

## Pass criteria

- Step 3: exit 0 and `{}` on stdout.
- Step 4: `state` `ended`, `ended_at` set; `end_reported_at` is a positive
  integer; the stand-in's SessionEnd hook exited `0`; the row's pane is gone
  from tmux (the pane process exited).
- Step 5: exactly one tick for the row, with `prior_state` `waiting`,
  `new_state` `ended`, `reconciliation_reason` `proc_absent` and `source`
  `ad_find_missing`; none with `new_state` `missing`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# fire ID EVENT [EXTRA_JSON]: type the Codex hook EVENT, for the session $sid and
# its rollout $rollout, into the row's pane; the hook must exit 0, print nothing.
fire() {
    local x="$1" extra='{}' body out
    [ $# -lt 3 ] || extra="$3"
    body="$(jq -cn --arg e "$2" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$x" "$body")" || fail "the $2 hook of $x exited non-zero"
    [ -z "$out" ] || fail "the $2 hook of $x printed [$out]; want nothing"
}

# make_rollout SID: the rollout file Codex would keep for session SID in the
# Codex folder; prints its path.
make_rollout() {
    local p="$codex_home/sessions/2026/10/12/rollout-2026-10-12T10-00-00-$1.jsonl"
    mkdir -p "$(dirname "$p")"
    printf '{"type":"session_meta"}\n' > "$p"
    printf '%s' "$p"
}

id="ag-pz-$uuid"
sid="$(cat /proc/sys/kernel/random/uuid)"
rollout=""

# Step 1: the config, the cwd, and a Codex folder with the rollout.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-pz-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
codex_home="$(mktemp -d)"
rollout="$(make_rollout "$sid")"

# Step 2: a waiting codex row.
spawn_codex "$id" cd-ag-pz --extra-env "CODEX_HOME=$codex_home"
fire "$id" SessionStart '{"source":"startup"}'
wait_state "$id" waiting
[ -z "$(col "$id" "IFNULL(end_reported_at, '')")" ] || fail "end_reported_at is set before any SessionEnd"
sock="$(col "$id" "IFNULL(tmux_socket, '')")"
pane="$(col "$id" "IFNULL(pane_id, '')")"
[ -n "$pane" ] || fail "the row records no pane_id"

# Step 3: pause types /exit and waits for ended.
set +e
out="$(agent-director pause --instance-id "$id" 2>&1)"
rc=$?
set -e
[ "$rc" -eq 0 ] || fail "pause exited $rc; want 0: $out"
[ "$(jq -c . <<<"$out")" = "{}" ] || fail "pause printed [$out]; want {}"

# Step 4: ended, with the end report in and the pane gone.
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '.state' <<<"$row")" = "ended" ] || fail "state is [$(jq -r '.state' <<<"$row")] after pause; want ended"
[ -n "$(jq -r '.ended_at // ""' <<<"$row")" ] || fail "ended_at is not set: $row"
reported="$(col "$id" "IFNULL(end_reported_at, '')")"
[[ "$reported" =~ ^[0-9]+$ ]] && [ "$reported" -gt 0 ] || fail "end_reported_at is [$reported]; want a positive integer"
[ "$(cat "$rec_root/$id/exit-hook.exit")" = "0" ] || fail "the stand-in's SessionEnd hook exited [$(cat "$rec_root/$id/exit-hook.exit")]; want 0"
if tmux ${sock:+-S "$sock"} list-panes -a -F '#{pane_id}' 2>/dev/null | grep -qx "$pane"; then fail "pane $pane still exists after pause"; fi

# Step 5: the one move to ended, as a find-missing tick.
[ -f "$trail" ] || fail "no trail file at $trail"
ticks="$(jq -c --arg x "$id" 'select(.event == "ad.find_missing.tick" and .instance_id == $x)' "$trail")"
[ "$(printf '%s' "$ticks" | grep -c . || true)" -eq 1 ] || fail "want exactly one ad.find_missing.tick for $id: [$ticks]"
got="$(jq -r '[.prior_state, .new_state, .reconciliation_reason, .source] | join("|")' <<<"$ticks")"
[ "$got" = "waiting|ended|proc_absent|ad_find_missing" ] || fail "tick [$got]; want [waiting|ended|proc_absent|ad_find_missing]: $ticks"

echo "agent-14 ok: pause on a codex row: /exit -> SessionEnd + process gone -> ended; end_reported_at set; one tick prior waiting, new ended, proc_absent"
```
