# agent-12: a forking codex launcher on the PATH: its hook is ignored (pid_mismatch) and the row ends up unreported

Maps to SRD SR-6.1 and SR-14.5 of Epic 13 (`b.ssz`): a hook moves
a row only when the hook's parent is the row's pane process. Where the
`codex` on the PATH is a launcher that runs Codex as a child (an npm shim
or a version manager's wrapper does), the Codex process is the hook's
parent and the launcher, the pane process, its grandparent. The hook is
ignored with `pid_mismatch`; because it is a `SessionStart` of a pending
row whose pane process is the parent's parent, one
`ad.hook.pane_is_grandparent` record advises naming the native program in
`[launch] codex_path`. The row stays `pending` and past the grace period
`find-missing` notes it `unreported`.

Agent and hooks (collector "Driver mode"): `[launch] codex_path` is unset.
The spawn runs with `PATH=/opt/driver/forking-codex:$PATH`; that
`codex` runs the Codex stand-in as a child (and `exit $?` keeps it from
exec-ing it). The stand-in is an interactive bash the case types the
`SessionStart` into with `/opt/driver/pane-hook.sh --agent codex`.

## Setup

Per-case DB-reset. The case writes a config holding only `[tmux]`
`pending_grace_seconds = 30` (the shortest the config accepts), makes a cwd
and removes both and the stand-in's record directory on exit.

## Steps

1. Write the config; make the cwd.
2. `PATH=/opt/driver/forking-codex:$PATH agent-director spawn --agent codex
   --cwd <cwd> --instance-id <id> --tmux-session-name cd-ag-fk`; wait for the
   stand-in's record.
3. Check the process tree: the pane process is the launcher (`codex`) and the
   stand-in's pid is its child.
4. Type a `SessionStart` (source `startup`) with `pane-hook.sh --agent codex`.
5. Read the trail's `ad.hook.ignored` and `ad.hook.pane_is_grandparent`
   records for the row; `get` the row.
6. Run `find-missing` every 3 s until `get` shows `liveness_note`
   `unreported`, for at most 75 s.

## Pass criteria

- Step 2: spawn exits 0; the row is `pending`, agent `codex`.
- Step 3: the row's `pane_pid` runs a process whose command is `codex`;
  the stand-in's recorded pid differs from it and its parent is `pane_pid`.
- Step 4: the hook exits 0 and prints nothing (a refused hook fails open).
- Step 5: the row is `pending` with no session id. Exactly one
  `ad.hook.ignored` for the row, with `hook_event` `SessionStart`, `reason`
  `pid_mismatch`, `row_pane_pid` the row's `pane_pid`, and `parent_pid` not
  the `pane_pid`. Exactly one `ad.hook.pane_is_grandparent`, with `pane_pid`
  the row's, `pane_command` `codex`, `parent_pid` the ignored record's, and
  an `advice` that names the `codex` launcher on the PATH, the shell-form
  hook and `[launch] codex_path`, and never `claude_path`.
- Step 6: every run has `count` 0 and empty `ids`; `unreported` shows within
  75 s and the row is still `pending`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# fire ID EVENT [EXTRA_JSON]: type the Codex hook EVENT, for the session $sid and
# its rollout $rollout, into the row's pane; the hook must exit 0, print nothing.
fire() {
    local x="$1" extra='{}' body out
    [ $# -lt 3 ] || extra="$3"
    body="$(jq -cn --arg e "$2" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$x" "$body")" || fail "the $2 hook of $x exited non-zero"
    [ -z "$out" ] || fail "the $2 hook of $x printed [$out]; want nothing"
}

# wait_unreported ID: find-missing, run every 3 s, ends by noting the pending
# row ID unreported, within 75 s of now; every run lists nothing (a pending row
# is never marked missing for being unreported).
wait_unreported() {
    local x="$1" end res note=""
    end=$(( $(date +%s) + 75 ))
    while :; do
        res="$(agent-director find-missing)" || fail "find-missing exited non-zero"
        [ "$(jq -c '[.count, .ids, (.unverified_ids // [])]' <<<"$res")" = '[0,[],[]]' ] \
            || fail "find-missing result [$res]; want count 0 and no ids (the row is only noted)"
        note="$(agent-director get --instance-id "$x" | jq -r '.liveness_note // ""')"
        [ "$note" != "unreported" ] || return 0
        [ -z "$note" ] || fail "liveness_note is [$note]; want none until unreported"
        [ "$(date +%s)" -lt "$end" ] || fail "row $x was not noted unreported within 75 s (pending_grace_seconds = 30)"
        sleep 3
    done
}

id="ag-fk-$uuid"
sid="$(cat /proc/sys/kernel/random/uuid)"
rollout="/tmp/ag-fk-$uuid-rollout.jsonl"

# Step 1: the config with the shortest pending grace (and no codex_path), and the cwd.
[ -x /opt/driver/forking-codex/codex ] || fail "/opt/driver/forking-codex/codex is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[tmux]\npending_grace_seconds = 30\n' > "$config"
cwd="$(mktemp -d /tmp/ag-fk-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"

# Step 2: spawn through the forking launcher on the PATH.
ids+=("$id")
out="$(PATH="/opt/driver/forking-codex:$PATH" agent-director spawn --agent codex --cwd "$cwd" --instance-id "$id" --tmux-session-name cd-ag-fk)" \
    || fail "spawn exited non-zero"
[ "$(jq -r '.instance_id' <<<"$out")" = "$id" ] || fail "spawn echoed [$out]; want id $id"
wait_record "$id"
[ "$(agent-director get --instance-id "$id" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] || fail "row is not codex|pending"

# Step 3: the pane process is the launcher; the stand-in is its child.
pane_pid="$(col "$id" pane_pid)"
[ -n "$pane_pid" ] || fail "the row records no pane_pid"
[ "$(ps -o comm= -p "$pane_pid" | tr -d ' ')" = "codex" ] || fail "the pane process $pane_pid runs [$(ps -o comm= -p "$pane_pid")]; want the launcher codex"
rec_pid="$(cat "$rec_root/$id/pid")"
[ "$rec_pid" != "$pane_pid" ] || fail "the stand-in's pid is the pane process $pane_pid; want the launcher's child"
[ "$(ps -o ppid= -p "$rec_pid" | tr -d ' ')" = "$pane_pid" ] || fail "the stand-in $rec_pid has parent [$(ps -o ppid= -p "$rec_pid")]; want the pane process $pane_pid"

# Step 4: a SessionStart typed into the stand-in: its parent is the stand-in, the launcher is its grandparent.
fire "$id" SessionStart '{"source":"startup"}'

# Step 5: ignored with pid_mismatch, one grandparent record, row unchanged.
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '.state' <<<"$row")" = "pending" ] || fail "state is [$(jq -r '.state' <<<"$row")] after the refused hook; want pending"
[ -z "$(jq -r '.session_id // ""' <<<"$row")" ] || fail "session_id is set after a refused hook: $row"
[ -f "$trail" ] || fail "no trail file at $trail"
ignored="$(jq -c --arg x "$id" 'select(.event == "ad.hook.ignored" and .instance_id == $x)' "$trail")"
[ "$(printf '%s' "$ignored" | grep -c . || true)" -eq 1 ] || fail "want exactly one ad.hook.ignored for $id: [$ignored]"
got="$(jq -r '[.hook_event, .reason, (.row_pane_pid | tostring)] | join("|")' <<<"$ignored")"
[ "$got" = "SessionStart|pid_mismatch|$pane_pid" ] || fail "ad.hook.ignored [$got]; want [SessionStart|pid_mismatch|$pane_pid]: $ignored"
parent_pid="$(jq -r '.parent_pid | tostring' <<<"$ignored")"
[ "$parent_pid" != "$pane_pid" ] || fail "ad.hook.ignored parent_pid is the pane process $pane_pid; want the launcher's child"
gp="$(jq -c --arg x "$id" 'select(.event == "ad.hook.pane_is_grandparent" and .instance_id == $x)' "$trail")"
[ "$(printf '%s' "$gp" | grep -c . || true)" -eq 1 ] || fail "want exactly one ad.hook.pane_is_grandparent for $id: [$gp]"
got="$(jq -r '[(.pane_pid | tostring), .pane_command, (.parent_pid | tostring)] | join("|")' <<<"$gp")"
[ "$got" = "$pane_pid|codex|$parent_pid" ] || fail "ad.hook.pane_is_grandparent [$got]; want [$pane_pid|codex|$parent_pid]: $gp"
advice="$(jq -r '.advice' <<<"$gp")"
for want in \
    "the codex on PATH is a launcher that runs Codex as a child instead of exec-ing it" \
    "Codex ran the hook through a shell instead of in exec form" \
    "[launch] codex_path"; do
    case "$advice" in *"$want"*) ;; *) fail "the advice [$advice] does not contain [$want]" ;; esac
done
case "$advice" in *claude_path*) fail "a codex row's advice names claude_path: [$advice]" ;; esac

# Step 6: past the grace period the row is noted unreported, still pending.
wait_unreported "$id"
[ "$(state "$id")" = "pending" ] || fail "state is [$(state "$id")] after the note; want pending"

echo "agent-12 ok: forking codex launcher on PATH: SessionStart ignored as pid_mismatch with one pane_is_grandparent naming codex; row pending, then noted unreported"
```
