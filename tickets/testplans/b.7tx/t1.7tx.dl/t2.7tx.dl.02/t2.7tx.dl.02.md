# Test case: configured OpenCode launch

Maps to t3.m6h.q8.37.uc Work 7. The shell driver runs the built CLI with
`test/driver/stand-in-codex/codex`; no native agent or account is used.

## Steps and pass criteria

Load the opt-in and launch path from the config file. With a synthetic host
key, observe a pending pane-only row and exactly two native arguments.
With the key absent from the invoking process, observe ErrHostCredentialMissing.

```bash
set -euo pipefail
case_dir=$(mktemp -d /tmp/opencode-lane.XXXXXX)
export TMUX_TMPDIR="$case_dir/tmux"
export AD_CODEX_RECORD_DIR="$case_dir/records"
unset AD_CODEX_HOOK
mkdir -p "$TMUX_TMPDIR" "$AD_CODEX_RECORD_DIR" "$case_dir/project"
cfg="$HOME/.config/agent-director/config.toml"
cleanup() {
    tmux kill-server 2>/dev/null || true
    rm -f "$cfg"
    rm -rf "$case_dir"
}
trap cleanup EXIT
cat >"$cfg" <<EOF
[experimental]
opencode = true
[launch]
opencode_path = "/opt/driver/stand-in-codex/codex"
EOF
export OPENROUTER_API_KEY=synthetic-docker-lane-key
agent-director spawn --agent opencode --cwd "$case_dir/project" -- --model openrouter/provider/model >"$case_dir/spawn"
id=$(jq -er .instance_id "$case_dir/spawn")
record="$AD_CODEX_RECORD_DIR/$id"
deadline=$((SECONDS + 45))
until test -s "$record/pid"; do
    (( SECONDS < deadline )) || { printf 'stand-in did not record argv\n' >&2; exit 1; }
    sleep 0.1
done
agent-director get --instance-id "$id" >"$case_dir/get"
jq -e '.agent == "opencode" and .state == "pending" and .integration_mode == "pane_only"' "$case_dir/get" >/dev/null
mapfile -d '' -t argv <"$record/argv"
test "${#argv[@]}" = 2
test "${argv[0]}" = --model
test "${argv[1]}" = openrouter/provider/model
unset OPENROUTER_API_KEY
rc=0
agent-director spawn --agent opencode --cwd "$case_dir/project" -- --model openrouter/provider/model >"$case_dir/out" 2>"$case_dir/error" || rc=$?
test "$rc" = 1
jq -e '.err_name == "ErrHostCredentialMissing"' "$case_dir/error" >/dev/null
printf 'OpenCode configured stand-in launch passed\n'
```
