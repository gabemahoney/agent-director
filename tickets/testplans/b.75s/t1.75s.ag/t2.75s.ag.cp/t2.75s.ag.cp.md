# agent-5: a permission list on a codex spawn is refused

Maps to SRD SR-14.5 and the permissions rule of Epic 13 (`b.ssz`): a
codex agent takes no allow, deny or ask entries. A `spawn --agent codex`
that carries one, on the call or inherited from its template, is
refused with `ErrPermissionsUnsupported` before anything is written.

## Setup

Per-case DB-reset: a fresh store, no tmux server, no templates
directory. The template variant is a template file the case writes by
hand: `agent = "codex"` with a `[permissions]` table (`make-template`
refuses to write one, which agent-6 pins).

## Steps

1. `agent-director spawn --agent codex --cwd /tmp --allow 'Bash(ls)'
   --tmux-session-name cd-ag-cp1`, then the same with `--deny` and with
   `--ask`.
2. Write `~/.agent-director/templates/ag-cp-tpl.toml` by hand with
   `agent = "codex"` and `[permissions]` `allow = ["Bash(ls)"]`.
3. `agent-director spawn --template ag-cp-tpl --cwd /tmp
   --tmux-session-name cd-ag-cp2` (no `--agent`: the template's codex
   applies).
4. Check nothing was written.

## Pass criteria

- Every refused call exits 1 with `err_name` `ErrPermissionsUnsupported`
  and an `err_description` containing `a codex agent takes no allow, deny
  or ask entries`.
- After each: `list` shows no rows, `spawns` holds none, and the tmux
  session ids are unchanged; the templates directory is empty before step
  2 and holds only `ag-cp-tpl.toml` after it.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN

db="$HOME/.agent-director/state.db"
tpl_dir="$HOME/.agent-director/templates"
# Should a refused call ever launch, it gets the stand-in, never Claude Code.
standin_path="/opt/driver/stand-in-claude:$PATH"
fail() { echo "FAIL: $*"; exit 1; }

session_ids() { tmux ls -F '#{session_id}' 2>/dev/null || true; }
session_names() { tmux ls -F '#{session_name}' 2>/dev/null || true; }
tpl_files() { ls -A "$tpl_dir" 2>/dev/null || true; }
# The templates directory is shared by every case in this epic: start from none.
rm -rf "$tpl_dir"
sessions_before="$(session_ids)"
cleanup() {
    local sid
    for sid in $(session_ids); do
        case " $(printf '%s ' $sessions_before)" in
            *" $sid "*) ;;
            *) tmux kill-session -t "$sid" 2>/dev/null || true ;;
        esac
    done
    # Leave no template behind for the next case.
    rm -rf "$tpl_dir"
}
trap cleanup EXIT

# refuse WANT_ERR_NAME WANT_TEXT ARGS...: agent-director ARGS... must exit 1
# with a JSON envelope on stderr whose err_name is WANT_ERR_NAME and whose
# err_description contains WANT_TEXT.
refuse() {
    local want_name="$1" want_text="$2" err code envelope got desc
    shift 2
    set +e
    err="$(PATH="$standin_path" agent-director "$@" 2>&1 >/dev/null)"
    code=$?
    set -e
    [ "$code" -eq 1 ] || fail "agent-director $* exited $code; want 1: $err"
    envelope="$(printf '%s\n' "$err" | awk '/^\{/{ln=$0} END{print ln}')"
    got="$(jq -r '.err_name' <<<"$envelope")"
    [ "$got" = "$want_name" ] || fail "agent-director $*: err_name=[$got]; want $want_name; stderr: $err"
    desc="$(jq -r '.err_description' <<<"$envelope")"
    case "$desc" in
        *"$want_text"*) ;;
        *) fail "agent-director $*: err_description=[$desc]; want it to contain [$want_text]" ;;
    esac
}

# assert_untouched TEMPLATE_FILES: nothing the refused calls could write exists:
# no row (list and the store), no new tmux session, and the templates dir
# holds exactly TEMPLATE_FILES (what the case itself wrote, "" for none).
assert_untouched() {
    local want_tpl="$1" n_list n_db after
    n_list="$(agent-director list | jq '.spawns | length')"
    [ "$n_list" = "0" ] || fail "list shows $n_list row(s); want 0"
    n_db="$(/opt/driver/sql.sh -readonly "$db" 'SELECT COUNT(*) FROM spawns;')"
    [ "$n_db" = "0" ] || fail "the store holds $n_db spawns row(s); want 0"
    after="$(session_ids)"
    [ "$after" = "$sessions_before" ] || fail "tmux sessions changed; before=[$sessions_before] after=[$after]"
    [ "$(tpl_files)" = "$want_tpl" ] || fail "templates dir holds [$(tpl_files)]; want [$want_tpl]"
}

want_text="a codex agent takes no allow, deny or ask entries"

# Step 1: the entry on the call.
n=0
for flag in --allow --deny --ask; do
    n=$((n + 1))
    refuse ErrPermissionsUnsupported "$want_text" spawn --agent codex --cwd /tmp "$flag" 'Bash(ls)' --tmux-session-name "cd-ag-cp-f$n"
    assert_untouched ""
    if session_names | grep -qx "cd-ag-cp-f$n"; then fail "a tmux session named cd-ag-cp-f$n exists"; fi
done

# Step 2: a hand-written codex template that holds permissions.
mkdir -p "$tpl_dir"
cat > "$tpl_dir/ag-cp-tpl.toml" <<'TOML'
agent = "codex"

[permissions]
allow = ["Bash(ls)"]
TOML

# Steps 3-4: the entry inherited from the template.
refuse ErrPermissionsUnsupported "$want_text" spawn --template ag-cp-tpl --cwd /tmp --tmux-session-name cd-ag-cp2
assert_untouched "ag-cp-tpl.toml"
if session_names | grep -qx 'cd-ag-cp2'; then fail "a tmux session named cd-ag-cp2 exists"; fi

echo "agent-5 ok: codex spawn with --allow, --deny, --ask and with a template's permissions -> ErrPermissionsUnsupported; no row, no session"
```
