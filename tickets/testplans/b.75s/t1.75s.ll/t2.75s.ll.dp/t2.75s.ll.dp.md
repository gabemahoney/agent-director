# Test case: long launch process delivery

Maps to Bug b.xhh's expected result and pre-write refusal.

## Setup

The driver's DB reset supplies a fresh store. Use the existing recording
Codex stand-in for both ordinary agents; no provider or credentials are used.

## Steps

1. Spawn Claude with a 30 KB append-system-prompt value, Codex with a 30 KB
   developer_instructions config plus its normal composed hooks/settings,
   and an ordinary Claude launch with a 30 KB explicit environment value.
2. Repeat on reused and resumed finished rows. Seed those rows from small
   launches after killing and verifying their pane processes exited; store
   a session id and transcript for resume. Do not wait out pending grace.
3. Compare exact argv/env bytes written by each launched process.
4. Load an over-limit argument from a template, so the outer CLI can start;
   require ErrInvalidFlags and no row, session or stand-in receipt.

## Pass criteria

All nine long launches exit 0 and each actual process records the exact
30,000-byte value. The oversized launch is refused before a row/session is
created. Any failed assertion exits nonzero; every lane is checked even if
another fails.

```bash
set -euo pipefail
# b.xhh: assertions replace the repro kit's log-only B, C2 and D3 steps.
python3 - <<'PY'
import json
import os
from pathlib import Path
import subprocess
import tempfile
import time
import uuid

work = Path(tempfile.mkdtemp(prefix='long-launch-'))
cfg = Path.home() / '.config/agent-director'
cfg.mkdir(parents=True, exist_ok=True)
# Reuse the recording fixture, including its terminal/process behavior.
(cfg / 'config.toml').write_text('[launch]\nclaude_path = "/opt/driver/stand-in-codex/codex"\ncodex_path = "/opt/driver/stand-in-codex/codex"\n')
record = work / 'records'
os.environ['AD_CODEX_RECORD_DIR'] = str(record)
os.environ.pop('AD_CODEX_HOOK', None)
db = cfg / 'state.db'
payload = 'x' * 30000
failures = []

def require(ok, text):
    if not ok:
        raise AssertionError(text)

def ad(*args):
    return subprocess.run(['agent-director', *args], capture_output=True, text=True, timeout=35)

def sql(statement):
    return subprocess.check_output(['/opt/driver/sql.sh', str(db), statement], text=True).strip()

def quoted(value):
    return "'" + value.replace("'", "''") + "'"

def wait(predicate, message):
    deadline = time.monotonic() + 10
    value = predicate()
    while not value:
        require(time.monotonic() < deadline, message)
        time.sleep(.02)
        value = predicate()
    return value

def recorded_pid(path):
    try:
        pid = int(path.read_text())
    except (FileNotFoundError, ValueError):
        return None
    return pid if pid > 0 else None

def gone(pid):
    path = Path('/proc') / str(pid) / 'stat'
    try:
        return path.read_text().rsplit(')', 1)[1].split()[0] in ('Z', 'X')
    except FileNotFoundError:
        return True

def spawn(id, agent, args, extra, reuse=False, template=None):
    argv = ['spawn', '--agent', agent, '--cwd', str(work), '--instance-id', id, '--no-pre-trust']
    if reuse:
        argv.append('--reuse-finished')
    if template:
        argv += ['--template', template]
    for key, value in extra.items():
        argv += ['--extra-env', key + '=' + value]
    if args:
        argv += ['--', *args]
    return ad(*argv)

# b.ug8: keep the lane index inside id[:8] for distinct default session names.
n = 0
for shape in ('claude-arg', 'codex-arg', 'claude-env'):
    for verb in ('spawn', 'reuse', 'resume'):
        lane = shape + '/' + verb
        id = 'long-' + str(n) + '-' + uuid.uuid4().hex
        n += 1
        rec = record / id
        agent = 'codex' if shape == 'codex-arg' else 'claude'
        args = ['--config=developer_instructions=' + payload] if agent == 'codex' else ['--append-system-prompt', payload]
        extra = {}
        if shape == 'claude-env':
            args, extra = [], {'LONG_LAUNCH_VALUE': payload}
        try:
            if verb != 'spawn':
                control = spawn(id, agent, [], {})
                require(control.returncode == 0, lane + ': small control did not launch: ' + control.stderr[:400])
                pid = wait(lambda: recorded_pid(rec / 'pid'), lane + ': small process never recorded a valid positive PID')
                killed = ad('kill', '--instance-id', id)
                require(killed.returncode == 0, lane + ': control kill failed')
                wait(lambda: gone(pid), lane + ': control pane still alive')
                # A finished-row seed, after verified exit: no grace sleep and
                # no dependence on whether the killed stand-in emitted hooks.
                sid = str(uuid.uuid4())
                transcript = work / (sid + '.jsonl')
                transcript.write_text('{}\n')
                sql('UPDATE spawns SET state=\'ended\', settled_state=\'ended\', ended_at=datetime(\'now\'), '
                    'session_id=' + quoted(sid) + ', jsonl_path=' + quoted(str(transcript)) + ', '
                    'pid=NULL, proc_starttime=NULL, pane_id=NULL, pane_pid=NULL, pane_starttime=NULL '
                    'WHERE instance_id=' + quoted(id))
                for path in rec.iterdir():
                    path.unlink()
                if verb == 'resume':
                    sql('UPDATE spawns SET agent_args=' + quoted(json.dumps(args)) + ', extra_env=' + quoted(json.dumps(extra)) +
                        ' WHERE instance_id=' + quoted(id))
            result = ad('resume', '--instance-id', id) if verb == 'resume' else spawn(id, agent, args, extra, verb == 'reuse')
            require(result.returncode == 0, lane + ': launch exit=' + str(result.returncode) + ' ' + (result.stderr or result.stdout)[:500])
            wait(lambda: recorded_pid(rec / 'pid'), lane + ': actual process never recorded a valid positive PID')
            actual_args = (rec / 'argv').read_bytes().split(b'\0')[:-1]
            actual_env = (rec / 'env').read_bytes().split(b'\0')[:-1]
            expected = ('LONG_LAUNCH_VALUE=' + payload).encode() if extra else args[-1].encode()
            require(expected in (actual_env if extra else actual_args), lane + ': process did not receive exact payload')
            if agent == 'codex':
                require(any(b'hooks.SessionStart' in arg for arg in actual_args), lane + ': composed Codex hook settings absent')
            if verb == 'resume':
                require(sql('SELECT session_id FROM spawns WHERE instance_id=' + quoted(id)) == sid, lane + ': resume lost conversation id')
            print('ASSERT_PASS ' + lane + ' exact_payload_bytes=30000', flush=True)
        except (AssertionError, subprocess.SubprocessError) as error:
            failures.append(str(error))
            print('ASSERT_FAIL ' + str(error), flush=True)

# A file-backed template avoids the outer CLI's MAX_ARG_STRLEN limit.
id = 'too-long-' + uuid.uuid4().hex
templates = cfg / 'templates'
templates.mkdir(exist_ok=True)
(templates / 'long-refusal.toml').write_text('agent = "claude"\nagent_args = [' + json.dumps('x' * 131072) + ']\n')
try:
    result = spawn(id, 'claude', [], {}, template='long-refusal')
    require(result.returncode != 0, 'oversize: launch accepted')
    error = json.loads(result.stderr or result.stdout)
    require(error.get('err_name') == 'ErrInvalidFlags', 'oversize: want ErrInvalidFlags, got ' + str(error.get('err_name')))
    require(sql('SELECT count(*) FROM spawns WHERE instance_id=' + quoted(id)) == '0', 'oversize: row was written')
    sessions = subprocess.run(['tmux', 'list-sessions', '-F', '#{session_name}'], capture_output=True, text=True)
    require(id[:8] not in sessions.stdout, 'oversize: session was created')
    require(not (record / id / 'pid').exists(), 'oversize: process ran')
    print('ASSERT_PASS oversize refused before row/session', flush=True)
except (AssertionError, subprocess.SubprocessError) as error:
    failures.append(str(error))
    print('ASSERT_FAIL ' + str(error), flush=True)

require(not failures, str(len(failures)) + ' long-launch assertion(s) failed')
PY
```
