# Test case: operator rules through the built CLI

Maps to b.i4t Epic t1.ba6.3q criterion 2 and SR-6.1; Subtask t3.ba6.3q.u6.uu.

## Setup

Use the harness's fresh database, the recording Codex stand-in as the Droid
program, and a temporary synthetic Rule Set. No native installation or account
is used. The stand-in's opt-in mode records the native check independently
of launch identity, then consumes the overlay before its usual shell starts.
An invented Secret Path lets the real Guard self-check pass before each rules check.

## Steps

1. Configure rules, launch Droid, and wait for a FIFO receipt from the pane's
   shell. Compare the checker input to the raw object embedded in the overlay,
   the copy consumed by the pane, and the stored launch token.
2. Kill the first launch, move its launch clock past the pending grace period,
   and reconcile it. Change the rules and reuse its ID. Check the new rules
   and token through the same executable.
3. Make that executable return a nonzero check status on a reuse. Compare
   the database, launch files and pane records before and after the refusal.

## Pass criteria

Exact bytes, keys, modes and token-qualified IDs agree on both launches;
the checker has no launch identity or forbidden credentials and its private
copy is gone before spawn returns. Native refusal creates no launch and
preserves the old row/files. The test HOME's Factory settings stay unchanged.

```bash
set -euo pipefail
case_dir=$(mktemp -d /tmp/droid-rules.XXXXXX)
export TMUX_TMPDIR="$case_dir/tmux"
export TMPDIR="$case_dir/tmp"
export AD_CODEX_RECORD_DIR="$case_dir/panes"
export AD_DROID_RULES_RECORD_DIR="$case_dir/native"
mkdir -p "$TMUX_TMPDIR" "$TMPDIR" "$AD_CODEX_RECORD_DIR" "$AD_DROID_RULES_RECORD_DIR" "$case_dir/project"
printf '%s\n' "$case_dir/invented-vault" >"$case_dir/secret-paths"
cfg="$HOME/.config/agent-director/config.toml"
db="$HOME/.config/agent-director/state.db"
launch="$HOME/.config/agent-director/launch"
factory="$HOME/.factory"
test ! -e "$cfg"
test ! -e "$factory"
mkdir -p "$factory"
printf '{"fixture":"unchanged"}\n' >"$factory/settings.json"
/opt/driver/tree-snapshot.sh "$factory" >"$case_dir/factory-before"
cleanup() {
    tmux kill-server 2>/dev/null || true
    rm -f "$cfg"
    rm -rf "$launch" "$factory" "$case_dir"
}
trap cleanup EXIT
cat >"$cfg" <<EOF
[experimental]
droid = true
[launch]
droid_path = "/opt/driver/stand-in-codex/codex"
droid_rules_path = "$case_dir/rules.json"
droid_secret_paths_file = "$case_dir/secret-paths"
EOF
rules() {
    jq -cn --arg id "$1" '{version:1,rules:[{id:$id,decision:"allow",match:{prefix:["tail"]},tests:{match:["tail x"],noMatch:["head x"]}}]}' >"$case_dir/rules.json"
}
# Synthetic inherited credentials must be absent from the native checker.
export OPENROUTER_API_KEY=PRIVATE_HOST_CREDENTIAL
export FACTORY_API_KEY=PRIVATE_FACTORY_CREDENTIAL
export AWS_BEARER_TOKEN_BEDROCK=PRIVATE_BEDROCK_CREDENTIAL
export FACTORY_HOME_OVERRIDE="$case_dir/forbidden-home"
export AGENT_DIRECTOR_PRIVATE_TEST=PRIVATE_IDENTITY
mkfifo "$case_dir/ready"
exec 7<>"$case_dir/ready"
check_launch() {
    local model="$1" operator_id="$2" settings checked token raw prefix reply
    # The fixture is a known shell. Queue one receipt command and block on
    # its FIFO; reaching it proves the shared recorder and overlay read ended.
    pane=$( /opt/driver/sql.sh -readonly "$db" "SELECT pane_id FROM spawns WHERE instance_id='$id';" )
    tmux send-keys -t "$pane" "printf 'ready\\n' > '$case_dir/ready'" Enter
    IFS= read -r -t 15 reply <&7
    test "$reply" = ready
    record="$AD_CODEX_RECORD_DIR/$id"
    mapfile -d '' -t argv <"$record/argv"
    test "${#argv[@]}" = 2
    test "${argv[0]}" = --settings
    settings="${argv[1]}"
    test "$(stat -c %a "$settings")" = 600
    test "$(stat -c %a "$(dirname "$settings")")" = 700
    cmp "$settings" "$AD_DROID_RULES_RECORD_DIR/pane-settings"
    jq -e --arg model "$model" '
        keys_unsorted == ["model","permissionRules","hooks"] and .model == $model and
        (.hooks.PreToolUse[0].hooks[0].command | type == "string" and length > 0) and
        .hooks == {hooksDisabled:false,PreToolUse:[{matcher:"*",hooks:[{
            type:"command",command:.hooks.PreToolUse[0].hooks[0].command,timeout:60
        }]}]}' "$settings" >/dev/null
    checked="$AD_DROID_RULES_RECORD_DIR/check"
    test "$(cat "$checked/file-mode")" = 600
    test "$(cat "$checked/dir-mode")" = 700
    test ! -e "$(cat "$checked/path")"
    test ! -e "$(dirname "$(cat "$checked/path")")"
    test -z "$(find "$TMPDIR" -mindepth 1 -print -quit)"
    raw=$(cat "$settings")
    prefix="{\"model\":\"$model\",\"permissionRules\":"
    [[ "$raw" == "$prefix"*'}' ]]
    raw="${raw#"$prefix"}"
    printf '%s' "${raw%%,\"hooks\":*}" >"$case_dir/embedded"
    cmp "$checked/input" "$case_dir/embedded"
    token=$(/opt/driver/sql.sh -readonly "$db" "SELECT launch_token FROM spawns WHERE instance_id='$id';")
    jq --arg id "agent-director/$token/$operator_id" '.rules[0].id = $id' "$case_dir/rules.json" >"$case_dir/expected"
    jq -S . "$checked/input" >"$case_dir/actual"
    jq -S . "$case_dir/expected" >"$case_dir/normalized"
    cmp "$case_dir/actual" "$case_dir/normalized"
    tr '\0' '\n' <"$checked/env" >"$case_dir/check-env"
    if grep -Eq '^(AGENT_DIRECTOR_[^=]*|TMUX|TMUX_PANE|OPENROUTER_API_KEY|FACTORY_API_KEY|AWS_BEARER_TOKEN_BEDROCK|FACTORY_HOME_OVERRIDE)=' "$case_dir/check-env"; then
        echo 'checker inherited identity or credentials' >&2
        return 1
    fi
    tr '\0' '\n' <"$record/env" | grep -Fx "AGENT_DIRECTOR_INSTANCE_ID=$id"
    /opt/driver/tree-snapshot.sh "$factory" >"$case_dir/factory-after"
    cmp "$case_dir/factory-before" "$case_dir/factory-after"
    current_token="$token"
    current_settings="$settings"
}
finish() {
    agent-director kill --instance-id "$id" | jq -e '.kill_sent == true' >/dev/null
    # Kill has verified process exit. This case tests rules/reuse, not the
    # pending grace timer: age only that clock on the real, now-dead launch.
    /opt/driver/sql.sh "$db" "UPDATE spawns SET launch_started_at='2000-01-01T00:00:00Z' WHERE instance_id='$id';"
    agent-director sweep >/dev/null
    agent-director get --instance-id "$id" | jq -e '.state == "missing"' >/dev/null
}
rules first-rule
# Keep this private server alive between launches, avoiding its last-session
# shutdown racing the reuse (this anchor is removed by the cleanup trap).
tmux new-session -d -s rules-anchor
agent-director spawn --agent droid --cwd "$case_dir/project" -- --model first-model >"$case_dir/spawn"
id=$(jq -er '.instance_id' "$case_dir/spawn")
check_launch first-model first-rule
first_token="$current_token"
first_settings="$current_settings"
finish
mv "$AD_CODEX_RECORD_DIR/$id" "$case_dir/first-pane"
rules second-rule
agent-director spawn --agent droid --instance-id "$id" --reuse-finished --cwd "$case_dir/project" -- --model second-model >"$case_dir/reuse"
check_launch second-model second-rule
test "$current_token" != "$first_token"
test "$current_settings" != "$first_settings"
finish
/opt/driver/sql.sh -readonly "$db" .dump >"$case_dir/db-before"
/opt/driver/tree-snapshot.sh "$launch" >"$case_dir/launch-before"
/opt/driver/tree-snapshot.sh "$AD_CODEX_RECORD_DIR" >"$case_dir/panes-before"
export AD_DROID_RULES_CHECK_EXIT=1
rc=0
agent-director spawn --agent droid --instance-id "$id" --reuse-finished --cwd "$case_dir/project" -- --model refused-model >"$case_dir/out" 2>"$case_dir/error" || rc=$?
test "$rc" = 1
test ! -s "$case_dir/out"
jq -e '.err_name == "ErrPermissionRulesRefused" and (.err_description | contains("Droid\u0027s rules check failed with a nonzero exit status"))' "$case_dir/error" >/dev/null
if grep -Eq 'PRIVATE_NATIVE_OUTPUT|second-rule|tail x' "$case_dir/error"; then exit 1; fi
/opt/driver/sql.sh -readonly "$db" .dump >"$case_dir/db-after"
/opt/driver/tree-snapshot.sh "$launch" >"$case_dir/launch-after"
/opt/driver/tree-snapshot.sh "$AD_CODEX_RECORD_DIR" >"$case_dir/panes-after"
cmp "$case_dir/db-before" "$case_dir/db-after"
cmp "$case_dir/launch-before" "$case_dir/launch-after"
cmp "$case_dir/panes-before" "$case_dir/panes-after"
test ! -e "$(dirname "$(cat "$AD_DROID_RULES_RECORD_DIR/check/path")")"
test "$(wc -l <"$AD_DROID_RULES_RECORD_DIR/calls")" = 3
/opt/driver/tree-snapshot.sh "$factory" >"$case_dir/factory-after"
cmp "$case_dir/factory-before" "$case_dir/factory-after"
printf 'Checked Droid Rule Set spawn, reuse and refusal passed\n'
```
