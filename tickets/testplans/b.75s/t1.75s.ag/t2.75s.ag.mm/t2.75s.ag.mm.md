# agent-4: inherited arguments for another agent are refused

Maps to SRD SR-14.5 and the template rule of Epic 13 (`b.ssz`): a
template's non-empty `agent_args` are inherited only by a spawn of the
template's own agent. A spawn of the other agent that passes no
`agent_args` of its own is refused with `ErrAgentArgsMismatch`, and the
description names both agents.

## Setup

Per-case DB-reset. The case writes two templates with `make-template`,
one for each agent, each with `agent_args`.

## Steps

1. `make-template --agent claude --name ag-mm-claude --cwd /tmp
   --agent-args --model --agent-args opus` and `make-template --agent
   codex --name ag-mm-codex --cwd /tmp --agent-args --model
   --agent-args gpt` (both succeed).
2. `agent-director spawn --agent codex --template ag-mm-claude
   --tmux-session-name cd-ag-mm1`.
3. `agent-director spawn --agent claude --template ag-mm-codex
   --tmux-session-name cd-ag-mm2`.
4. Check nothing was written.

## Pass criteria

- Steps 2 and 3 exit 1 with `err_name` `ErrAgentArgsMismatch`; the
  descriptions contain `agent_args are for agent claude and this spawn's
  agent is codex` and `agent_args are for agent codex and this spawn's
  agent is claude` respectively.
- After each: `list` shows no rows, `spawns` holds none, the tmux
  session ids are unchanged, and the templates directory holds exactly
  `ag-mm-claude.toml` and `ag-mm-codex.toml`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN

db="$HOME/.agent-director/state.db"
tpl_dir="$HOME/.agent-director/templates"
# Should a refused call ever launch, it gets the stand-in, never Claude Code.
standin_path="/opt/driver/stand-in-claude:$PATH"
fail() { echo "FAIL: $*"; exit 1; }

session_ids() { tmux ls -F '#{session_id}' 2>/dev/null || true; }
session_names() { tmux ls -F '#{session_name}' 2>/dev/null || true; }
tpl_files() { ls -A "$tpl_dir" 2>/dev/null || true; }
sessions_before="$(session_ids)"
cleanup() {
    local sid
    for sid in $(session_ids); do
        case " $(printf '%s ' $sessions_before)" in
            *" $sid "*) ;;
            *) tmux kill-session -t "$sid" 2>/dev/null || true ;;
        esac
    done
}
trap cleanup EXIT

# refuse WANT_ERR_NAME WANT_TEXT ARGS...: agent-director ARGS... must exit 1
# with a JSON envelope on stderr whose err_name is WANT_ERR_NAME and whose
# err_description contains WANT_TEXT.
refuse() {
    local want_name="$1" want_text="$2" err code envelope got desc
    shift 2
    set +e
    err="$(PATH="$standin_path" agent-director "$@" 2>&1 >/dev/null)"
    code=$?
    set -e
    [ "$code" -eq 1 ] || fail "agent-director $* exited $code; want 1: $err"
    envelope="$(printf '%s\n' "$err" | awk '/^\{/{ln=$0} END{print ln}')"
    got="$(jq -r '.err_name' <<<"$envelope")"
    [ "$got" = "$want_name" ] || fail "agent-director $*: err_name=[$got]; want $want_name; stderr: $err"
    desc="$(jq -r '.err_description' <<<"$envelope")"
    case "$desc" in
        *"$want_text"*) ;;
        *) fail "agent-director $*: err_description=[$desc]; want it to contain [$want_text]" ;;
    esac
}

# assert_untouched TEMPLATE_FILES: nothing the refused calls could write exists:
# no row (list and the store), no new tmux session, and the templates dir
# holds exactly TEMPLATE_FILES (what the case itself wrote, "" for none).
assert_untouched() {
    local want_tpl="$1" n_list n_db after
    n_list="$(agent-director list | jq '.spawns | length')"
    [ "$n_list" = "0" ] || fail "list shows $n_list row(s); want 0"
    n_db="$(/opt/driver/sql.sh -readonly "$db" 'SELECT COUNT(*) FROM spawns;')"
    [ "$n_db" = "0" ] || fail "the store holds $n_db spawns row(s); want 0"
    after="$(session_ids)"
    [ "$after" = "$sessions_before" ] || fail "tmux sessions changed; before=[$sessions_before] after=[$after]"
    [ "$(tpl_files)" = "$want_tpl" ] || fail "templates dir holds [$(tpl_files)]; want [$want_tpl]"
}

agent-director make-template --agent claude --name ag-mm-claude --cwd /tmp --agent-args --model --agent-args opus >/dev/null
agent-director make-template --agent codex --name ag-mm-codex --cwd /tmp --agent-args --model --agent-args gpt >/dev/null
want_tpl="$(printf 'ag-mm-claude.toml\nag-mm-codex.toml')"
[ "$(tpl_files)" = "$want_tpl" ] || fail "templates dir holds [$(tpl_files)]; want [$want_tpl]"

refuse ErrAgentArgsMismatch "agent_args are for agent claude and this spawn's agent is codex" \
    spawn --agent codex --template ag-mm-claude --tmux-session-name cd-ag-mm1
assert_untouched "$want_tpl"
if session_names | grep -qx 'cd-ag-mm1'; then fail "a tmux session named cd-ag-mm1 exists"; fi

refuse ErrAgentArgsMismatch "agent_args are for agent codex and this spawn's agent is claude" \
    spawn --agent claude --template ag-mm-codex --tmux-session-name cd-ag-mm2
assert_untouched "$want_tpl"
if session_names | grep -qx 'cd-ag-mm2'; then fail "a tmux session named cd-ag-mm2 exists"; fi

echo "agent-4 ok: claude args to a codex spawn and codex args to a claude spawn -> ErrAgentArgsMismatch; no row, no session, templates unchanged"
```
