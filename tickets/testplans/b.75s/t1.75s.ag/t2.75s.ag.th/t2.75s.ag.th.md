# agent-17: the Codex folder named by CODEX_HOME is unchanged by spawn, pause, resume, kill and expire

Maps to SRD SR-5.6 and SR-14.5 of Epic 13 (`b.ssz`): agent-director writes
nothing into a Codex folder (its configuration, logins, rollouts and trust
entries are Codex's own); everything a launch needs is on the command line
(case agent-6 covers the spawn). This case extends the check over the whole
life of a codex row: the folder's tree (paths, modes and bytes) is identical
before the spawn and after the spawn, `pause`, `resume`, `kill`, `find-missing`
and `expire`.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex`, named by `[launch] codex_path`.
Hooks are typed into it with `/opt/driver/pane-hook.sh --agent codex`;
`pause` types `/exit`, which makes it fire `SessionEnd` and exit.
`/opt/driver/tree-snapshot.sh` lists a tree.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"`, and makes
a cwd and a temporary Codex folder holding a `config.toml`, an `auth.json`
(mode 0600), a `hooks.json`, and one rollout file, and removes
them, the config and the stand-in's record directory on exit.

## Steps

1. Write the config; make the cwd and the folder; snapshot the folder.
2. Spawn a codex row with `--extra-env CODEX_HOME=<folder>`; type a
   `SessionStart`; snapshot.
3. `pause`; the row is `ended`; snapshot.
4. `resume`, then type a `SessionStart` with source `resume`; snapshot.
5. `kill`, `find-missing` (the row becomes `missing`), snapshot.
6. `expire --older-than 0d`; snapshot.

## Pass criteria

- Each snapshot is byte for byte the first one.
- Step 3: `ended`. Step 5: `missing`. Step 6: `ids` lists the row and `get`
  no longer finds it.
- The folder's `auth.json` is still mode 600.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# fire ID EVENT [EXTRA_JSON]: type the Codex hook EVENT, for the session $sid and
# its rollout $rollout, into the row's pane; the hook must exit 0, print nothing.
fire() {
    local x="$1" extra='{}' body out
    [ $# -lt 3 ] || extra="$3"
    body="$(jq -cn --arg e "$2" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$x" "$body")" || fail "the $2 hook of $x exited non-zero"
    [ -z "$out" ] || fail "the $2 hook of $x printed [$out]; want nothing"
}

# make_rollout SID: the rollout file Codex would keep for session SID in the
# Codex folder; prints its path.
make_rollout() {
    local p="$codex_home/sessions/2026/10/12/rollout-2026-10-12T10-00-00-$1.jsonl"
    mkdir -p "$(dirname "$p")"
    printf '{"type":"session_meta"}\n' > "$p"
    printf '%s' "$p"
}

id="ag-th-$uuid"
sid="$(cat /proc/sys/kernel/random/uuid)"
rollout=""
snap_before=""
snap_now=""
trap 'rm -f ${snap_before:+"$snap_before"} ${snap_now:+"$snap_now"}; cleanup' EXIT

# same WHAT: the Codex folder's tree is the one snapshotted in step 1.
same() {
    /opt/driver/tree-snapshot.sh "$codex_home" > "$snap_now"
    diff -u "$snap_before" "$snap_now" || fail "the Codex folder $codex_home changed: $1"
}

# Step 1: the config, the cwd and a Codex folder with content of its own.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-th-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
codex_home="$(mktemp -d)"
printf 'model = "gpt-5"\n' > "$codex_home/config.toml"
printf '{"stand-in":true}\n' > "$codex_home/auth.json"
chmod 600 "$codex_home/auth.json"
printf '{"hooks":{}}\n' > "$codex_home/hooks.json"
rollout="$(make_rollout "$sid")"
snap_before="$(mktemp)"
snap_now="$(mktemp)"
/opt/driver/tree-snapshot.sh "$codex_home" > "$snap_before"
[ -s "$snap_before" ] || fail "the snapshot of $codex_home is empty"

# Step 2: spawn and the first hook.
spawn_codex "$id" cd-ag-th --extra-env "CODEX_HOME=$codex_home"
fire "$id" SessionStart '{"source":"startup"}'
wait_state "$id" waiting
same "after spawn"

# Step 3: pause.
old_pid="$(cat "$rec_root/$id/pid")"
agent-director pause --instance-id "$id" >/dev/null || fail "pause exited non-zero"
[ "$(state "$id")" = "ended" ] || fail "state is [$(state "$id")] after pause; want ended"
same "after pause"

# Step 4: resume and the resumed session's first hook.
agent-director resume --instance-id "$id" >/dev/null || fail "resume exited non-zero"
wait_record "$id" "$old_pid"
fire "$id" SessionStart '{"source":"resume"}'
wait_state "$id" waiting
same "after resume"

# Step 5: kill; find-missing marks the row missing.
agent-director kill --instance-id "$id" >/dev/null || fail "kill exited non-zero"
end=$(( $(date +%s) + 15 ))
while :; do
    res="$(agent-director find-missing)" || fail "find-missing exited non-zero"
    [ "$(state "$id")" != "missing" ] || break
    [ "$(date +%s)" -lt "$end" ] || fail "row $id is [$(state "$id")] 15 s after kill; want missing"
    sleep 1
done
same "after kill and find-missing"

# Step 6: expire deletes the row.
res="$(agent-director expire --older-than 0d)" || fail "expire exited non-zero: $res"
[ "$(jq -c '.ids' <<<"$res")" = "[\"$id\"]" ] || fail "expire result [$res]; want ids [$id]"
if agent-director get --instance-id "$id" >/dev/null 2>&1; then fail "row $id still found after expire"; fi
same "after expire"
[ "$(stat -c %a "$codex_home/auth.json")" = "600" ] || fail "auth.json is mode $(stat -c %a "$codex_home/auth.json"); want 600"

echo "agent-17 ok: the Codex folder's tree is identical after spawn, pause, resume, kill, find-missing and expire"
```
