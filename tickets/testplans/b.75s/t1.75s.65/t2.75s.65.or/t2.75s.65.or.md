# Test case: Hook origin and closeout authority

Maps to t3.7tz.6o.5h.ov, SR-5.5 and audit row 9. The harness reset supplies a clean store and tmux namespace. Use shipped CLI/hooks and the existing stand-ins; no authenticated provider qualification is claimed.

## Steps

For each provider, admit through a real hook in its owned pane. Compare exact allow, deny and own-timeout bytes against controls. Inject observational closure on an admitted request while its parent is still alive, then verify zero bytes and no acknowledgement. SQL closure is explicit fault injection; atomic lifecycle closure is covered by the store tests.

## Pass criteria

Exact envelope files match, WAL is active, committed origin/claim fields match each positive response, and the closed negative retains zero response bytes. A fresh public reader cannot turn closure into authority.

```bash
set -euo pipefail
config="$HOME/.config/agent-director/config.toml"
db="$HOME/.config/agent-director/state.db"
trap 'rm -f "$config"' EXIT
printf '[relay]\ntimeout_seconds=7\n[launch]\nclaude_path="/opt/driver/stand-in-claude/claude"\ncodex_path="/opt/driver/stand-in-codex/codex"\n' > "$config"
[[ $(/opt/driver/sql.sh -readonly "$db" 'PRAGMA journal_mode;') == wal ]]
for agent in claude codex; do
 for history in allow deny timeout closed; do
  id=$(agent-director spawn --agent "$agent" --cwd /tmp --relay-mode on --no-pre-trust | jq -er .instance_id)
  session=$(cat /proc/sys/kernel/random/uuid)
  startup=$(jq -cn --arg sid "$session" '{hook_event_name:"SessionStart",source:"startup",session_id:$sid,transcript_path:("/tmp/rollout-2026-10-30T00-00-00-"+$sid+".jsonl")}')
  /opt/driver/pane-hook.sh --agent "$agent" "$id" "$startup" >/dev/null
  payload=$(jq -cn --arg sid "$session" '{hook_event_name:"PermissionRequest",session_id:$sid,tool_name:"Bash",tool_input:{command:"true"},tool_use_id:"origin-tool"}')
  job=$(/opt/driver/pane-hook.sh --agent "$agent" --no-wait "$id" "$payload")
  token=''
  deadline=$((SECONDS+5))
  until [[ -n $token ]]; do
   token=$(/opt/driver/sql.sh -readonly "$db" "SELECT request_token FROM permission_requests WHERE instance_id='$id';")
   (( SECONDS < deadline )); sleep 0.02
  done
  if [[ $history == allow || $history == deny ]]; then
   agent-director decide --instance-id "$id" --request-token "$token" --decision "$history" >/dev/null
  elif [[ $history == closed ]]; then
   /opt/driver/sql.sh "$db" "UPDATE permission_requests SET closed_at=CAST(strftime('%s','now') AS INTEGER)*1000,decision='deny',decision_reason='ended',decision_origin='observation' WHERE request_token='$token';"
  fi
  /opt/driver/pane-hook.sh --timeout 12 --wait "$job" > "$job/actual"
  case $history in
   allow) printf '%s\n' '{"hookSpecificOutput":{"decision":{"behavior":"allow"},"hookEventName":"PermissionRequest"}}' > "$job/expected" ;;
   deny) printf '%s\n' '{"hookSpecificOutput":{"decision":{"behavior":"deny","message":"operator"},"hookEventName":"PermissionRequest"}}' > "$job/expected" ;;
   timeout) printf '%s\n' '{"hookSpecificOutput":{"decision":{"behavior":"deny","message":"Denied by orchestrator"},"hookEventName":"PermissionRequest"}}' > "$job/expected" ;;
   closed) : > "$job/expected" ;;
  esac
  cmp "$job/expected" "$job/actual"
  if [[ $history == closed ]]; then
   [[ $(/opt/driver/sql.sh -readonly "$db" "SELECT decision_origin='observation' AND delivered_at IS NULL AND response_state IS NULL FROM permission_requests WHERE request_token='$token';") == 1 ]]
   if agent-director decide --instance-id "$id" --request-token "$token" --decision allow > "$job/retry" 2> "$job/error"; then exit 1; fi
   jq -e '.err_name=="ErrNoOpenPermissionRequest"' "$job/error" >/dev/null
  else
   origin=caller; [[ $history != timeout ]] || origin=timeout
   [[ $(/opt/driver/sql.sh -readonly "$db" "SELECT decision_origin='$origin' AND response_state='relay_claimed' AND delivered_at IS NOT NULL AND proven_gone_at IS NULL FROM permission_requests WHERE request_token='$token';") == 1 ]]
  fi
  agent-director get-permission --request-token "$token" >/dev/null
  agent-director kill --instance-id "$id" >/dev/null
 done
done
printf '%s\n' '{"case":"t2.75s.65.or","status":"pass","checks":["Claude-Codex-exact-envelopes","own-timeout","living-parent-closed-no-output","origin-and-claim","WAL"],"authenticated_provider":false}'
```
