# interact-14-input-lock-causes

Regression for Bug b.vxo, using the existing stand-in Codex, pane hook,
real CLI and real tmux. No reader rate loop or sleep establishes lock ownership.

## Setup

The per-case DB-reset gives a fresh store. Spawn an uncovered Codex row and
fire SessionStart from its own pane with `pane-hook.sh --agent codex`.

## Steps

1. Run `list` on the waiting row and check neither synchronization file exists.
2. Hold its `.lock` with `flock`; a FIFO acknowledges acquisition and another
   FIFO keeps the holder alive until both calls finish.
3. Call `send-keys` and `pause`. Release the holder, then read the row again.

## Pass criteria

`list` exposes the waiting row and creates neither `.lock` nor `.intent.lock`.
Both calls exit 1 with empty stdout. Their errors retain `input lock stayed
busy for 2s` in `err_description` and `err_details.reason`; names remain
`ErrSubmissionUnconfirmed` and `ErrInternal`. Both carry `not_started`,
`pre_input`, zero bytes, false Enter/keys/confirmation and `waiting` facts.
The row is still waiting after both refusals. Cleanup ends only this case's
session and removes its private files and config.

```bash
set -euo pipefail
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME
driver="${AD_TEST_DRIVER:-/opt/driver}"
case_dir="$(mktemp -d /tmp/vxo-input-XXXXXX)"
id="vxo-$(cat /proc/sys/kernel/random/uuid)"
sid="$(cat /proc/sys/kernel/random/uuid)"
db="$HOME/.config/agent-director/state.db"
config="$HOME/.config/agent-director/config.toml"
holder=""
socket=""
session=""
cleanup() {
    if [[ -n "$holder" ]]; then kill "$holder" 2>/dev/null || true; wait "$holder" 2>/dev/null || true; fi
    if [[ -n "$session" ]]; then tmux -S "$socket" kill-session -t "$session" 2>/dev/null || true; fi
    rm -f "$config"
    rm -rf "$case_dir" "/tmp/stand-in-codex/$id"
}
trap cleanup EXIT
mkdir -p "$HOME/.config/agent-director" "$case_dir/codex/sessions"
printf '[launch]\ncodex_path = "%s"\n' "$driver/stand-in-codex/codex" >"$config"
rollout="$case_dir/codex/sessions/rollout-2026-11-23T00-00-00-$sid.jsonl"
printf '{"type":"session_meta"}\n' >"$rollout"
agent-director spawn --agent codex --cwd "$case_dir" --instance-id "$id" \
    --tmux-session-name "cd-$id" --no-pre-trust --extra-env "CODEX_HOME=$case_dir/codex" >/dev/null
socket="$("$driver/sql.sh" -readonly "$db" "SELECT tmux_socket FROM spawns WHERE instance_id='$id';")"
pane="$("$driver/sql.sh" -readonly "$db" "SELECT pane_id FROM spawns WHERE instance_id='$id';")"
session="$(tmux -S "$socket" display-message -p -t "$pane" '#{session_id}')"
"$driver/pane-hook.sh" --agent codex "$id" "$(jq -cn --arg sid "$sid" --arg tp "$rollout" --arg cwd "$case_dir" \
    '{hook_event_name:"SessionStart",source:"startup",session_id:$sid,transcript_path:$tp,cwd:$cwd,model:"stand-in",permission_mode:"default"}')"
token="$("$driver/sql.sh" -readonly "$db" "SELECT launch_token FROM spawns WHERE instance_id='$id';")"
store_id="$("$driver/sql.sh" -readonly "$db" "SELECT value FROM store_meta WHERE key='store_id';")"
[[ -n "$token" && -n "$store_id" ]]
lock="$HOME/.config/agent-director/input/$store_id/$token.lock"
agent-director list >"$case_dir/list.json"
jq -e --arg id "$id" '.spawns | any(.instance_id==$id and .state=="waiting")' "$case_dir/list.json" >/dev/null
failed=0
for path in "$lock" "${lock%.lock}.intent.lock"; do
    if [[ -e "$path" ]]; then echo "FAIL idle list created $path"; failed=1; fi
done
mkdir -p "$(dirname "$lock")"
mkfifo "$case_dir/acquired" "$case_dir/release"
# The driver's diagnostic/xtrace streams own descriptors 8 and 9.
exec 6<>"$case_dir/acquired" 7<>"$case_dir/release"
flock -F -x "$lock" bash -c 'printf "held\n" >&6; read -r _ <&7' &
holder=$!
read -r -t 10 -u 6 acquired
[[ "$acquired" == held ]]
for verb in send-keys pause; do
    args=("$verb" --instance-id "$id")
    want=ErrInternal
    if [[ "$verb" == send-keys ]]; then args+=(--text ': must not arrive'); want=ErrSubmissionUnconfirmed; fi
    rc=0
    agent-director "${args[@]}" >"$case_dir/out" 2>"$case_dir/error" || rc=$?
    cat "$case_dir/error"
    if [[ "$rc" != 1 || -s "$case_dir/out" ]] || ! jq -e --arg name "$want" '
        .err_name==$name and (.err_description|contains("input lock stayed busy for 2s")) and
        .err_details.outcome=="not_started" and .err_details.phase=="pre_input" and
        .err_details.bytes_attempted==0 and .err_details.enter_may_have_been_sent==false and
        .err_details.keys_sent==false and .err_details.submission_confirmed==false and
        .err_details.row_state=="waiting" and (.err_details.reason|contains("input lock stayed busy for 2s"))
        ' "$case_dir/error" >/dev/null; then
        echo "FAIL $verb lost the lock cause or zero-input facts"; failed=1
    fi
done
printf 'release\n' >&7
wait "$holder"
holder=""
agent-director get --instance-id "$id" | jq -e '.state=="waiting"' >/dev/null
[[ "$failed" == 0 ]]
echo 'interact-14-input-lock-causes: idle read and both writer refusals passed'
```
