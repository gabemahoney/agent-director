# agent-16: find-missing on codex rows: an exited agent with its end report becomes ended, a vanished one without a report becomes missing

Maps to SRD SR-7.2, SR-14.5, SR-22.9 and SR-27.4 of Epic 13 (`b.ssz`): a
Codex row whose process is gone is judged by whether the agent reported its
end. Row A: `/exit` was typed at its pane outside `pause`, so Codex sent
`SessionEnd` (the end report) and exited; the row keeps `waiting` until
`find-missing` moves it to `ended` (one `ad.find_missing.tick`, `new_state`
`ended`, `proc_absent`); it is not counted and not in `ids`. Row B: the pane
process ended with no end report (a kill); `find-missing` marks it `missing`
and lists it in `ids`.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex`, named by `[launch] codex_path`.
Typing `/exit` at its pane makes it fire a Codex `SessionEnd` through
`agent-director hook --agent codex` and exit; it sends nothing on a signal.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"`, makes a
cwd and a temporary Codex folder holding the rollouts the payloads name, and
removes them, the config and the stand-in's record directories on exit.

## Steps

1. Write the config; make the cwd and the Codex folder.
2. Spawn rows A and B (`--extra-env CODEX_HOME=<folder>`), each with its own
   session id; type a `SessionStart` into each; both are `waiting`.
3. Row A: type `/exit` at its pane and wait for the stand-in's `SessionEnd`
   hook to finish. Row B: `agent-director kill`.
4. `get` both rows (find-missing has not run).
5. `agent-director find-missing`.
6. `get` both rows; read the trail's `ad.find_missing.tick` records.

## Pass criteria

- Step 3: the stand-in's SessionEnd hook exited `0`; `kill` exits 0.
- Step 4: row A is still `waiting` with `end_reported_at` a positive
  integer; row B is `waiting` with `end_reported_at` NULL.
- Step 5: exit 0; `count` is `1` and `ids` is exactly `[B]`.
- Step 6: row A `ended` (`ended_at` set, `end_reported_at` kept), row B
  `missing`. One tick for A (`waiting`, `ended`, `proc_absent`) and one for B
  (`waiting`, `missing`, `proc_absent`), both `source` `ad_find_missing`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# fire ID EVENT [EXTRA_JSON]: type the Codex hook EVENT, for the session $sid and
# its rollout $rollout, into the row's pane; the hook must exit 0, print nothing.
fire() {
    local x="$1" extra='{}' body out
    [ $# -lt 3 ] || extra="$3"
    body="$(jq -cn --arg e "$2" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$x" "$body")" || fail "the $2 hook of $x exited non-zero"
    [ -z "$out" ] || fail "the $2 hook of $x printed [$out]; want nothing"
}

# make_rollout SID: the rollout file Codex would keep for session SID in the
# Codex folder; prints its path.
make_rollout() {
    local p="$codex_home/sessions/2026/10/12/rollout-2026-10-12T10-00-00-$1.jsonl"
    mkdir -p "$(dirname "$p")"
    printf '{"type":"session_meta"}\n' > "$p"
    printf '%s' "$p"
}

a="ag-fm-a-$uuid"
b="ag-fm-b-$uuid"
sid_a="$(cat /proc/sys/kernel/random/uuid)"
sid_b="$(cat /proc/sys/kernel/random/uuid)"
sid=""
rollout=""

# Step 1: the config, the cwd and a Codex folder.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-fm-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
codex_home="$(mktemp -d)"

# Step 2: two waiting codex rows, each with its own session.
spawn_codex "$a" cd-ag-fm-a --extra-env "CODEX_HOME=$codex_home"
spawn_codex "$b" cd-ag-fm-b --extra-env "CODEX_HOME=$codex_home"
sid="$sid_a"; rollout="$(make_rollout "$sid")"
fire "$a" SessionStart '{"source":"startup"}'
sid="$sid_b"; rollout="$(make_rollout "$sid")"
fire "$b" SessionStart '{"source":"startup"}'
wait_state "$a" waiting
wait_state "$b" waiting

# Step 3: A exits by /exit typed outside pause; B is killed.
row="$(col "$a" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')")"
sock="${row%%|*}"
pane="${row#*|}"
[ -n "$pane" ] || fail "row A records no pane_id"
tmux ${sock:+-S "$sock"} send-keys -t "$pane" -l '/exit'
tmux ${sock:+-S "$sock"} send-keys -t "$pane" Enter
end=$(( $(date +%s) + 15 ))
until [ -s "$rec_root/$a/exit-hook.exit" ]; do
    [ "$(date +%s)" -lt "$end" ] || fail "the stand-in's SessionEnd hook did not finish within 15 s of /exit"
    sleep 0.2
done
[ "$(cat "$rec_root/$a/exit-hook.exit")" = "0" ] || fail "the stand-in's SessionEnd hook exited [$(cat "$rec_root/$a/exit-hook.exit")]; want 0"
agent-director kill --instance-id "$b" >/dev/null || fail "kill of row B exited non-zero"
end=$(( $(date +%s) + 15 ))
until ! tmux ${sock:+-S "$sock"} list-panes -a -F '#{pane_id}' 2>/dev/null | grep -qx "$pane"; do
    [ "$(date +%s)" -lt "$end" ] || fail "row A's pane $pane still exists 15 s after /exit"
    sleep 0.2
done

# Step 4: nothing has judged them yet.
[ "$(state "$a")" = "waiting" ] || fail "row A is [$(state "$a")] before find-missing; want waiting"
[ "$(state "$b")" = "waiting" ] || fail "row B is [$(state "$b")] before find-missing; want waiting"
rep_a="$(col "$a" "IFNULL(end_reported_at, '')")"
[[ "$rep_a" =~ ^[0-9]+$ ]] && [ "$rep_a" -gt 0 ] || fail "row A end_reported_at is [$rep_a]; want a positive integer"
[ -z "$(col "$b" "IFNULL(end_reported_at, '')")" ] || fail "row B has an end report"

# Step 5: find-missing.
res="$(agent-director find-missing)" || fail "find-missing exited non-zero"
[ "$(jq -c '[.count, .ids]' <<<"$res")" = "[1,[\"$b\"]]" ] || fail "find-missing result [$res]; want count 1 and ids [$b]"

# Step 6: A ended, B missing; one tick each.
rowa="$(agent-director get --instance-id "$a")"
[ "$(jq -r '.state' <<<"$rowa")" = "ended" ] || fail "row A is [$(jq -r '.state' <<<"$rowa")]; want ended"
[ -n "$(jq -r '.ended_at // ""' <<<"$rowa")" ] || fail "row A has no ended_at: $rowa"
[ "$(col "$a" "IFNULL(end_reported_at, '')")" = "$rep_a" ] || fail "row A's end_reported_at changed"
[ "$(state "$b")" = "missing" ] || fail "row B is [$(state "$b")]; want missing"
[ -f "$trail" ] || fail "no trail file at $trail"
for x in "$a:ended" "$b:missing"; do
    ticks="$(jq -c --arg x "${x%%:*}" 'select(.event == "ad.find_missing.tick" and .instance_id == $x)' "$trail")"
    [ "$(printf '%s' "$ticks" | grep -c . || true)" -eq 1 ] || fail "want exactly one tick for ${x%%:*}: [$ticks]"
    got="$(jq -r '[.prior_state, .new_state, .reconciliation_reason, .source] | join("|")' <<<"$ticks")"
    [ "$got" = "waiting|${x#*:}|proc_absent|ad_find_missing" ] || fail "tick of ${x%%:*} is [$got]; want [waiting|${x#*:}|proc_absent|ad_find_missing]"
done

echo "agent-16 ok: find-missing: the codex row that exited by /exit (end report) is ended and not listed; the killed one without a report is missing and listed"
```
