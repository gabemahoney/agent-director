# Test case: Native interrupt with independent permission plan and denied question

Maps E3 AC2/3/5/6; SR-0.2, SR-3.2–3.3, SR-4.1–4.2, SR-9.2.

## Setup

Exact-revision shipped CLI/MCP, shared helper and existing captured authorization
bodies rebound only to this synthetic connection. Direct question below is
source-derived synthetic, not a captured or authenticated question condition.

## Steps

1. Emit independent permission and blocking plan plus a question. Observe two
   public request tokens and a question denial with the exact protocol ID.
2. Deny the plan through CLI. Recreate MCP inspection, preserve the permission
   token, refuse pause, and interrupt natively without allowing a model follow-up.
3. Leave the independent permission unanswered. Verify the finite cancellation
   shutdown kills only the owned accepting runtime and gateway.

## Pass criteria

Question denial cannot answer either independent request. Plan denial cannot
decide the permission; native interrupt cannot prove unknown-origin request gone.
No model bytes for guarded input. One JSON verdict; real permissions remain pending.

```bash
set -euo pipefail
source /work/source/test/driver/cursor-pane-controls.sh
relay_scenario plan
jq '.requests += [{jsonrpc:"2.0",id:"collector-question",method:"cursor/ask_question",params:{questions:[{text:"Synthetic question"}]}}] | .waitForRelease=true' "$case_dir/scenario" > "$case_dir/next"
mv "$case_dir/next" "$case_dir/scenario"
pc_start
ad send-keys --instance-id "$id" --text 'controlled permission plan question' >/dev/null
check hold.two-requests wait_until request_count_is 2
question_denied() { frames | jq -se 'any(.[]; .id=="collector-question" and (has("method")|not) and ((.result|tostring)|contains("cancelled")))' >/dev/null; }
check question.denied wait_until question_denied
plan_tool=$(jq -r '.requests[]|select(.id=="plan")|.params.toolCallId' "$case_dir/scenario")
permission_tool=$(jq -r '.requests[]|select(.id=="permission")|.params.toolCall.toolCallId' "$case_dir/scenario")
plan_token=$(request_token "$plan_tool"); permission_token=$(request_token "$permission_tool")
check hold.distinct test "$plan_token" != "$permission_token"
relay_decide "$plan_token" deny >/dev/null
plan_rejected() { frames | jq -se 'any(.[];.id=="plan" and .result.outcome.outcome=="rejected")' >/dev/null; }
check plan.rejected wait_until plan_rejected
check hold.permission-undecided test "$(sql "SELECT decision IS NULL AND proven_gone_at IS NULL FROM permission_requests WHERE request_token='$permission_token';")" = 1
pc_public MCP get-permission "$(jq -cn --arg t "$permission_token" '{request_token:$t}')" | jq -e --arg t "$permission_token" '.request_token==$t' >/dev/null
cp "$case_dir/raw" "$case_dir/raw-before-guard"
check hold.pause-input-guards relay_guard
check hold.guard-raw-unchanged cmp "$case_dir/raw-before-guard" "$case_dir/raw"
tmux -S "$socket" send-keys -t "$pane" C-c
cancel_seen() { frames | jq -se 'any(.[];.method=="session/cancel")' >/dev/null; }
check control.native-interrupt wait_until cancel_seen
check hold.not-proven-by-cancel test "$(sql "SELECT proven_gone_at IS NULL FROM permission_requests WHERE request_token='$permission_token';")" = 1
check owned.runtime-gone wait_gone "$runtime"
check owned.gateway-gone wait_gone "$gateway"
check hold.raw-complete pc_raw_complete
# After the guard snapshot, only the single native cancel may add wire bytes.
check hold.raw-guard-window node -e 'const fs=require("fs"),a=require("assert/strict"),before=fs.readFileSync(process.argv[1]),raw=fs.readFileSync(process.argv[2]);a.ok(raw.subarray(0,before.length).equals(before));const tail=raw.subarray(before.length).toString();a.ok(tail.endsWith("\n"));const lines=tail.slice(0,-1).split("\n");a.equal(lines.length,1);const f=JSON.parse(lines[0]);a.equal(f.method,"session/cancel");a.equal(f.params.sessionId,process.argv[3]);a.equal(f.id,undefined);' "$case_dir/raw-before-guard" "$case_dir/raw" "$session"
check hold.no-fabricated-permission-answer test "$(frames | jq -s '[.[]|select(.id=="permission" and (has("method")|not))]|length')" = 0
frames | jq -se '[.[]|select(.method=="session/prompt")]|length==1 and .[0].params.prompt[0].text=="controlled permission plan question"' >/dev/null
pc_finish t2.jf1.pc.02 '["question.denied","hold.independent-plan-permission","control.native-interrupt","hold.no-false-proof","owned.finite-cancel-death"]'
```
