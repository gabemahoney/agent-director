# agent-15: resume of a paused codex row: pending with resume <session id> on the argv, then waiting with the session id kept

Maps to SRD SR-5.0, SR-6.3, SR-14.5 and SR-22.9 of Epic 13 (`b.ssz`): a
finished codex row resumes by its session id. `resume` moves it to
`pending` and launches `codex resume <session id>` (agent-director's
settings before it, as for a spawn); the agent's first prompt of the resumed
session sends `SessionStart` with source `resume` and the same id, which
moves the row to `waiting` and keeps the session id.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex`, named by `[launch] codex_path`.
It records each launch's argv in `/tmp/stand-in-codex/<id>/argv` (rewritten
by the second launch; its `pid` file, written last, changes), and typing
`/exit` makes it fire the Codex `SessionEnd` and exit, which is how `pause`
ends it.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"`, makes a
cwd and a temporary Codex folder holding the rollout the payloads name, and
removes them, the config and the stand-in's record directory on exit.

## Steps

1. Write the config; make the cwd and the Codex folder with its rollout.
2. Spawn a codex row (`--extra-env CODEX_HOME=<folder>`), type a `SessionStart`
   (source `startup`); `pause` it; the row is `ended`. Note the stand-in's pid.
3. `agent-director resume --instance-id <id>`; `get`.
4. Wait for the stand-in's second record (a new pid); read its `argv`.
5. Type a `SessionStart` with source `resume` and the same session id.
6. `get` the row.

## Pass criteria

- Step 2: `pause` exits 0; the row is `ended` with the session id of the
  `SessionStart`.
- Step 3: resume exits 0 and echoes the id; `state` is `pending`.
- Step 4: the argv holds `resume` immediately followed by the session id,
  before the first `--`, and agent-director's own `-c hooks.SessionStart=`
  setting.
- Step 5: the hook exits 0 and prints nothing.
- Step 6: `state` `waiting`, `session_id` still the original id, `agent`
  `codex`, and no `end_reported_at` (the new session supersedes the earlier
  end report).

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# fire ID EVENT [EXTRA_JSON]: type the Codex hook EVENT, for the session $sid and
# its rollout $rollout, into the row's pane; the hook must exit 0, print nothing.
fire() {
    local x="$1" extra='{}' body out
    [ $# -lt 3 ] || extra="$3"
    body="$(jq -cn --arg e "$2" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$x" "$body")" || fail "the $2 hook of $x exited non-zero"
    [ -z "$out" ] || fail "the $2 hook of $x printed [$out]; want nothing"
}

# make_rollout SID: the rollout file Codex would keep for session SID in the
# Codex folder; prints its path.
make_rollout() {
    local p="$codex_home/sessions/2026/10/12/rollout-2026-10-12T10-00-00-$1.jsonl"
    mkdir -p "$(dirname "$p")"
    printf '{"type":"session_meta"}\n' > "$p"
    printf '%s' "$p"
}

id="ag-rz-$uuid"
sid="$(cat /proc/sys/kernel/random/uuid)"
rollout=""

# Step 1: the config, the cwd, and a Codex folder with the rollout.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-rz-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
codex_home="$(mktemp -d)"
rollout="$(make_rollout "$sid")"

# Step 2: a paused (ended) codex row with a session id.
spawn_codex "$id" cd-ag-rz --extra-env "CODEX_HOME=$codex_home"
fire "$id" SessionStart '{"source":"startup"}'
wait_state "$id" waiting
agent-director pause --instance-id "$id" >/dev/null || fail "pause exited non-zero"
[ "$(state "$id")" = "ended" ] || fail "state is [$(state "$id")] after pause; want ended"
[ "$(col "$id" session_id)" = "$sid" ] || fail "session_id is [$(col "$id" session_id)] after pause; want $sid"
old_pid="$(cat "$rec_root/$id/pid")"

# Step 3: resume.
out="$(agent-director resume --instance-id "$id")" || fail "resume exited non-zero"
[ "$(jq -r '.instance_id' <<<"$out")" = "$id" ] || fail "resume echoed [$out]; want id $id"
[ "$(state "$id")" = "pending" ] || fail "state is [$(state "$id")] after resume; want pending"

# Step 4: the second launch's argv.
wait_record "$id" "$old_pid"
mapfile -d '' -t argv < "$rec_root/$id/argv"
dd=${#argv[@]}
for i in "${!argv[@]}"; do
    if [ "${argv[$i]}" = "--" ]; then dd="$i"; break; fi
done
at=-1
for ((i = 0; i + 1 < dd; i++)); do
    if [ "${argv[$i]}" = "resume" ] && [ "${argv[$((i + 1))]}" = "$sid" ]; then at="$i"; break; fi
done
[ "$at" -ge 0 ] || fail "the resumed launch's argv has no 'resume $sid' before the first --: [${argv[*]}]"
case " ${argv[*]} " in *" hooks.SessionStart="*) ;; *) fail "the resumed launch carries no hooks.SessionStart setting: [${argv[*]}]" ;; esac

# Steps 5-6: the resumed session's first prompt reports in.
fire "$id" SessionStart '{"source":"resume"}'
wait_state "$id" waiting
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '[.agent, .session_id] | join("|")' <<<"$row")" = "codex|$sid" ] || fail "agent|session_id is [$(jq -r '[.agent, .session_id] | join("|")' <<<"$row")]; want codex|$sid"
[ -z "$(col "$id" "IFNULL(end_reported_at, '')")" ] || fail "end_reported_at is still set after the resumed SessionStart"

echo "agent-15 ok: resume of a paused codex row: pending, argv holds 'resume $sid', SessionStart(resume) -> waiting with the session id kept"
```
