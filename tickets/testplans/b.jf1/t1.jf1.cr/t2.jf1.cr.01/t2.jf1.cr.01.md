# Test case: Requested fields and once-only mappings

Maps to E2 AC1; SR-4.1/4.2, SR-5.1/5.2, SR-9.1.

## Setup

The harness reset supplies its clean baseline. Source the shared driver, which creates an additional private synthetic Cursor namespace with an override store and control sentinel. All SQL, including raw-byte observer queries and faults, goes through /opt/driver/sql.sh. Source mount and shipped binary must match the scheduler's full commit. No authenticated runtime is used.

## Steps

Execute the ordered block below. Each check tests the stated caller-visible state, preserved fields, request correlation, committed evidence, actual wire bytes or process lifetime. Any injected fault is a test fixture action; it does not establish a production configuration feature.

## Pass criteria

Every assertion passes and one final JSON object contains this case ID, stable check names, status pass, authenticated_cursor false and separate variable facts. Do not treat an exit code alone as evidence. The collector must be executed externally before claiming a case pass.

```bash
set -euo pipefail
source /work/source/test/driver/cursor-owned-launch.sh
relay_scenario coverage
relay_observe
relay_start
expected=$(jq '.requests|length' "$case_dir/scenario")
wait_until request_count_is "$expected"
decoy="$HOME/.config/agent-director/state.db"
agent-director list --home "$HOME" --store-path "$decoy" >/dev/null
sql "ATTACH DATABASE '$decoy' AS decoy; INSERT INTO decoy.spawns SELECT * FROM main.spawns WHERE instance_id='$id'; UPDATE decoy.spawns SET session_id='unrelated-default-session',state='waiting' WHERE instance_id='$id'; DETACH DATABASE decoy;"
decoy_before=$(/opt/driver/sql.sh -readonly -json "$decoy" "SELECT * FROM spawns WHERE instance_id='$id';")
ad get --instance-id "$id" > "$case_dir/get.json"
ad list > "$case_dir/list.json"
[[ $(sql "SELECT count(*) FROM permission_requests WHERE instance_id='$id' AND hook_pid IS NULL AND agent_id IS NULL AND main_turn_end_eligible=0;") == "$expected" ]]
while IFS= read -r frame; do
 tool=$(jq -r '.params.toolCall.toolCallId // .params.toolCallId' <<< "$frame")
 token=$(request_token "$tool")
 [[ -n $token ]]
 ad get-permission --request-token "$token" > "$case_dir/permission.json"
 jq -e --argjson frame "$frame" '.tool_input==($frame.params|tojson) and .tool_use_id==($frame.params.toolCall.toolCallId // $frame.params.toolCallId) and .hook_alive==null and .decision==null and .delivery=="not_confirmed"' "$case_dir/permission.json" >/dev/null
 jq -e --arg t "$token" '..|objects|select(.request_token?==$t)' "$case_dir/get.json" >/dev/null
 jq -e --arg t "$token" '..|objects|select(.request_token?==$t)' "$case_dir/list.json" >/dev/null
 relay_decide "$token" allow | jq -e '.delivery=="delivered" or .delivery=="not_confirmed"' >/dev/null
 if relay_decide "$token" deny >"$case_dir/duplicate" 2>"$case_dir/duplicate-error"; then exit 1; fi
 jq -e '.err_name=="ErrAlreadyDecided"' "$case_dir/duplicate-error" >/dev/null
done < <(jq -c '.requests[]' "$case_dir/scenario")
wait_until reply_count_is "$expected"
frames | jq -se '[.[]|select(has("method")|not)]|all(.[]; .result.outcome.optionId=="allow-once" or .result.outcome.outcome=="accepted")' >/dev/null
[[ $(/opt/driver/sql.sh -readonly -json "$decoy" "SELECT * FROM spawns WHERE instance_id='$id';") == "$decoy_before" ]]
[[ ! -e $case_dir/configured.db ]]
# Captured execution/edit/search/MCP/plan retain original fields; never-requested
# read/delete/fetch approvals are explicit synthetic variants, not live evidence.
relay_receipt t2.jf1.cr.01 '["AC1.raw-fields","AC1.actual-once","AC1.first-writer","AC1.null-origin","AC1.get-list-correlation","AC1.explicit-store-decoy-unchanged"]'
```
