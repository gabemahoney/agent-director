# Journal cutover: bounded completion, conservative blockers and literal restart

Maps to t3.7tz.tb.ib.yc and migration_retained row 12. This ordinary Docker
refinement does not claim a formal liveness pass. Seven synthetic fault classes
exercise the installed administrator against a real prior-release SQLite store,
custom database and invoking-home evidence roots. The old binary creates the
schema; SQL sets private identity fields with no public setter. Synthetic Claude
records use the existing supported version and qualify no new provider.

## Setup and steps

Create each fixture in its own temporary home. Derive the target schema from
a fresh current store. A preexisting exact sentinel must not let the ordinary
runtime migrate. Call the administrator once per fault under a 15-second bound.
Keep ambiguous/corrupt/conflicting/bad-proof evidence as a blocker. Unreadable
and cross-owner input must refuse with the exact restart advice. Retry once
unchanged, then repair only the condition and literally restart verification
with a fresh backup destination. Preserve the original launch lock and logs.

## Pass criteria

All seven classes terminate within the bound, with the specified outcome;
only committed SQLite authority enables a current runtime open. Successful
cutover preserves exact proof BLOB bytes and current-over-history precedence.
No test sends text or Enter. Later reads/retries cannot replay imported input.

```bash
set -euo pipefail
work=$(mktemp -d)
trap 'chmod -R u+rwX "$work"; rm -rf "$work"' EXIT
admin=/opt/bin/agent-director-admin
prior=/opt/previous-release/agent-director
sql=/opt/driver/sql.sh
hash() { sha256sum | cut -d' ' -f1; }
ref="$work/reference.db"
agent-director list --store-path "$ref" >/dev/null
target=$("$sql" "$ref" 'PRAGMA user_version')
token=aaaaaaaaaaaaaaaa
advice='keep the maintenance boundary, stop incompatible mutators, and restart administrator verification'
for fault in valid ambiguous corrupt unreadable conflict bad_proof wrong_identity; do
    home="$work/$fault"
    mkdir -m 700 "$home" "$home/custom"
    db="$home/custom/state.db"
    HOME="$home" "$prior" list --store-path "$db" >/dev/null
    old=$("$sql" "$db" 'PRAGMA user_version')
    sid=$("$sql" "$db" "SELECT value FROM store_meta WHERE key='store_id'")
    idcol=$("$sql" "$db" "SELECT name FROM pragma_table_info('spawns') WHERE name IN ('claude_instance_id','instance_id')")
    "$sql" "$db" "INSERT INTO spawns($idcol,state,cwd,tmux_session_name,relay_mode,launch_token,tmux_socket,pane_id,pane_pid,pane_starttime,life_number) VALUES('journal-case','waiting','/tmp','journal-case','off','$token','/tmp/journal.sock','%7',123,'12345678',3)"
    root="$home/.config/agent-director/input"
    dir="$root/$sid/$token"
    mkdir -p -m 700 "$dir"
    chmod -R go-rwx "$home"
    : >"$dir.lock"
    inode=$(stat -c '%d:%i' "$dir.lock")
    transcript="$dir/provider.jsonl"
    printf '%s' '{ "type":"user", "uuid":"9c8b2ab4-1880-48df-b2f7-8fc0a7e49311", "sessionId":"session", "cwd":"/tmp", "version":"2.1.280", "message":{"role":"user","content":"hello <&>"} }' >"$transcript"
    proofhash=$(hash <"$transcript")
    jq -n --arg token "$token" --arg transcript "$transcript" '{Agent:"claude",Version:"2.1.280",SessionID:"session",CWD:"/tmp",Socket:"/tmp/journal.sock",Token:$token,InstanceID:"journal-case",Transcript:$transcript,Created:"2026-10-01T00:00:00Z",Safe:true,KeymapPinned:false}' >"$dir/launch.json"
    emptyhash=$(printf '' | hash)
    payloadhash=$(printf 'hello <&>' | hash)
    raw=$(base64 -w0 "$transcript")
    jq -n --arg sid "$sid" --arg token "$token" --arg path "$transcript" --arg empty "$emptyhash" --arg payload "$payloadhash" --arg raw "$raw" --arg proofhash "$proofhash" '{request_id:"current",identity:{StoreID:$sid,Token:$token,InstanceID:"journal-case",Socket:"/tmp/journal.sock",PaneID:"%7",PID:123,Starttime:"12345678",Agent:"claude",CWD:"/tmp"},payload:"hello <&>",payload_sha256:$payload,watermark:{($path):{Size:0,PrefixSHA256:$empty}},phase:"submitted",bytes_attempted:9,enter_attempted:true,proof:{Path:$path,SessionID:"session",RecordID:"9c8b2ab4-1880-48df-b2f7-8fc0a7e49311",Offset:0,Raw:$raw,RawSHA256:$proofhash}}' >"$dir/attempt.json"
    cp "$dir/attempt.json" "$home/valid-attempt"
    currenthash=$(printf current | hash)
    printf '{older malformed history' >"$dir/request-$currenthash.json"
    blocker=''
    case "$fault" in
        ambiguous) "$sql" "$db" 'DELETE FROM spawns'; blocker=unresolved_life ;;
        corrupt) printf '{broken' >"$dir/attempt.json"; blocker=corrupt_current ;;
        unreadable) chmod 000 "$dir/attempt.json" ;;
        conflict) "$sql" "$db" 'UPDATE spawns SET pane_pid=124'; blocker=conflicting_process ;;
        bad_proof) jq '.proof.RawSHA256="bad"' "$home/valid-attempt" >"$dir/attempt.json"; blocker=invalid_attempt_or_proof ;;
        wrong_identity) jq '.identity.StoreID="bbbbbbbbbbbbbbbb"' "$home/valid-attempt" >"$dir/attempt.json" ;;
    esac
    printf '{"from":%s,"to":%s}\n' "$old" "$target" >"$home/custom/migrate-authorized"
    before=$(hash <"$db")
    rc=0
    HOME="$home" agent-director list --store-path "$db" >"$home/out" 2>"$home/err" || rc=$?
    test "$rc" -ne 0
    jq -e '.err_name=="ErrSchemaMigrationRequired"' "$home/err" >/dev/null
    test "$(hash <"$db")" = "$before"
    args=(--store-path "$db" migrate-journal --maintenance-window --roots-complete)
    rc=0
    HOME="$home" timeout 15 "$admin" "${args[@]}" --backup-dir "$home/backup-1" >"$home/out" 2>"$home/err" || rc=$?
    test "$rc" -ne 124
    if [[ "$fault" == unreadable || "$fault" == wrong_identity ]]; then
        test "$rc" -eq 1
        jq -e --arg advice "$advice" '.err_name=="ErrJournalMigrationRefused" and .details.refusal.action==$advice and .details.result.authority==""' "$home/err" >/dev/null
        test "$("$sql" "$db" "SELECT count(*) FROM sqlite_master WHERE name='input_launches'")" = 0
        rc=0
        HOME="$home" timeout 15 "$admin" "${args[@]}" --backup-dir "$home/backup-2" >"$home/out" 2>"$home/err" || rc=$?
        test "$rc" -eq 1
        jq -e --arg advice "$advice" '.details.refusal.action==$advice' "$home/err" >/dev/null
        chmod 600 "$dir/attempt.json"
        cp "$home/valid-attempt" "$dir/attempt.json"
        # Earlier schema hops can commit before the journal hop refuses.
        # Re-authorize the actual remaining chain, never reset its version.
        remaining=$("$sql" "$db" 'PRAGMA user_version')
        printf '{"from":%s,"to":%s}\n' "$remaining" "$target" >"$home/custom/migrate-authorized"
        HOME="$home" timeout 15 "$admin" "${args[@]}" --backup-dir "$home/backup-3" >"$home/out" 2>"$home/err"
    else
        test "$rc" -eq 0 || { cat "$home/err"; exit 1; }
    fi
    jq -e --arg sid "$sid" '.authority=="sqlite" and .store_id==$sid and .retirement_pending==false' "$home/out" >/dev/null
    test "$("$sql" "$db" 'PRAGMA user_version')" = "$target"
    test "$(stat -c '%d:%i' "$dir.lock")" = "$inode"
    test "$(hash <"$transcript")" = "$proofhash"
    test "$("$sql" "$db" 'SELECT json_extract(record,"$.Blocker") FROM input_launches')" = "$blocker"
    if [[ -z "$blocker" ]]; then
        expected=$(od -An -v -tx1 "$transcript" | tr -d ' \n' | tr '[:lower:]' '[:upper:]')
        test "$("$sql" "$db" 'SELECT hex(proof_raw) FROM input_attempts WHERE request_id="current"')" = "$expected"
    else
        test "$("$sql" "$db" 'SELECT count(*) FROM input_attempts')" = 0
        test "$("$sql" "$db" 'SELECT count(*) FROM input_imports WHERE length(evidence)>0')" = 1
    fi
    HOME="$home" agent-director list --store-path "$db" >/dev/null
    echo "$fault: bounded completion, preserved evidence and conservative authority"
done
```
