# Test case: supervised pane lifecycle

Maps to t3.m6h.7o.w8.u8 Work 6.

## Setup

Use the harness's fresh database and `/opt/driver/stand-in-codex/codex` as
`droid_path`. The stand-in records argv, then runs an interactive shell;
its `/exit` path invokes the existing SessionEnd hook before exiting.

## Steps

1. Refuse a flag-off Droid spawn without changing the database or creating
   launch files, templates or trust files.
2. Enable Droid and launch with an explicit model. Inspect its pending
   pane-only result and recorded settings argument, private modes and JSON.
3. Send two lines through the public manual-input verb; observe their exact
   output file. Refuse pause as an unsupported capability.
4. Send `/exit`; verify the hook returned success with empty output and left
   the stored row pending. Reconcile the exited process with find-missing.
5. Reuse the finished ID with a different model and a fresh settings path.
   Turn the flag off; inspect the live row and pane, then kill and reconcile it.

## Pass criteria

All exact assertions below pass. The two launches use distinct token
directories and the same instance ID. No native agent runs.

```bash
set -euo pipefail
case_dir=$(mktemp -d /tmp/droid-lane.XXXXXX)
export TMUX_TMPDIR="$case_dir/tmux"
export AD_CODEX_RECORD_DIR="$case_dir/records"
mkdir -p "$TMUX_TMPDIR" "$AD_CODEX_RECORD_DIR" "$case_dir/project"
cfg="$HOME/.config/agent-director/config.toml"
db="$HOME/.config/agent-director/state.db"
cleanup() {
    tmux kill-server 2>/dev/null || true
    rm -f "$cfg"
    # This case starts with no launch directory and owns every overlay it makes.
    if [[ -s "$case_dir/spawn" ]]; then
        rm -rf "$HOME/.config/agent-director/launch"
    fi
    rm -rf "$case_dir"
}
trap cleanup EXIT
configure() {
    cat >"$cfg" <<EOF
[experimental]
droid = $1
[launch]
droid_path = "/opt/driver/stand-in-codex/codex"
[tmux]
pending_grace_seconds = 30
EOF
}
wait_for() {
    local deadline=$((SECONDS + 45))
    until "$@"; do
        (( SECONDS < deadline )) || {
            printf 'timed out: %s\n' "$*" >&2
            if [[ -f "$case_dir/pane" ]]; then cat "$case_dir/pane" >&2; fi
            return 1
        }
        sleep 0.1
    done
}
configure false
/opt/driver/sql.sh -readonly "$db" .dump >"$case_dir/before"
rc=0
agent-director spawn --agent droid --cwd "$case_dir/project" -- --model first-model >"$case_dir/out" 2>"$case_dir/error" || rc=$?
test "$rc" = 1
test ! -s "$case_dir/out"
jq -e '.err_name == "ErrExperimentalAgentDisabled"' "$case_dir/error" >/dev/null
/opt/driver/sql.sh -readonly "$db" .dump >"$case_dir/after"
cmp "$case_dir/before" "$case_dir/after"
test ! -e "$HOME/.config/agent-director/launch"
test ! -e "$HOME/.config/agent-director/templates"
test ! -e "$HOME/.claude.json"

configure true
agent-director spawn --agent droid --cwd "$case_dir/project" -- --model first-model >"$case_dir/spawn"
jq -e '.pre_trust == "skipped"' "$case_dir/spawn" >/dev/null
id=$(jq -r .instance_id "$case_dir/spawn")
record="$AD_CODEX_RECORD_DIR/$id"
wait_for test -s "$record/pid"
agent-director get --instance-id "$id" >"$case_dir/get"
jq -e '.agent == "droid" and .state == "pending" and .integration_mode == "pane_only" and .idle_screen_availability == "unavailable" and (.context | all(.[]; . == null))' "$case_dir/get" >/dev/null
mapfile -d '' -t argv <"$record/argv"
test "${#argv[@]}" = 2
test "${argv[0]}" = --settings
settings="${argv[1]}"
test "$(stat -c %a "$settings")" = 600
test "$(stat -c %a "$(dirname "$settings")")" = 700
jq -e '. == {model:"first-model"}' "$settings" >/dev/null

# First inspect the shell prompt, then send the two literal lines once.
pane_ready() {
    agent-director read-pane --instance-id "$id" >"$case_dir/pane"
    jq -er '.pane | test("bash-[0-9.]+[$#]")' "$case_dir/pane" >/dev/null
}
wait_for pane_ready
text=$'printf "first\\n" > received\nprintf "second\\n" >> received'
agent-director send-keys --instance-id "$id" --allow-pending --text "$text" >"$case_dir/sent"
jq -e '. == {}' "$case_dir/sent" >/dev/null
printf 'first\nsecond\n' >"$case_dir/expected"
received() { cmp -s "$case_dir/expected" "$case_dir/project/received"; }
wait_for received
rc=0
agent-director pause --instance-id "$id" >"$case_dir/out" 2>"$case_dir/error" || rc=$?
test "$rc" = 1
jq -e '.err_name == "ErrAgentCapabilityUnsupported"' "$case_dir/error" >/dev/null
agent-director read-pane --instance-id "$id" >"$case_dir/pane"
agent-director send-keys --instance-id "$id" --allow-pending --text /exit >"$case_dir/sent"
wait_for test -s "$record/exit-hook.exit"
test "$(cat "$record/exit-hook.exit")" = 0
test ! -s "$record/exit-hook.out"
test "$(/opt/driver/sql.sh -readonly "$db" "SELECT state FROM spawns WHERE instance_id='$id';")" = pending
reconciled() {
    agent-director find-missing >"$case_dir/sweep"
    jq -e --arg id "$id" '.ids == [$id] and .count == 1' "$case_dir/sweep" >/dev/null
}
wait_for reconciled
agent-director get --instance-id "$id" | jq -e '.state == "missing"' >/dev/null

mv "$record" "$case_dir/first-record"
agent-director spawn --agent droid --instance-id "$id" --reuse-finished --cwd "$case_dir/project" -- --model second-model >"$case_dir/reuse"
jq -e --arg id "$id" '.instance_id == $id and .pre_trust == "skipped"' "$case_dir/reuse" >/dev/null
wait_for test -s "$record/pid"
mapfile -d '' -t argv <"$record/argv"
test "${#argv[@]}" = 2
test "${argv[0]}" = --settings
test "${argv[1]}" != "$settings"
jq -e '. == {model:"second-model"}' "${argv[1]}" >/dev/null
configure false
agent-director get --instance-id "$id" | jq -e '.state == "pending" and .integration_mode == "pane_only"' >/dev/null
wait_for pane_ready
agent-director kill --instance-id "$id" | jq -e '.kill_sent == true' >/dev/null
wait_for reconciled
agent-director get --instance-id "$id" | jq -e '.state == "missing" and .integration_mode == "pane_only"' >/dev/null
printf 'Droid stand-in lifecycle passed\n'
```
