# spawn-17: the reuse opt-in over a live row collides and changes nothing

Maps to SRD SR-20.9 spawn-17 and SR-10.2 (Epic 17 `t1.h98.i4`, Task
`t2.h98.i4.tm`, Subtask `t3.h98.i4.tm.pm`; spec `t3.h98.i4.y9.hq`): the
opt-in (`--reuse-finished`) applies only to a finished row. Over a live
row, `pending` included, `spawn --reuse-finished` is refused with
`ErrInstanceIdCollision` and nothing changes: the row (life, token and
`row_version` included), its running session and the session set stay as
they were, and no `ad.spawn.reused` record is written. The case checks this
twice: while X is `pending` (its agent has not reported in) and again once
it is `waiting`.

Agent and hooks (SR-20.9, SR-22.9; collector "Driver mode"): X's spawn and
both reuse attempts run with `PATH=/opt/driver/stand-in-claude:$PATH`
prefixed, so X's pane process is the stand-in's interactive bash, which
sends no hook by itself: X stays `pending` until the case types a
SessionStart into its pane with `/opt/driver/pane-hook.sh <id>
'<payload>'`. No hook is piped from the case's own shell.

Credential isolation (Epic 10's kill-10 pattern), before any tmux or
agent-director call: `ANTHROPIC_API_KEY` and `CLAUDE_CODE_OAUTH_TOKEN` are
unset and removed from the tmux server's global environment (tolerating no
server); after the launch, neither appears in
`/proc/<recorded pane_pid>/environ` ("credentials reached the agent").

## Setup

Per-case DB-reset. X's id is `sp17-<uuid>` (the trail is not reset
between cases, so records are selected by this id) and its session is
named `cd-sp17`; the reuse attempts ask for `cd-sp17r`, so a regression
that launched would make a second session the session-set check sees.
Both names are swept by the reset; an EXIT trap also ends every session
that was not there before by its session id.

1. Credential isolation as above.
2. `PATH=/opt/driver/stand-in-claude:$PATH agent-director spawn --cwd /tmp
   --tmux-session-name cd-sp17 --claude-instance-id X` → exit 0. Read the
   socket, pane, `pane_pid` and the pane's `$N`; check the pane environ.
   `status` X is `pending`.

## Steps

1. Capture X's full `spawns` row (`SELECT *` with a header) and the set of
   tmux session ids on X's socket.
2. `PATH=/opt/driver/stand-in-claude:$PATH agent-director spawn --cwd /tmp
   --tmux-session-name cd-sp17r --claude-instance-id X --reuse-finished`
   → exit 1, `err_name` `ErrInstanceIdCollision` (last JSON line on
   stderr).
3. The row read is byte-identical to the capture; X's session still runs
   by `$N`; the session set is unchanged; no `ad.spawn.reused` record for X.
4. Type a SessionStart into X's pane → a `sqlite3` read shows `waiting`
   and `pid` = the recorded `pane_pid`.
5. Repeat steps 1 to 3 on the `waiting` row.

## Pass criteria

- Setup: the spawn exits 0, X is `pending` and no credential variable is
  in the pane environ.
- Steps 2-3 and step 5, for the `pending` and then the `waiting` row:
  exit 1, `ErrInstanceIdCollision`; the full row (`life_number`,
  `launch_token` and `row_version` included) unchanged; X's session
  running by `$N`; the session set unchanged; no `ad.spawn.reused` record.
- Step 4: `waiting` with `pid` = `pane_pid`.

Each failed check prints a `FAIL:` line naming what was expected and what
was seen; the final line summarises what was checked.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN
tmux set-environment -g -u ANTHROPIC_API_KEY 2>/dev/null || true
tmux set-environment -g -u CLAUDE_CODE_OAUTH_TOKEN 2>/dev/null || true

db="$HOME/.agent-director/state.db"
trail="$HOME/.agent-director/ad-trail.jsonl"
standin_path="/opt/driver/stand-in-claude:$PATH"
uuid="$(cat /proc/sys/kernel/random/uuid)"
tag="${uuid:0:8}"
x="sp17-$uuid"
name=cd-sp17
name_r=cd-sp17r
transcript="/tmp/sp17-never-$tag.jsonl"
err_file="$(mktemp)"
sock=""
fail() { echo "FAIL: $*"; exit 1; }
session_set() { { tmux ${sock:+-S "$sock"} list-sessions -F '#{session_id}' 2>/dev/null || true; } | sort; }
before="$(session_set)"
cleanup() {
    local s
    for s in $(session_set); do
        case " $(printf '%s ' $before)" in
            *" $s "*) ;;
            *) tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true ;;
        esac
    done
    rm -f "$err_file"
}
trap cleanup EXIT

row_field() { sqlite3 -readonly "$db" "SELECT IFNULL($2,'') FROM spawns WHERE claude_instance_id = '$1';"; }
no_creds() {
    local v
    for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN; do
        if grep -qz "^$v=" "/proc/$1/environ"; then
            fail "credentials reached the agent: $v is in /proc/$1/environ ($2)"
        fi
    done
}
last_json() { printf '%s' "$1" | awk '/^\{/{ln=$0} END{print ln}'; }
full_row() { sqlite3 -readonly -header -nullvalue '<null>' -separator '|' "$db" "SELECT * FROM spawns WHERE claude_instance_id = '$x';"; }
trail_count() {
    [ -f "$trail" ] || { echo 0; return; }
    jq -c --arg e "$1" --arg x "$2" 'select(.event == $e and .claude_instance_id == $x)' "$trail" | grep -c . || true
}
# spawn_x <args...>: a stand-in spawn of X; stdout in $out, stderr in $err_file, exit status in $rc.
spawn_x() {
    set +e
    out="$(PATH="$standin_path" agent-director spawn --claude-instance-id "$x" "$@" 2>"$err_file")"
    rc=$?
    set -e
}
# refused_reuse <phase>: the reuse attempt over live X is refused and changes nothing.
refused_reuse() {
    local phase="$1" row_before set_before row_after set_after err_name n
    row_before="$(full_row)"
    set_before="$(session_set)"
    [ -n "$row_before" ] || fail "X's row is missing before the $phase reuse attempt"
    spawn_x --cwd /tmp --tmux-session-name "$name_r" --reuse-finished
    [ "$rc" -eq 1 ] || fail "spawn --reuse-finished of $phase X exited $rc; want 1: $(cat "$err_file")"
    err_name="$(last_json "$(cat "$err_file")" | jq -r '.err_name')"
    [ "$err_name" = "ErrInstanceIdCollision" ] || fail "spawn --reuse-finished of $phase X err_name=[$err_name]; want ErrInstanceIdCollision: $(cat "$err_file")"
    row_after="$(full_row)"
    [ "$row_after" = "$row_before" ] || fail "$phase X's row changed: [$row_after]; was [$row_before]"
    tmux ${sock:+-S "$sock"} has-session -t "$x_sid" 2>/dev/null || fail "$phase X's session $x_sid no longer runs"
    set_after="$(session_set)"
    [ "$set_after" = "$set_before" ] || fail "tmux session set changed on the $phase reuse attempt: [$set_after]; was [$set_before]"
    n="$(trail_count ad.spawn.reused "$x")"
    [ "$n" -eq 0 ] || fail "$n ad.spawn.reused record(s) for X after the $phase reuse attempt; want none"
}

# Setup 2: spawn X with the stand-in; it stays pending.
spawn_x --cwd /tmp --tmux-session-name "$name"
[ "$rc" -eq 0 ] || fail "spawn of X exited $rc; want 0: $(cat "$err_file")"
[ "$(jq -r '.claude_instance_id' <<<"$out")" = "$x" ] || fail "spawn of X echoed [$out]; want id $x"
sock="$(row_field "$x" tmux_socket)"
pane="$(row_field "$x" pane_id)"
pane_pid="$(row_field "$x" pane_pid)"
[ -n "$pane" ] && [[ "$pane_pid" =~ ^[0-9]+$ ]] || fail "X records pane_id|pane_pid [$pane|$pane_pid]; want both set"
x_sid="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}')"
[[ "$x_sid" =~ ^\$[0-9]+$ ]] || fail "X's pane $pane gave session id [$x_sid]"
no_creds "$pane_pid" "X after spawn"
st="$(agent-director status --claude-instance-id "$x" | jq -r '.state')"
[ "$st" = "pending" ] || fail "X after spawn state=$st; want pending (the stand-in reported in on its own?)"

# Steps 1-3: the reuse attempt over the pending row.
refused_reuse pending

# Step 4: SessionStart typed into X's pane turns it waiting.
/opt/driver/pane-hook.sh "$x" "{\"hook_event_name\":\"SessionStart\",\"source\":\"startup\",\"transcript_path\":\"$transcript\"}" >/dev/null \
    || fail "SessionStart hook for X exited non-zero"
got="$(sqlite3 -readonly "$db" "SELECT state || '|' || IFNULL(pid,'') FROM spawns WHERE claude_instance_id = '$x';")"
[ "$got" = "waiting|$pane_pid" ] || fail "X after SessionStart state|pid [$got]; want waiting|$pane_pid"

# Step 5: the same attempt over the waiting row.
refused_reuse waiting

echo "spawn-17 ok: live X ($x, session $x_sid) spawned with --reuse-finished while pending and while waiting -> ErrInstanceIdCollision both times; row, session and session set unchanged; no ad.spawn.reused"
```
