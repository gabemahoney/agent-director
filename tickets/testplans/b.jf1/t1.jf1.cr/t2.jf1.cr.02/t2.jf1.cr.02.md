# Test case: Blocking plan, independent hold and cancellation

Maps to E2 AC2; SR-4.1, SR-5.1/5.2, SR-9.2.

## Setup

The harness reset supplies its clean baseline. Source the shared driver, which creates an additional private synthetic Cursor namespace with an override store and control sentinel. All SQL, including raw-byte observer queries and faults, goes through /opt/driver/sql.sh. Source mount and shipped binary must match the scheduler's full commit. No authenticated runtime is used.

## Steps

Execute the ordered block below. Each check tests the stated caller-visible state, preserved fields, request correlation, committed evidence, actual wire bytes or process lifetime. Any injected fault is a test fixture action; it does not establish a production configuration feature.

## Pass criteria

Every assertion passes and one final JSON object contains this case ID, stable check names, status pass, authenticated_cursor false and separate variable facts. Do not treat an exit code alone as evidence. The collector must be executed externally before claiming a case pass.

```bash
set -euo pipefail
receipts=$(mktemp /tmp/cr-plan-receipts-XXXXXX)
for mode in early_end cancel_busy; do
(
source /work/source/test/driver/cursor-owned-launch.sh
relay_scenario plan
if [[ $mode == early_end ]]; then jq '.completeBeforeAnswers=true' "$case_dir/scenario" > "$case_dir/next"; mv "$case_dir/next" "$case_dir/scenario"; fi
relay_start
wait_until request_count_is 2
plan_tool=$(jq -r '.requests[]|select(.id=="plan")|.params.toolCallId' "$case_dir/scenario")
permission_tool=$(jq -r '.requests[]|select(.id=="permission")|.params.toolCall.toolCallId' "$case_dir/scenario")
plan_token=$(request_token "$plan_tool"); permission_token=$(request_token "$permission_tool")
wait_state check_permission
if [[ $mode == early_end ]]; then
 transcript=$(sql "SELECT jsonl_path FROM spawns WHERE instance_id='$id';")
 turn_ended() { jq -se 'any(.[]; .kind=="turn_outcome" and .data.stop_reason=="end_turn")' "$transcript" >/dev/null; }
 wait_until turn_ended
fi
[[ $(sql "SELECT count(*) FROM permission_requests WHERE instance_id='$id' AND proven_gone_at IS NULL;") == 2 ]]
relay_guard
relay_decide "$plan_token" deny >/dev/null
wait_until reply_count_is 1
frames | jq -se '[.[]|select(.id=="plan")]|length==1 and .[0].result.outcome.outcome=="rejected"' >/dev/null
[[ $(sql "SELECT decision IS NULL AND delivered_at IS NULL AND proven_gone_at IS NULL FROM permission_requests WHERE request_token='$permission_token';") == 1 ]]
# Real terminal cancellation retires the still-open permission; no fabricated answer.
if [[ $mode == cancel_busy ]]; then tmux -S "$socket" send-keys -t "$pane" C-c; else tmux -S "$socket" send-keys -t "$pane" -l -- /exit; tmux -S "$socket" send-keys -t "$pane" Enter; fi
wait_gone "$gateway"; wait_gone "$runtime"
cancels=$(frames | jq -sc '[.[]|select(.method=="session/cancel")]|length')
if [[ $mode == cancel_busy ]]; then [[ $cancels == 1 ]]; else [[ $cancels == 0 ]]; fi
[[ $(frames | jq -sc '[.[]|select(.id=="permission" and (has("method")|not))]|length') == 0 ]]
kill -0 "$control_sentinel"
relay_receipt t2.jf1.cr.02 '["AC2.blocking-plan","AC2.end-turn-held","AC2.independent-permission","AC2.reject","AC2.cancel-no-late-answer"]' >> "$receipts"
)
done
jq -s '{case:"t2.jf1.cr.02",status:"pass",checks:["AC2.blocking-plan","AC2.end-turn-held","AC2.independent-permission","AC2.reject","AC2.cancel-no-late-answer"],authenticated_cursor:false,facts:{runs:map(.facts)}}' "$receipts"
```
