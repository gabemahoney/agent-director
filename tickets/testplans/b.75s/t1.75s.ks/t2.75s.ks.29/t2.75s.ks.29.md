# kill-9: the finished-row option refuses a live row (pending, then waiting)

Maps to SRD SR-20.9 kill-9 (AC-KILL-16; Epic `t1.h98.9b`, Task
`t2.h98.9b.qv`; SR-6.5, SR-6.8, SR-1.4). `kill`'s operator-only
finished-row option applies only to an `ended` or `missing` row. On a
live row, `pending` included, it refuses right after the row read:
`ErrSpawnNotResumable`, exit 1, with a description that says the row
is live, makes no lookup and sends nothing. The description never names
a way to end a session (no tmux command, no `agent-director kill`, not
the option's own flag). X is spawned for real and checked twice: first
while `pending` (its agent has not reported in), then after a
SessionStart typed into its pane makes it `waiting`. Both times X's
session and its pane process still run afterwards and X's row is
unchanged.

## Setup

Per-case DB-reset. The spawn runs with
`PATH=/opt/driver/stand-in-claude:$PATH` prefixed to the
`agent-director` command, so X's pane process is the stand-in claude's
interactive bash, which sends no hook by itself. The one hook (a
SessionStart) is typed into X's recorded pane with
`/opt/driver/pane-hook.sh <id> '<payload>'`, so its parent is the pane
process (SR-22.9). No config file is written: the refusal needs none.
Before any tmux or agent-director call the script:

1. unsets `ANTHROPIC_API_KEY` and `CLAUDE_CODE_OAUTH_TOKEN`;
2. removes both from the global environment of a tmux server already
   answering on the default socket (`tmux set-environment -g -u`),
   tolerating "no server";
3. right after the spawn, before any state assertion, checks neither
   name appears in `/proc/<recorded pane_pid>/environ`, failing with
   "credentials reached the agent".

X's session is named `cd-k9` (the harness's reserved prefix, which the
DB-reset sweeps); an EXIT trap also removes it by its session id `$N`
on the recorded socket.

## Steps

1. `PATH=/opt/driver/stand-in-claude:$PATH agent-director spawn --cwd
   /tmp --tmux-session-name cd-k9 --claude-instance-id id-k9`.
2. Read X's `tmux_socket`, `pane_id` and `pane_pid` with `sqlite3`; get
   X's session id `$N` from the recorded pane's `#{session_id}` on that
   socket. Run the credential check. `status` X is `pending`.
3. `agent-director kill --include-finished --claude-instance-id id-k9`.
4. Type `{"hook_event_name":"SessionStart","transcript_path":"<path>"}`
   into X's pane with `pane-hook.sh` (as resume-8, `t2.75s.ue.5y`).
   X is `waiting` and its `pid` equals the recorded `pane_pid`.
5. Repeat step 3.

## Pass criteria

- Neither credential variable is in the pane process's environment.
- Each kill (on `pending`, then on `waiting`) exits 1; the last JSON
  line on stderr has `err_name` `ErrSpawnNotResumable`, and its
  `err_description` contains "the row is live", the row's state and
  "nothing was sent".
- The description contains none of `tmux `, `kill-session`,
  `kill-pane`, `kill-server`, `agent-director kill`,
  `--include-finished`.
- After each kill: `tmux -S <socket> has-session -t '<$N>'` succeeds,
  the recorded pane pid still runs (present, not a zombie), and X's row
  (`state`, `pid`, `pane_id`, `pane_pid`, `row_version`,
  `launch_token`, `ended_at`) is exactly as before the kill; `status`
  X is still `pending`, then still `waiting`.
- After the SessionStart, X is `waiting` with `pid` = `pane_pid`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN
tmux set-environment -g -u ANTHROPIC_API_KEY 2>/dev/null || true
tmux set-environment -g -u CLAUDE_CODE_OAUTH_TOKEN 2>/dev/null || true

db="$HOME/.agent-director/state.db"
standin_path="/opt/driver/stand-in-claude:$PATH"
id="id-k9"
fail() { echo "FAIL: $*"; exit 1; }
last_json() { printf '%s
' "$1" | awk '/^\{/{ln=$0} END{print ln}'; }
sock=""
sid=""
trap '[ -z "$sid" ] || tmux -S "$sock" kill-session -t "$sid" 2>/dev/null || true' EXIT

# row_snapshot: every column the refusal must leave alone.
row_snapshot() {
    sqlite3 -readonly -separator '|' "$db" \
        "SELECT state, IFNULL(pid,''), IFNULL(pane_id,''), IFNULL(pane_pid,''),
                row_version, IFNULL(launch_token,''), IFNULL(ended_at,'')
         FROM spawns WHERE claude_instance_id = '$id';"
}

# refused_kill <want state>: the opt-in kill of X is refused as a live row;
# nothing about X's session, pane process or row changes.
refused_kill() {
    local want="$1" before after err code envelope name desc bad st state
    before="$(row_snapshot)"
    [ "${before%%|*}" = "$want" ] || fail "before the opt-in kill X's row is [$before]; want state $want"
    set +e
    err="$(agent-director kill --include-finished --claude-instance-id "$id" 2>&1 >/dev/null)"
    code=$?
    set -e
    [ "$code" -eq 1 ] || fail "opt-in kill of $want X exited $code; want 1 (stderr: $err)"
    envelope="$(last_json "$err")"
    name="$(jq -r '.err_name' <<<"$envelope")"
    [ "$name" = "ErrSpawnNotResumable" ] || fail "err_name=$name on $want X; want ErrSpawnNotResumable (stderr: $err)"
    desc="$(jq -r '.err_description' <<<"$envelope")"
    for w in "the row is live" "(state $want)" "nothing was sent"; do
        case "$desc" in *"$w"*) ;; *) fail "description on $want X lacks [$w]: $desc" ;; esac
    done
    for bad in "tmux " kill-session kill-pane kill-server "agent-director kill" --include-finished; do
        case "$desc" in *"$bad"*) fail "description on $want X names [$bad]: $desc" ;; esac
    done
    tmux -S "$sock" has-session -t "$sid" 2>/dev/null \
        || fail "X's session $sid is gone after the refused kill of $want X"
    [ -e "/proc/$pane_pid" ] || fail "pane process $pane_pid is gone after the refused kill of $want X"
    st="$(sed -E 's/^.*\) //' "/proc/$pane_pid/stat" | cut -d' ' -f1)"
    [ "$st" != "Z" ] || fail "pane process $pane_pid is a zombie after the refused kill of $want X"
    after="$(row_snapshot)"
    [ "$after" = "$before" ] || fail "X's row changed on the refused kill: [$before] -> [$after]"
    state="$(agent-director status --claude-instance-id "$id" | jq -r '.state')"
    [ "$state" = "$want" ] || fail "status after the refused kill is $state; want $want"
}

# The spawn's stderr (a "pre-trust failed" note: no ~/.claude.json here) is
# shown only if the spawn fails, so the case's last line stays the summary.
spawn_err="$(mktemp)"
spawn_out="$(PATH="$standin_path" agent-director spawn --cwd /tmp --tmux-session-name cd-k9 \
    --claude-instance-id "$id" 2>"$spawn_err")" \
    || fail "spawn exited non-zero: $(cat "$spawn_err")"
got_id="$(jq -r '.claude_instance_id' <<<"$spawn_out")"
[ "$got_id" = "$id" ] || fail "spawn returned claude_instance_id [$got_id]; want $id: $spawn_out"

IFS='|' read -r sock pane pane_pid < <(sqlite3 -readonly "$db" \
    "SELECT IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'') || '|' || IFNULL(pane_pid,'') FROM spawns WHERE claude_instance_id = '$id';")
[ -n "$sock" ] && [ -n "$pane" ] && [[ "$pane_pid" =~ ^[0-9]+$ ]] \
    || fail "recorded tmux_socket|pane_id|pane_pid=[$sock|$pane|$pane_pid]; want all set"
sid="$(tmux -S "$sock" display-message -p -t "$pane" '#{session_id}')"
[[ "$sid" =~ ^\$[0-9]+$ ]] || fail "pane $pane gave session id [$sid]"

for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN; do
    if grep -qz "^$v=" "/proc/$pane_pid/environ"; then
        fail "credentials reached the agent: $v is in /proc/$pane_pid/environ"
    fi
done

state="$(agent-director status --claude-instance-id "$id" | jq -r '.state')"
[ "$state" = "pending" ] || fail "after spawn state=$state; want pending (the agent reported in on its own?)"

# Step 3: refused on pending.
refused_kill pending

# Step 4: X reports in through its own pane process.
jsonl_path="$HOME/.claude/projects/-tmp/k9-session-uuid.jsonl"
/opt/driver/pane-hook.sh "$id" \
    "{\"hook_event_name\":\"SessionStart\",\"transcript_path\":\"${jsonl_path}\"}" >/dev/null \
    || fail "SessionStart typed into $pane failed"
IFS='|' read -r state pid < <(sqlite3 -readonly "$db" \
    "SELECT state || '|' || IFNULL(pid,'') FROM spawns WHERE claude_instance_id = '$id';")
[ "$state" = "waiting" ] || fail "after SessionStart state=$state; want waiting"
[ "$pid" = "$pane_pid" ] || fail "after SessionStart pid=[$pid]; want the recorded pane_pid $pane_pid"

# Step 5: refused on waiting.
refused_kill waiting

echo "kill-9 ok: opt-in kill of X ($id) refused with ErrSpawnNotResumable on pending and on waiting; session $sid and pane process $pane_pid still run; row unchanged"
```
