# agent-24: Claude Codex Claude switching drops long and short resume arguments and starts fresh

Maps to SRD SR-6.8.9 Docker agent-24 and Subtask `t3.9r1.dn.ro.o8`.

## Setup

The driver runs `db-reset.sh`. Configure both existing stand-ins. Each
variant has an independent id, S1/S2 transcripts, Codex session and
`CODEX_HOME`; S1 stays on disk throughout. Isolate Claude config and cwd,
remove credentials, restore launch config and clean recorded sessions.

## Steps

For each `--resume S1`, `-r S1` and `-c`:

1. Start Claude, seed/report S1 from its pane, then end and wait for its
   recorded processes within 5 s, as resume-9 does.
2. Reuse as Codex; report SessionStart with `pane-hook.sh --agent codex`
   before pause. Pause within 30 s and verify its successful end hook and
   recorded processes absent within 5 s after pause, as agent-14 does.
3. Reuse as Claude with the variant after `--` and `--no-pre-trust`.
   Read the fresh pane's `#{pane_start_command}`; check exact dropped
   original arguments in the result and final `ad.spawn.reused` trail.
4. Report new Claude S2 through `pane-hook.sh` and read get. End this
   stand-in and wait before the next independent variant.

## Pass criteria

Every launch exits 0. Each switch increments life once (two total).
Final Claude launches the simple interactive stand-in without long/short
resume options or S1 in its tmux command; before its report it is pending
with no session or history. Exact `dropped_args` equals the variant's
original tokens in the spawn result and final reuse trail (the first
reuse has null drops). After report-in, get says Claude, waiting, S2 and
`prior_sessions` `[]`, while S1's transcript remains on disk. Both Codex
and fresh Claude report through recorded-pane helpers; no real agents,
login or network are used. Bounded cleanup restores the config on failure.

## Script

```bash
set -euo pipefail
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY CODEX_HOME CLAUDE_CONFIG_DIR AD_CODEX_HOOK AD_CODEX_RECORD_DIR AGENT_DIRECTOR_INSTANCE_ID
for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
    tmux set-environment -g -u "$v" 2>/dev/null || true
done
db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
work="$(mktemp -d /tmp/agent-24-XXXXXX)"
ids=()
had_config=false
fail() { echo "FAIL: $*"; exit 1; }
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }
cleanup() {
    local x sock pane target
    for x in "${ids[@]}"; do
        sock="$(col "$x" "IFNULL(tmux_socket,'')" 2>/dev/null || true)"
        pane="$(col "$x" "IFNULL(pane_id,'')" 2>/dev/null || true)"
        target="$(timeout 5s tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
        [ -z "$target" ] || timeout 5s tmux ${sock:+-S "$sock"} kill-session -t "$target" 2>/dev/null || true
        rm -rf "/tmp/stand-in-codex/$x"
    done
    if $had_config; then cp -p "$work/config.before" "$config"; else rm -f "$config"; fi
    rm -rf "$work"
}
trap cleanup EXIT
if [ -e "$config" ]; then had_config=true; cp -p "$config" "$work/config.before"; fi
mkdir -p "$HOME/.config/agent-director" "$work/cwd" "$work/claude"
printf '[launch]\nclaude_path = "/opt/driver/stand-in-claude/claude"\ncodex_path = "/opt/driver/stand-in-codex/codex"\n' > "$config"
uuid="$(cat /proc/sys/kernel/random/uuid)"
# All hook payloads are typed into the row's recorded pane, never this shell.
fire() {
    local x="$1" agent="$2" event="$3" session="$4" transcript="$5" extra='{}' body out
    [ $# -lt 6 ] || extra="$6"
    body="$(jq -cn --arg e "$event" --arg s "$session" --arg t "$transcript" --arg c "$work/cwd" --argjson x "$extra" \
        '{hook_event_name:$e,session_id:$s,transcript_path:$t,cwd:$c} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent "$agent" "$x" "$body")" || fail "$event of $x failed"
    [ -z "$out" ] || fail "$event of $x printed [$out]; want empty stdout"
}
spawn_row() {
    local x="$1" agent="$2" name="$3" out
    shift 3
    ids+=("$x")
    out="$(agent-director spawn --agent "$agent" --instance-id "$x" --cwd "$work/cwd" --tmux-session-name "$name" \
        --extra-env "CODEX_HOME=$codex_home" --extra-env "CLAUDE_CONFIG_DIR=$work/claude" "$@")" || fail "spawn $agent $x failed"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed wrong id: $out"
    spawn_out="$out"
    local pane_pid deadline
    pane_pid="$(col "$x" pane_pid)"
    deadline=$(( $(date +%s) + 10 ))
    until [ "$(cat /proc/"$pane_pid"/comm 2>/dev/null || true)" = bash ]; do
        [ "$(date +%s)" -lt "$deadline" ] || fail "$x stand-in did not become interactive bash within 10 s"
        sleep 0.1
    done
    for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
        if grep -qz "^$v=" "/proc/$pane_pid/environ"; then fail "credentials reached $x pane"; fi
    done
    if [ "$agent" = codex ]; then
        until [ -s "/tmp/stand-in-codex/$x/pid" ]; do
            [ "$(date +%s)" -lt "$deadline" ] || fail "no Codex record for $x within 10 s"
            sleep 0.1
        done
        [ "$(cat /tmp/stand-in-codex/"$x"/pid)" = "$pane_pid" ] || fail "Codex record differs from pane pid"
    fi
}
end_and_wait() {
    local id="$1" target="$2" rec rpid rpane rserver rsock n left p end
    rec="$(/opt/driver/sql.sh -readonly "$db" "SELECT IFNULL(pid,'') || '|' || IFNULL(pane_pid,'') || '|' || IFNULL(tmux_server_pid,'') || '|' || IFNULL(tmux_socket,'') FROM spawns WHERE instance_id = '$id';")"
    IFS='|' read -r rpid rpane rserver rsock <<<"$rec"
    [[ "$rpid|$rpane|$rserver" =~ ^[0-9]+\|[0-9]+\|[0-9]+$ ]] || fail "$id records pid|pane_pid|tmux_server_pid [$rpid|$rpane|$rserver]; want all set"
    n="$(tmux ${rsock:+-S "$rsock"} list-sessions 2>/dev/null | wc -l)"
    timeout 5s tmux ${rsock:+-S "$rsock"} kill-session -t "$target" || fail "kill-session -t $target of $id failed"
    set -- "$rpid" "$rpane"
    [ "$n" -gt 1 ] || set -- "$@" "$rserver"
    end=$(( $(date +%s%N) + 5000000000 ))
    while :; do
        left=""
        for p in "$@"; do [ ! -e "/proc/$p" ] || left+=" $p"; done
        [ -n "$left" ] || break
        [ "$(date +%s%N)" -lt "$end" ] || fail "$id's recorded process(es)$left still running 5 s after its session $target ended"
        sleep 0.1
    done
    [ "$n" -le 1 ] || [ -e "/proc/$rserver" ] || fail "tmux server $rserver of $id ended although it held $((n - 1)) other session(s)"
}
finish_claude() {
    local x="$1" session="$2" transcript="$3" sock pane target
    fire "$x" claude SessionEnd "$session" "$transcript" '{"reason":"logout"}'
    [ "$(col "$x" state)" = ended ] || fail "Claude SessionEnd did not end $x"
    sock="$(col "$x" "IFNULL(tmux_socket,'')")"
    pane="$(col "$x" pane_id)"
    target="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}')"
    end_and_wait "$x" "$target"
}

# Each spelling gets an independent id, transcript, Codex folder and three launches.
for spelling in long short continue; do
    id="ag24-$spelling-$uuid"
    codex_home="$work/codex-$spelling"
    mkdir -p "$codex_home"
    s1="ag24-s1-$spelling-$uuid"
    s2="ag24-s2-$spelling-$uuid"
    s1_file="$work/claude/projects/case/$s1.jsonl"
    s2_file="$work/claude/projects/case/$s2.jsonl"
    mkdir -p "$(dirname "$s1_file")"
    printf '{}\n' > "$s1_file"
    printf '{}\n' > "$s2_file"
    spawn_row "$id" claude "cd-ag24-$spelling"
    fire "$id" claude SessionStart "$s1" "$s1_file" '{"source":"startup"}'
    [ "$(col "$id" session_id)" = "$s1" ] || fail "$spelling setup did not report S1"
    finish_claude "$id" "$s1" "$s1_file"
    life0="$(col "$id" life_number)"
    spawn_row "$id" codex "cd-ag24-$spelling" --reuse-finished
    cs="$(cat /proc/sys/kernel/random/uuid)"
    rollout="$codex_home/sessions/2026/10/13/rollout-2026-10-13T10-00-00-$cs.jsonl"
    mkdir -p "$(dirname "$rollout")"
    printf '{"type":"session_meta"}\n' > "$rollout"
    printf '%s\n' "$cs" > "/tmp/stand-in-codex/$id/session-id"
    fire "$id" codex SessionStart "$cs" "$rollout" '{"source":"startup"}'
    [ "$(col "$id" state)" = waiting ] || fail "$spelling Codex did not report before pause"
    [ "$(col "$id" life_number)" -eq "$((life0 + 1))" ] || fail "$spelling first switch life"
    # Save identities before pause, then require every process to exit within 5 s.
    rec="$(col "$id" "pid || '|' || pane_pid || '|' || tmux_server_pid")"
    IFS='|' read -r rpid rpane rserver <<<"$rec"
    sock="$(col "$id" "IFNULL(tmux_socket,'')")"
    count="$(tmux ${sock:+-S "$sock"} list-sessions | wc -l)"
    paused="$(timeout 30s agent-director pause --instance-id "$id")" || fail "$spelling Codex pause failed or exceeded 30 s"
    [ "$(jq -c . <<<"$paused")" = '{}' ] || fail "$spelling pause result: $paused"
    [ "$(col "$id" state)" = ended ] || fail "$spelling Codex not ended after pause"
    [ "$(cat /tmp/stand-in-codex/"$id"/exit-hook.exit)" = 0 ] || fail "$spelling Codex exit hook failed"
    [[ "$rpid|$rpane|$rserver" =~ ^[0-9]+\|[0-9]+\|[0-9]+$ ]] || fail "$spelling missing recorded process ids"
    pids=("$rpid" "$rpane")
    [ "$count" -gt 1 ] || pids+=("$rserver")
    deadline=$(( $(date +%s%N) + 5000000000 ))
    while :; do
        left=""
        for p in "${pids[@]}"; do [ ! -e "/proc/$p" ] || left+=" $p"; done
        [ -n "$left" ] || break
        [ "$(date +%s%N)" -lt "$deadline" ] || fail "$spelling Codex processes$left still present 5 s after pause"
        sleep 0.1
    done
    case "$spelling" in
        long) args=(--resume "$s1");;
        short) args=(-r "$s1");;
        continue) args=(-c);;
    esac
    expected="$(printf '%s\n' "${args[@]}" | jq -Rsc 'split("\n")[:-1]')"
    # The final launch uses the simple interactive stand-in, without pre-trust.
    spawn_row "$id" claude "cd-ag24-$spelling" --reuse-finished --no-pre-trust -- "${args[@]}"
    jq -e --argjson expected "$expected" '.dropped_args == $expected' <<<"$spawn_out" >/dev/null || fail "$spelling result did not report exact originals: $spawn_out"
    [ "$(col "$id" life_number)" -eq "$((life0 + 2))" ] || fail "$spelling switch back did not start another life"
    sock="$(col "$id" "IFNULL(tmux_socket,'')")"
    pane="$(col "$id" pane_id)"
    command="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{pane_start_command}')"
    [[ "$command" == *'/opt/driver/stand-in-claude/claude'* ]] || fail "$spelling launched wrong Claude command: $command"
    [[ "$command" != *'--resume'* && "$command" != *'--continue'* && "$command" != *"$s1"* ]] || fail "$spelling launched old resume command: $command"
    # Match short forms as whole shell words; settings JSON may contain -r text.
    if [[ "$command" =~ (^|[[:space:]\'])-r([[:space:]\']|$) || "$command" =~ (^|[[:space:]\'])-c([[:space:]\']|$) ]]; then
        fail "$spelling launched a short resume option: $command"
    fi
    row="$(agent-director get --instance-id "$id")"
    jq -e '.agent == "claude" and .state == "pending" and .prior_sessions == [] and ((.session_id // "") == "")' <<<"$row" >/dev/null \
        || fail "$spelling final launch inherited a session: $row"
    fire "$id" claude SessionStart "$s2" "$s2_file" '{"source":"startup"}'
    row="$(agent-director get --instance-id "$id")"
    jq -e --arg sid "$s2" '.agent == "claude" and .state == "waiting" and .session_id == $sid and .prior_sessions == []' <<<"$row" >/dev/null \
        || fail "$spelling final Claude did not report fresh S2: $row"
    reuses="$(jq -sc --arg id "$id" '[.[] | select(.event == "ad.spawn.reused" and .instance_id == $id)]' "$trail")"
    jq -e --argjson expected "$expected" --arg cs "$cs" \
        'length == 2 and .[0].dropped_args == null and .[1].dropped_args == $expected and .[1].archived_session_id == $cs and .[1].source == "ad_spawn"' \
        <<<"$reuses" >/dev/null || fail "$spelling trail did not report exact dropped originals: $reuses"
    [ -f "$s1_file" ] || fail "$spelling original transcript disappeared"
    finish_claude "$id" "$s2" "$s2_file"
    echo "variant $spelling ok: independent Claude S1 -> Codex reported/paused -> fresh Claude S2, life +2, exact dropped originals in result/trail, old transcript present and history empty"
done
echo "agent-24 ok: --resume S1, -r S1 and -c all start fresh across Claude/Codex/Claude switches"
```
