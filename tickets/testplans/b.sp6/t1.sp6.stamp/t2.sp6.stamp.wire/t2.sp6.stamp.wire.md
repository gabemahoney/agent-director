# get and list show the stored raw state change time exactly

Maps to Bug b.k6i: get and list expose the stored raw `state_changed_ms` as
the required `state_changed_at` on every row, all agents and pending
included; status does not.

## Setup

The DB-reset fixture gives a fresh store at the binary's current schema and
no `config.toml` (no settle time). All SQL uses `/opt/driver/sql.sh`.

## Steps

1. Insert three spawns, each with `settled_state` equal to its `state`:
   `stamp-claude` (claude, `pending`, `state_changed_ms` 1767225600123),
   `stamp-codex` (codex, `waiting`, 0) and `stamp-cursor` (cursor, `ended`,
   1767225600456).
2. Run `agent-director list`, then `get` and `status` for each id.
3. Read `instance_id` and `state_changed_ms` of every spawn from the store.

## Pass criteria

- `get` and `list` give each row's `state_changed_at` as exactly
  `2026-01-01T00:00:00.123Z`, `1970-01-01T00:00:00Z` and
  `2026-01-01T00:00:00.456Z` respectively, and its seeded `state`.
- `list` returns exactly the 3 rows.
- No `status` result has a `state_changed_at` key.
- The stored `state_changed_ms` values are still the seeded ones.
- The evidence line (at most 200 characters, so the driver keeps it whole)
  is built from the observed outputs, keyed by agent, as
  `[get state_changed_at, list's equals get's, status has the key, stored
  state_changed_ms after the reads]`.

```bash
set -euo pipefail
db="$HOME/.config/agent-director/state.db"
ids=(stamp-claude stamp-codex stamp-cursor)
declare -A agent=([stamp-claude]=claude [stamp-codex]=codex [stamp-cursor]=cursor)
declare -A state=([stamp-claude]=pending [stamp-codex]=waiting [stamp-cursor]=ended)
declare -A ms=([stamp-claude]=1767225600123 [stamp-codex]=0 [stamp-cursor]=1767225600456)
declare -A want=([stamp-claude]=2026-01-01T00:00:00.123Z [stamp-codex]=1970-01-01T00:00:00Z [stamp-cursor]=2026-01-01T00:00:00.456Z)
for id in "${ids[@]}"; do
    /opt/driver/sql.sh "$db" "INSERT INTO spawns(instance_id,state,settled_state,state_changed_ms,cwd,agent,relay_mode,tmux_session_name)
        VALUES('$id','${state[$id]}','${state[$id]}',${ms[$id]},'/tmp','${agent[$id]}','off','ts-$id')"
done
list="$(agent-director list)"
if [[ "$(jq '.spawns | length' <<<"$list")" != 3 ]]; then
    echo "list returned $(jq -c '[.spawns[].instance_id]' <<<"$list"); want the 3 seeded rows" >&2
    exit 1
fi
seen='{}'
for id in "${ids[@]}"; do
    get="$(agent-director get --instance-id "$id")"
    row="$(jq -c --arg id "$id" '.spawns[] | select(.instance_id == $id)' <<<"$list")"
    for doc in "get:$get" "list:$row"; do
        verb="${doc%%:*}"
        body="${doc#*:}"
        if ! jq -e --arg at "${want[$id]}" --arg st "${state[$id]}" '.state_changed_at == $at and .state == $st' <<<"$body" >/dev/null; then
            echo "$verb $id: state_changed_at $(jq -c .state_changed_at <<<"$body"), state $(jq -c .state <<<"$body"); want ${want[$id]}, ${state[$id]}" >&2
            exit 1
        fi
    done
    status="$(agent-director status --instance-id "$id")"
    seen="$(jq -c --arg a "${agent[$id]}" --argjson g "$get" --argjson r "$row" --argjson s "$status" \
        '. + {($a): [$g.state_changed_at, ($r.state_changed_at == $g.state_changed_at), ($s | has("state_changed_at"))]}' <<<"$seen")"
    if jq -e 'has("state_changed_at")' <<<"$status" >/dev/null; then
        echo "status $id carries state_changed_at: $status" >&2
        exit 1
    fi
done
stored="$(/opt/driver/sql.sh -readonly -separator '|' "$db" "SELECT instance_id, state_changed_ms FROM spawns ORDER BY instance_id")"
if [[ "$stored" != $'stamp-claude|1767225600123\nstamp-codex|0\nstamp-cursor|1767225600456' ]]; then
    echo "stored stamps changed by the reads: $stored" >&2
    exit 1
fi
while IFS='|' read -r id stored_ms; do
    seen="$(jq -c --arg a "${agent[$id]}" --argjson m "$stored_ms" '.[$a] += [$m]' <<<"$seen")"
done <<<"$stored"
jq -cS . <<<"$seen"
```
