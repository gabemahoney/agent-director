# agent-22: Codex manual subagent lifecycle hooks only soft-refresh the row

Maps to SRD SR-6.8.4 and Subtask `t3.9r1.dn.ro.o8`.

## Setup

The driver runs `db-reset.sh`. Use the existing Codex stand-in through
`[launch] codex_path`, a unique id, cwd, rollout and per-case `CODEX_HOME`.
Save and restore any config. Remove credentials from the case and tmux;
cleanup kills only recorded case sessions and removes case files.

## Steps

1. Spawn Codex and inspect its recorded argv for absent SubagentStart and
   SubagentStop registrations. Agent-9 already owns the eight-hook count.
2. Report SessionStart from its recorded pane with `pane-hook.sh --agent
   codex`; verify `waiting`.
3. Type SubagentStart and SubagentStop through the same helper, each with
   `agent_id` and `agent_type`. Check every predicate after each event.

## Pass criteria

Spawn and both hooks exit 0; hooks print nothing. No lifecycle registration
occurs. Each manual event keeps state, increments `row_version` by exactly
one, emits exactly one matching `ad.spawn.state_transition` with
`soft_refresh` true and unchanged prior/new state, and leaves both
`subagents` and `subagent_launches` empty for the id. After each,
`list-subagents` is exactly `{"tracked":false,"subagents":[]}` (key order
ignored). All hooks come from the recorded pane; no real Codex is run.

## Script

```bash
set -euo pipefail
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY CODEX_HOME CLAUDE_CONFIG_DIR AD_CODEX_HOOK AD_CODEX_RECORD_DIR AGENT_DIRECTOR_INSTANCE_ID
for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
    tmux set-environment -g -u "$v" 2>/dev/null || true
done
db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
work="$(mktemp -d /tmp/agent-22-XXXXXX)"
ids=()
had_config=false
fail() { echo "FAIL: $*"; exit 1; }
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }
cleanup() {
    local x sock pane target
    for x in "${ids[@]}"; do
        sock="$(col "$x" "IFNULL(tmux_socket,'')" 2>/dev/null || true)"
        pane="$(col "$x" "IFNULL(pane_id,'')" 2>/dev/null || true)"
        target="$(timeout 5s tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
        [ -z "$target" ] || timeout 5s tmux ${sock:+-S "$sock"} kill-session -t "$target" 2>/dev/null || true
        rm -rf "/tmp/stand-in-codex/$x"
    done
    if $had_config; then cp -p "$work/config.before" "$config"; else rm -f "$config"; fi
    rm -rf "$work"
}
trap cleanup EXIT
if [ -e "$config" ]; then had_config=true; cp -p "$config" "$work/config.before"; fi
mkdir -p "$HOME/.config/agent-director" "$work/cwd" "$work/claude"
printf '[launch]\nclaude_path = "/opt/driver/stand-in-claude/claude"\ncodex_path = "/opt/driver/stand-in-codex/codex"\n' > "$config"
uuid="$(cat /proc/sys/kernel/random/uuid)"
# All hook payloads are typed into the row's recorded pane, never this shell.
fire() {
    local x="$1" agent="$2" event="$3" session="$4" transcript="$5" extra='{}' body out
    [ $# -lt 6 ] || extra="$6"
    body="$(jq -cn --arg e "$event" --arg s "$session" --arg t "$transcript" --arg c "$work/cwd" --argjson x "$extra" \
        '{hook_event_name:$e,session_id:$s,transcript_path:$t,cwd:$c} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent "$agent" "$x" "$body")" || fail "$event of $x failed"
    [ -z "$out" ] || fail "$event of $x printed [$out]; want empty stdout"
}
spawn_row() {
    local x="$1" agent="$2" name="$3" out
    shift 3
    ids+=("$x")
    out="$(agent-director spawn --agent "$agent" --instance-id "$x" --cwd "$work/cwd" --tmux-session-name "$name" \
        --extra-env "CODEX_HOME=$codex_home" --extra-env "CLAUDE_CONFIG_DIR=$work/claude" "$@")" || fail "spawn $agent $x failed"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed wrong id: $out"
    spawn_out="$out"
    local pane_pid deadline
    pane_pid="$(col "$x" pane_pid)"
    deadline=$(( $(date +%s) + 10 ))
    until [ "$(cat /proc/"$pane_pid"/comm 2>/dev/null || true)" = bash ]; do
        [ "$(date +%s)" -lt "$deadline" ] || fail "$x stand-in did not become interactive bash within 10 s"
        sleep 0.1
    done
    for v in ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY; do
        if grep -qz "^$v=" "/proc/$pane_pid/environ"; then fail "credentials reached $x pane"; fi
    done
    if [ "$agent" = codex ]; then
        until [ -s "/tmp/stand-in-codex/$x/pid" ]; do
            [ "$(date +%s)" -lt "$deadline" ] || fail "no Codex record for $x within 10 s"
            sleep 0.1
        done
        [ "$(cat /tmp/stand-in-codex/"$x"/pid)" = "$pane_pid" ] || fail "Codex record differs from pane pid"
    fi
}

id="ag22-$uuid"
codex_home="$work/codex"
session="$(cat /proc/sys/kernel/random/uuid)"
rollout="$codex_home/sessions/2026/10/13/rollout-2026-10-13T10-00-00-$session.jsonl"
mkdir -p "$(dirname "$rollout")"
printf '{"type":"session_meta"}\n' > "$rollout"
spawn_row "$id" codex cd-ag22
argv="$(tr '\0' '\n' < "/tmp/stand-in-codex/$id/argv")"
[[ "$argv" != *SubagentStart* && "$argv" != *SubagentStop* ]] || fail "Codex launch registered lifecycle hooks: $argv"
fire "$id" codex SessionStart "$session" "$rollout" '{"source":"startup"}'
[ "$(col "$id" state)" = waiting ] || fail "Codex report-in did not give waiting"
for event in SubagentStart SubagentStop; do
    before_state="$(col "$id" state)"
    before_version="$(col "$id" row_version)"
    fire "$id" codex "$event" "$session" "$rollout" '{"agent_id":"manual-subagent","agent_type":"worker"}'
    [ "$(col "$id" state)" = "$before_state" ] || fail "$event changed main state"
    [ "$(col "$id" row_version)" -eq "$((before_version + 1))" ] || fail "$event did not advance row_version by exactly one"
    for table in subagents subagent_launches; do
        [ "$(/opt/driver/sql.sh -readonly "$db" "SELECT COUNT(*) FROM $table WHERE instance_id = '$id';")" = 0 ] || fail "$event wrote $table"
    done
    transitions="$(jq -sc --arg id "$id" --arg event "$event" \
        '[.[] | select(.event == "ad.spawn.state_transition" and .instance_id == $id and .triggering_event_name == $event)]' "$trail")"
    jq -e --arg state "$before_state" 'length == 1 and .[0].soft_refresh == true and .[0].prior_state == $state and .[0].new_state == $state' \
        <<<"$transitions" >/dev/null || fail "$event did not emit one matching soft refresh: $transitions"
    result="$(agent-director list-subagents --instance-id "$id" | jq -cS .)"
    [ "$result" = '{"subagents":[],"tracked":false}' ] || fail "$event list-subagents: $result"
done
echo "agent-22 ok: codex_path launch has no lifecycle registrations; both manual lifecycle events kept state, advanced version once, emitted one soft refresh and wrote no entry/claim; untracked empty"
```
