# agent-13: the Codex hook states on a codex row

Maps to SRD SR-6.1, SR-6.3 and SR-14.5 of Epic 13 (`b.ssz`): a
hook moves a row only when the `agent-director hook --agent codex` process
is a direct child of the row's pane process, and then by the Codex state
table: `SessionStart` gives `waiting` and writes the payload's `session_id`
and rollout path (`transcript_path`), `UserPromptSubmit` gives `working`,
`Stop` gives `waiting`, and `Interrupt` (Codex's Esc) gives `waiting` from
`working`.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex` named by `[launch] codex_path`,
which ends as an interactive bash under the pane's pid. The case types each
hook into it with `/opt/driver/pane-hook.sh --agent codex`, which types
`agent-director hook --agent codex < payload > out 2> err` and waits for its
exit status. Each payload is a small inline JSON object whose
`transcript_path` is a rollout under a temporary `CODEX_HOME`.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"`, makes a
cwd and a temporary Codex folder holding the one rollout file the payloads
name, and removes them, the config and the stand-in's record directory on
exit.

## Steps

1. Write the config; make the cwd and the Codex folder with its rollout.
2. `agent-director spawn --agent codex --cwd <cwd> --instance-id <id>
   --tmux-session-name cd-ag-hk --extra-env CODEX_HOME=<folder>`; wait for
   the stand-in's record; `get` shows `pending`.
3. Type a `SessionStart` (source `startup`) with `pane-hook.sh --agent
   codex`.
4. Type `UserPromptSubmit`, then `Stop`, then `UserPromptSubmit`, then
   `Interrupt`, checking the row's state after each.
5. Read the trail's `ad.hook.ignored` records for the row.

## Pass criteria

- Step 2: spawn exits 0 and the row is `pending` with agent `codex`.
- Step 3: the hook exits 0 and prints nothing; the row is `waiting`; `get`
  shows agent `codex`, `session_id` equal to the payload's, and
  `jsonl_path` equal to the rollout path; `end_reported_at` is NULL.
- Step 4, after each hook (exit 0, no output): `working`, `waiting`,
  `working`, `waiting`, in that order; `session_id` still the payload's.
- Step 5: no `ad.hook.ignored` record names the row.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
id="ag-hk-$uuid"
sid="$(cat /proc/sys/kernel/random/uuid)"
codex_home=""
cwd=""
rollout=""
fail() { echo "FAIL: $*"; exit 1; }

col() { /opt/driver/sql.sh -readonly "$db" "SELECT $1 FROM spawns WHERE instance_id = '$id';"; }

cleanup() {
    local row sock pane s
    row="$(col "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
    sock="${row%%|*}"
    pane="${row#*|}"
    if [ -n "$pane" ]; then
        s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
        [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
    fi
    rm -f "$config"
    rm -rf "$rec_root/$id" ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# wait_state WANT: the row's state (get) is WANT within 10 s.
wait_state() {
    local want="$1" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(agent-director get --instance-id "$id" | jq -r '.state')"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $id state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# fire EVENT [EXTRA_JSON]: type the Codex hook EVENT into the row's pane; the
# hook must exit 0 and print nothing.
fire() {
    local extra='{}' body out
    [ $# -lt 2 ] || extra="$2"
    body="$(jq -cn --arg e "$1" --arg sid "$sid" --arg tp "$rollout" --arg cwd "$cwd" --argjson x "$extra" \
        '{hook_event_name: $e, session_id: $sid, transcript_path: $tp, cwd: $cwd, model: "codex-stand-in", permission_mode: "default"} + $x')"
    out="$(/opt/driver/pane-hook.sh --agent codex "$id" "$body")" || fail "the $1 hook exited non-zero"
    [ -z "$out" ] || fail "the $1 hook printed [$out]; want nothing"
}

# Step 1: the config, the cwd, and a Codex folder with the rollout the payloads name.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-hk-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"
codex_home="$(mktemp -d)"
rollout="$codex_home/sessions/2026/10/12/rollout-2026-10-12T10-00-00-$sid.jsonl"
mkdir -p "$(dirname "$rollout")"
printf '{"type":"session_meta"}\n' > "$rollout"

# Step 2: spawn; the row is pending until the first hook.
out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$id" --tmux-session-name cd-ag-hk \
    --extra-env "CODEX_HOME=$codex_home")" || fail "spawn exited non-zero"
[ "$(jq -r '.instance_id' <<<"$out")" = "$id" ] || fail "spawn echoed [$out]; want id $id"
end=$(( $(date +%s%N) + 10000000000 ))
until [ -s "$rec_root/$id/pid" ]; do
    [ "$(date +%s%N)" -lt "$end" ] || fail "no stand-in record at $rec_root/$id within 10 s"
    sleep 0.1
done
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '[.agent, .state] | join("|")' <<<"$row")" = "codex|pending" ] || fail "row is not codex|pending: $row"

# Step 3: SessionStart gives waiting, the payload's session id and the rollout.
fire SessionStart '{"source":"startup"}'
wait_state waiting
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '.agent' <<<"$row")" = "codex" ] || fail "agent is [$(jq -r '.agent' <<<"$row")]; want codex"
[ "$(jq -r '.session_id' <<<"$row")" = "$sid" ] || fail "session_id is [$(jq -r '.session_id' <<<"$row")]; want $sid"
[ "$(jq -r '.jsonl_path' <<<"$row")" = "$rollout" ] || fail "jsonl_path is [$(jq -r '.jsonl_path' <<<"$row")]; want $rollout"
[ -z "$(col "IFNULL(end_reported_at, '')")" ] || fail "end_reported_at is set after a SessionStart"

# Step 4: the turn states.
fire UserPromptSubmit '{"prompt":"say hello","turn_id":"turn-1"}'
wait_state working
fire Stop '{"last_assistant_message":"hello","stop_hook_active":false,"turn_id":"turn-1"}'
wait_state waiting
fire UserPromptSubmit '{"prompt":"again","turn_id":"turn-2"}'
wait_state working
fire Interrupt '{"turn_id":"turn-2"}'
wait_state waiting
[ "$(agent-director get --instance-id "$id" | jq -r '.session_id')" = "$sid" ] || fail "session_id changed during the turns"

# Step 5: nothing was ignored.
if [ -f "$trail" ]; then
    n="$(jq -c --arg x "$id" 'select(.event == "ad.hook.ignored" and .instance_id == $x)' "$trail" | wc -l)"
    [ "$n" -eq 0 ] || fail "$n ad.hook.ignored record(s) name $id"
fi

echo "agent-13 ok: SessionStart -> waiting (session $sid, rollout kept), UserPromptSubmit -> working, Stop -> waiting, UserPromptSubmit -> working, Interrupt -> waiting"
```
