# agent-8: the Codex arguments agent-director must own are refused

Maps to SRD SR-1.3 step 7 and SR-2.3 (RN-13) of Epic 13 (`b.ssz`): a
`spawn --agent codex` whose agent arguments hold a Codex subcommand, a
`-c` override of `hooks`, `--remote`, `--disable hooks`, or (with
`disable_ask_user` on) a form that turns Codex's question tool on, is
refused with `ErrSpawnDeniedFlag` before anything is written. The error
names the flag and its key or feature, or the subcommand.

Agent and hooks (collector "Driver mode"): `[launch] codex_path` names the
Codex stand-in, so a regression that launched would start
`/opt/driver/stand-in-codex/codex`, which records its launch in
`/tmp/stand-in-codex/<instance id>/`; that directory must never appear.

## Setup

Per-case DB-reset: a fresh store, no tmux server. The case writes
`~/.config/agent-director/config.toml` with `[launch]` `codex_path` and
`[defaults]` `disable_ask_user = true`, and a Codex folder of its own
(`config.toml`, `auth.json` and `sessions/`) given to every call as
`--extra-env CODEX_HOME=<folder>`. It removes the config, the folder and the
stand-in's record root on exit.

## Steps

1. `agent-director spawn --agent codex --cwd /tmp --extra-env
   CODEX_HOME=<folder> --instance-id <id> -- <arg> ...` once for
   each of these argument lists:
   - `exec` (a subcommand);
   - `-c`, `hooks.Stop=[]`;
   - `--remote`, `ws://127.0.0.1:1`;
   - `--disable`, `hooks`;
   - `--enable`, `default_mode_request_user_input` (question tool, with
     `disable_ask_user` on);
   - `-c`, `features.default_mode_request_user_input=true` (the same
     setting as a config key).
2. After each call, check nothing was written.

## Pass criteria

- Every call exits 1 with `err_name` `ErrSpawnDeniedFlag` and an
  `err_description` containing, respectively: `codex subcommand exec`,
  `-c hooks.Stop`, `--remote: it attaches`, `--disable hooks`, `--enable
  default_mode_request_user_input`, `-c features.default_mode_request_user_input`.
- After each: `list` shows no rows, `spawns` holds none, the tmux session
  ids are unchanged, `/tmp/stand-in-codex` does not exist (no stand-in
  record), the Codex folder's tree is identical to the one taken before the
  first call, and `~/.codex` does not exist.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call. CODEX_HOME is
# unset here: the case gives the Codex folder to each call as --extra-env.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
codex_home=""
snap_before=""
fail() { echo "FAIL: $*"; exit 1; }

session_ids() { tmux ls -F '#{session_id}' 2>/dev/null || true; }
rm -rf "$rec_root"
sessions_before="$(session_ids)"
cleanup() {
    local sid
    for sid in $(session_ids); do
        case " $(printf '%s ' $sessions_before)" in
            *" $sid "*) ;;
            *) tmux kill-session -t "$sid" 2>/dev/null || true ;;
        esac
    done
    rm -f "$config" ${snap_before:+"$snap_before"}
    rm -rf "$rec_root" ${codex_home:+"$codex_home"}
}
trap cleanup EXIT

[ -x "$standin" ] || fail "$standin is not an executable file"
[ ! -e "$HOME/.codex" ] || fail "$HOME/.codex exists before the case"

# Setup: the config, and a Codex folder the refused calls must leave alone.
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n\n[defaults]\ndisable_ask_user = true\n' "$standin" > "$config"
codex_home="$(mktemp -d)"
printf 'model = "gpt-5"\n' > "$codex_home/config.toml"
printf '{"stand-in":true}\n' > "$codex_home/auth.json"
chmod 600 "$codex_home/auth.json"
mkdir -p "$codex_home/sessions"
snap_before="$(mktemp)"
/opt/driver/tree-snapshot.sh "$codex_home" > "$snap_before"

# refuse WANT_TEXT AGENT_ARG...: a codex spawn with the AGENT_ARGs after -- must
# exit 1 with a JSON envelope on stderr whose err_name is ErrSpawnDeniedFlag and
# whose err_description contains WANT_TEXT.
refuse() {
    local want_text="$1" err code envelope got desc id a
    shift
    id="ag-ca-$(cat /proc/sys/kernel/random/uuid)"
    local args=(--)
    for a in "$@"; do args+=("$a"); done
    set +e
    err="$(agent-director spawn --agent codex --cwd /tmp --extra-env "CODEX_HOME=$codex_home" --instance-id "$id" "${args[@]}" 2>&1 >/dev/null)"
    code=$?
    set -e
    [ "$code" -eq 1 ] || fail "spawn with agent args [$*] exited $code; want 1: $err"
    envelope="$(printf '%s\n' "$err" | awk '/^\{/{ln=$0} END{print ln}')"
    got="$(jq -r '.err_name' <<<"$envelope")"
    [ "$got" = "ErrSpawnDeniedFlag" ] || fail "agent args [$*]: err_name=[$got]; want ErrSpawnDeniedFlag; stderr: $err"
    desc="$(jq -r '.err_description' <<<"$envelope")"
    case "$desc" in
        *"$want_text"*) ;;
        *) fail "agent args [$*]: err_description=[$desc]; want it to contain [$want_text]" ;;
    esac
    assert_untouched "agent args [$*]" "$id"
}

# assert_untouched WHAT ID: nothing the refused call could write exists.
assert_untouched() {
    local what="$1" id="$2" n_list n_db after
    n_list="$(agent-director list | jq '.spawns | length')"
    [ "$n_list" = "0" ] || fail "$what: list shows $n_list row(s); want 0"
    n_db="$(/opt/driver/sql.sh -readonly "$db" 'SELECT COUNT(*) FROM spawns;')"
    [ "$n_db" = "0" ] || fail "$what: the store holds $n_db spawns row(s); want 0"
    after="$(session_ids)"
    [ "$after" = "$sessions_before" ] || fail "$what: tmux sessions changed; before=[$sessions_before] after=[$after]"
    [ ! -e "$rec_root" ] || fail "$what: $rec_root exists: the stand-in codex was launched (id $id)"
    /opt/driver/tree-snapshot.sh "$codex_home" | diff -u "$snap_before" - || fail "$what: the Codex folder changed"
    [ ! -e "$HOME/.codex" ] || fail "$what: $HOME/.codex was created"
}

# Steps 1-2: each refused argument list.
refuse "codex subcommand exec" exec
refuse "-c hooks.Stop" -c 'hooks.Stop=[]'
refuse "--remote: it attaches" --remote ws://127.0.0.1:1
refuse "--disable hooks" --disable hooks
refuse "--enable default_mode_request_user_input" --enable default_mode_request_user_input
refuse "-c features.default_mode_request_user_input" -c features.default_mode_request_user_input=true

echo "agent-8 ok: codex agent args with a subcommand, -c hooks, --remote, --disable hooks and two question-tool forms -> ErrSpawnDeniedFlag; no row, no session, no stand-in record, Codex folder unchanged, no ~/.codex"
```
