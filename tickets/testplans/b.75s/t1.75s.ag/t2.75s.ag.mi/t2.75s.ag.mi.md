# agent-1: no agent given anywhere is ErrAgentMissing, nothing written

Maps to SRD SR-14.5 and the agent-input rule of Epic 13 (`b.ssz`): a
`spawn` names its agent (`--agent claude` or `--agent codex`) or takes
its template's; with neither it is refused with `ErrAgentMissing` before
anything is written. No config key or environment variable supplies an
agent.

Agent and hooks (collector "Driver mode"): each refused call carries the
`PATH=/opt/driver/stand-in-claude:$PATH` prefix, so a regression that
launched would start the stand-in, not Claude Code.

## Setup

Per-case DB-reset: a fresh store, no tmux server, no templates
directory.

## Steps

1. `agent-director spawn --cwd /tmp --tmux-session-name cd-ag-mi1`
   (no `--agent`, no template).
2. Check nothing was written.
3. `agent-director make-template --name ag-mi-tpl --cwd /tmp` (a
   template that names no agent; this is allowed).
4. `agent-director spawn --template ag-mi-tpl --tmux-session-name
   cd-ag-mi2` (no `--agent`; the template names none).
5. Check nothing was written but the template file of step 3.

## Pass criteria

- Steps 1 and 4 exit 1 with `err_name` `ErrAgentMissing` and an
  `err_description` containing `gives no agent`.
- After step 1: `list` shows no rows, `spawns` holds none, the tmux
  session ids are unchanged, and the templates directory is empty.
- After step 4: still no rows and no new session; the templates
  directory holds only `ag-mi-tpl.toml`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN

db="$HOME/.config/agent-director/state.db"
tpl_dir="$HOME/.config/agent-director/templates"
# Should a refused call ever launch, it gets the stand-in, never Claude Code.
standin_path="/opt/driver/stand-in-claude:$PATH"
fail() { echo "FAIL: $*"; exit 1; }

session_ids() { tmux ls -F '#{session_id}' 2>/dev/null || true; }
session_names() { tmux ls -F '#{session_name}' 2>/dev/null || true; }
tpl_files() { ls -A "$tpl_dir" 2>/dev/null || true; }
# The templates directory is shared by every case in this epic: start from none.
rm -rf "$tpl_dir"
sessions_before="$(session_ids)"
cleanup() {
    local sid
    for sid in $(session_ids); do
        case " $(printf '%s ' $sessions_before)" in
            *" $sid "*) ;;
            *) tmux kill-session -t "$sid" 2>/dev/null || true ;;
        esac
    done
    # Leave no template behind for the next case.
    rm -rf "$tpl_dir"
}
trap cleanup EXIT

# refuse WANT_ERR_NAME WANT_TEXT ARGS...: agent-director ARGS... must exit 1
# with a JSON envelope on stderr whose err_name is WANT_ERR_NAME and whose
# err_description contains WANT_TEXT.
refuse() {
    local want_name="$1" want_text="$2" err code envelope got desc
    shift 2
    set +e
    err="$(PATH="$standin_path" agent-director "$@" 2>&1 >/dev/null)"
    code=$?
    set -e
    [ "$code" -eq 1 ] || fail "agent-director $* exited $code; want 1: $err"
    envelope="$(printf '%s\n' "$err" | awk '/^\{/{ln=$0} END{print ln}')"
    got="$(jq -r '.err_name' <<<"$envelope")"
    [ "$got" = "$want_name" ] || fail "agent-director $*: err_name=[$got]; want $want_name; stderr: $err"
    desc="$(jq -r '.err_description' <<<"$envelope")"
    case "$desc" in
        *"$want_text"*) ;;
        *) fail "agent-director $*: err_description=[$desc]; want it to contain [$want_text]" ;;
    esac
}

# assert_untouched TEMPLATE_FILES: nothing the refused calls could write exists:
# no row (list and the store), no new tmux session, and the templates dir
# holds exactly TEMPLATE_FILES (what the case itself wrote, "" for none).
assert_untouched() {
    local want_tpl="$1" n_list n_db after
    n_list="$(agent-director list | jq '.spawns | length')"
    [ "$n_list" = "0" ] || fail "list shows $n_list row(s); want 0"
    n_db="$(/opt/driver/sql.sh -readonly "$db" 'SELECT COUNT(*) FROM spawns;')"
    [ "$n_db" = "0" ] || fail "the store holds $n_db spawns row(s); want 0"
    after="$(session_ids)"
    [ "$after" = "$sessions_before" ] || fail "tmux sessions changed; before=[$sessions_before] after=[$after]"
    [ "$(tpl_files)" = "$want_tpl" ] || fail "templates dir holds [$(tpl_files)]; want [$want_tpl]"
}

want_text="gives no agent"

# Steps 1-2: no agent, no template.
refuse ErrAgentMissing "$want_text" spawn --cwd /tmp --tmux-session-name cd-ag-mi1
assert_untouched ""
if session_names | grep -qx 'cd-ag-mi1'; then fail "a tmux session named cd-ag-mi1 exists"; fi

# Step 3: a template that names no agent.
agent-director make-template --name ag-mi-tpl --cwd /tmp >/dev/null
[ "$(tpl_files)" = "ag-mi-tpl.toml" ] || fail "templates dir holds [$(tpl_files)]; want only ag-mi-tpl.toml"

# Steps 4-5: the template names none either.
refuse ErrAgentMissing "$want_text" spawn --template ag-mi-tpl --tmux-session-name cd-ag-mi2
assert_untouched "ag-mi-tpl.toml"
if session_names | grep -qx 'cd-ag-mi2'; then fail "a tmux session named cd-ag-mi2 exists"; fi

echo "agent-1 ok: no agent from the call or the template -> ErrAgentMissing twice; no row, no session, no template written by a refused call"
```
