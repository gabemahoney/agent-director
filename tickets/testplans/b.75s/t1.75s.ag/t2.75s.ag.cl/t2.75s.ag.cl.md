# agent-9: the Codex launch the stand-in records

Maps to SRD SR-5.0, SR-5.1, SR-5.2, SR-5.3, SR-10.1 and SR-14.5 of Epic 13
(`b.ssz`): `[launch] codex_path` is the program every Codex launch runs. A
`spawn --agent codex` with `--relay-mode on` launches it as
`<codex> [caller's options] <agent-director's settings> -- <prompt>`: the
settings come after every pass-through option and before the first `--`,
and the prompt after it is untouched. The result says `pre_trust` `ok`
(Codex pre-trust puts the folder trust on the command line and writes no
file), the row is `pending` with agent `codex`, and the launch's
environment carries `AGENT_DIRECTOR_AGENT=codex` and an empty `BASH_ENV`.

Agent and hooks (collector "Driver mode"): the stand-in is
`/opt/driver/stand-in-codex/codex`, named by `codex_path`; the PATH has no
stand-in directory on it. It records its argv and environment in
`/tmp/stand-in-codex/<instance id>/` (`argv` and `env`, NUL-terminated
entries) and runs no hook (a case has to opt in with `AD_CODEX_HOOK`).

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"` and
removes it, the cwd it makes and the stand-in's record directory on exit.

## Steps

1. Write the config. Make a cwd `/tmp/ag-cl-XXXXXX`.
2. `agent-director spawn --agent codex --cwd <cwd> --instance-id <id>
   --tmux-session-name cd-ag-cl --relay-mode on -- --model gpt-5
   -- 'say hello'` → exit 0.
3. `agent-director get --instance-id <id>` and the row's `agent`, `state`,
   `relay_mode` and `agent_args` from the store.
4. Within 10 s the stand-in's record exists. Read its `argv` and `env`.

## Pass criteria

- Step 2: exit 0, `instance_id` is the id sent, `pre_trust` is `ok`, and
  stderr is empty (no pre-trust warning).
- Step 3: `get` shows `agent` `codex` and `state` `pending`; the store has
  `agent` `codex`, `relay_mode` `on` and `agent_args` exactly
  `["--model","gpt-5","--","say hello"]` (agent-director's own settings are
  not stored).
- Step 4, the argv: it starts with the caller's `--model` `gpt-5`; the
  first `--` is the last but one argument and the last is `say hello`.
  Before the `--`, and only there: `--no-daemon` once; eight `-c`
  `hooks.<Event>=` values (SessionStart, UserPromptSubmit, PreToolUse,
  PostToolUse, PermissionRequest, Stop, Interrupt, SessionEnd), in that
  order, each holding exactly one `exec '<binary>' hook --agent codex` with
  `<binary>` the running agent-director; one `-c` `hooks.state={...}` after
  them with eight `trusted_hash="sha256:` entries; `-c`
  `tui.disable_paste_burst=true`; `--enable` `write_stdin_approval`; `-c`
  `projects={"<cwd>"={trust_level="trusted"}}`; and
  `--dangerously-bypass-hook-trust`. No `resume`, no
  `tools.experimental_request_user_input.enabled=false` (`disable_ask_user`
  is off) and no `--remote`.
- Step 4, the environment: `AGENT_DIRECTOR_AGENT=codex`, `BASH_ENV=` (set,
  empty), `AGENT_DIRECTOR_RELAY_MODE=on`, `AGENT_DIRECTOR_INSTANCE_ID=<id>`,
  and no `CODEX_HOME`.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
ad="$(command -v agent-director)"
ad_real="$(readlink -f "$ad")"
uuid="$(cat /proc/sys/kernel/random/uuid)"
id="ag-cl-$uuid"
name="cd-ag-cl"
cwd=""
sock=""
sid=""
fail() { echo "FAIL: $*"; exit 1; }
spawn_err="$(mktemp)"
cleanup() {
    [ -z "$sid" ] || tmux ${sock:+-S "$sock"} kill-session -t "$sid" 2>/dev/null || true
    rm -f "$config" "$spawn_err"
    rm -rf "$rec_root/$id" ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# Step 1: the config and the cwd.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-cl-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"

# Step 2: the spawn, with an option of the caller's and a prompt after the first --.
set +e
out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$id" --tmux-session-name "$name" --relay-mode on \
    -- --model gpt-5 -- 'say hello' 2>"$spawn_err")"
rc=$?
set -e
[ "$rc" -eq 0 ] || fail "spawn exited $rc; want 0: $(cat "$spawn_err")"
[ "$(jq -r '.instance_id' <<<"$out")" = "$id" ] || fail "spawn echoed [$out]; want id $id"
[ "$(jq -r '.pre_trust' <<<"$out")" = "ok" ] || fail "spawn result pre_trust is [$(jq -r '.pre_trust' <<<"$out")]; want ok: $out"
[ ! -s "$spawn_err" ] || fail "spawn wrote to stderr: $(cat "$spawn_err")"

# Step 3: get and the store.
row="$(agent-director get --instance-id "$id")"
got="$(jq -r '[.agent, .state] | join("|")' <<<"$row")"
[ "$got" = "codex|pending" ] || fail "get agent|state=[$got]; want codex|pending: $row"
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $1 FROM spawns WHERE instance_id = '$id';"; }
[ "$(col agent)" = "codex" ] || fail "spawns.agent=[$(col agent)]; want codex"
[ "$(col relay_mode)" = "on" ] || fail "spawns.relay_mode=[$(col relay_mode)]; want on"
stored_args="$(col agent_args | jq -c .)"
[ "$stored_args" = '["--model","gpt-5","--","say hello"]' ] || fail "spawns.agent_args=[$stored_args]; want [\"--model\",\"gpt-5\",\"--\",\"say hello\"]"
sock="$(col "IFNULL(tmux_socket,'')")"
pane="$(col "IFNULL(pane_id,'')")"
[ -n "$pane" ] || fail "the row records no pane_id"
sid="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null)" || fail "the row's pane $pane is not in any tmux session: the launch did not stay up"

# Step 4: the stand-in's record (its pid file is written last).
rec="$rec_root/$id"
end=$(( $(date +%s%N) + 10000000000 ))
until [ -s "$rec/pid" ]; do
    [ "$(date +%s%N)" -lt "$end" ] || fail "no stand-in record at $rec within 10 s: the program named by codex_path did not run"
    sleep 0.1
done
mapfile -d '' -t argv < "$rec/argv"
mapfile -d '' -t envv < "$rec/env"

# The first --, and only agent-director's settings before it.
dd=-1
for i in "${!argv[@]}"; do
    if [ "${argv[$i]}" = "--" ]; then dd="$i"; break; fi
done
n="${#argv[@]}"
[ "$dd" -ge 0 ] || fail "argv has no --: [${argv[*]}]"
[ "$dd" -eq $((n - 2)) ] && [ "${argv[$((n - 1))]}" = "say hello" ] || fail "argv does not end with -- 'say hello': [${argv[*]}]"
[ "${argv[0]}" = "--model" ] && [ "${argv[1]}" = "gpt-5" ] || fail "argv starts [${argv[0]} ${argv[1]:-}]; want the caller's --model gpt-5 first"
pre=("${argv[@]:0:dd}")

count_pre() { local c=0 a; for a in "${pre[@]}"; do [ "$a" = "$1" ] && c=$((c + 1)); done; echo "$c"; }
# value_after_c PREFIX: the value of the -c in pre whose value starts with PREFIX ("" when none; one match expected).
value_after_c() {
    local j found="" hits=0
    for ((j = 0; j + 1 < ${#pre[@]}; j++)); do
        if [ "${pre[$j]}" = "-c" ] && [[ "${pre[$((j + 1))]}" == "$1"* ]]; then
            found="${pre[$((j + 1))]}"
            hits=$((hits + 1))
        fi
    done
    # Runs in a command substitution: report on stderr and end it, which ends the case.
    [ "$hits" -eq 1 ] || { echo "FAIL: $hits -c values start with [$1] before the --; want 1: [${pre[*]}]" >&2; exit 1; }
    printf '%s' "$found"
}
index_of_c() {
    local j
    for ((j = 0; j + 1 < ${#pre[@]}; j++)); do
        if [ "${pre[$j]}" = "-c" ] && [[ "${pre[$((j + 1))]}" == "$1"* ]]; then echo "$j"; return; fi
    done
    echo -1
}

[ "$(count_pre --no-daemon)" = "1" ] || fail "--no-daemon appears $(count_pre --no-daemon) times before the --; want 1"
[ "$(count_pre --dangerously-bypass-hook-trust)" = "1" ] || fail "--dangerously-bypass-hook-trust appears $(count_pre --dangerously-bypass-hook-trust) times before the --; want 1 (the folder is trusted)"
[ "$(count_pre --enable)" = "1" ] && [[ " ${pre[*]} " == *" --enable write_stdin_approval "* ]] || fail "want one --enable write_stdin_approval before the --: [${pre[*]}]"
[ "$(count_pre --remote)" = "0" ] && [ "$(count_pre resume)" = "0" ] || fail "--remote or resume in the argv: [${pre[*]}]"

last=-1
for ev in SessionStart UserPromptSubmit PreToolUse PostToolUse PermissionRequest Stop Interrupt SessionEnd; do
    val="$(value_after_c "hooks.$ev=")"
    at="$(index_of_c "hooks.$ev=")"
    [ "$at" -gt "$last" ] || fail "hooks.$ev is at argv index $at, not after the one before it ($last)"
    last="$at"
    n_exec="$(grep -o "exec '[^']*' hook --agent codex" <<<"$val" | wc -l)"
    [ "$n_exec" -eq 1 ] || fail "hooks.$ev holds $n_exec exec ... hook --agent codex entries; want 1: $val"
    bin="$(grep -o "exec '[^']*' hook --agent codex" <<<"$val" | sed "s/^exec '//; s/' hook --agent codex\$//")"
    [ "$(readlink -f "$bin")" = "$ad_real" ] || fail "hooks.$ev names [$bin]; want the running agent-director $ad_real"
done
state="$(value_after_c "hooks.state=")"
[ "$(index_of_c "hooks.state=")" -gt "$last" ] || fail "hooks.state is not after the eight events"
n_hash="$(grep -o 'trusted_hash="sha256:[0-9a-f]\{64\}"' <<<"$state" | wc -l)"
[ "$n_hash" -eq 8 ] || fail "hooks.state holds $n_hash trusted_hash entries; want 8: $state"
[ "$(value_after_c "tui.disable_paste_burst=")" = "tui.disable_paste_burst=true" ] || fail "tui.disable_paste_burst is not true"
[ "$(value_after_c "projects=")" = "projects={\"$cwd\"={trust_level=\"trusted\"}}" ] || fail "the folder trust is [$(value_after_c "projects=")]; want one for $cwd"
case " ${pre[*]} " in
    *"tools.experimental_request_user_input"*) fail "the question-tool setting is on the argv, disable_ask_user is off: [${pre[*]}]" ;;
esac

# The environment of the launch.
env_value() {
    local e
    for e in "${envv[@]}"; do
        case "$e" in "$1="*) printf '%s' "${e#*=}"; return 0 ;; esac
    done
    return 1
}
v="$(env_value AGENT_DIRECTOR_AGENT)" || fail "AGENT_DIRECTOR_AGENT is not set in the launch environment"
[ "$v" = "codex" ] || fail "AGENT_DIRECTOR_AGENT=[$v]; want codex"
v="$(env_value BASH_ENV)" || fail "BASH_ENV is not set in the launch environment; want it set and empty"
[ -z "$v" ] || fail "BASH_ENV=[$v]; want it empty"
[ "$(env_value AGENT_DIRECTOR_RELAY_MODE)" = "on" ] || fail "AGENT_DIRECTOR_RELAY_MODE=[$(env_value AGENT_DIRECTOR_RELAY_MODE || true)]; want on"
[ "$(env_value AGENT_DIRECTOR_INSTANCE_ID)" = "$id" ] || fail "AGENT_DIRECTOR_INSTANCE_ID=[$(env_value AGENT_DIRECTOR_INSTANCE_ID || true)]; want $id"
if env_value CODEX_HOME >/dev/null; then fail "CODEX_HOME is set in the launch environment; agent-director never sets it"; fi

echo "agent-9 ok: codex_path=$standin, relay on: pre_trust ok, row codex|pending; argv [--model gpt-5 | settings | -- 'say hello'], 8 hook entries naming $ad_real, 8 trust hashes, folder trust $cwd; env AGENT_DIRECTOR_AGENT=codex, BASH_ENV empty"
```
