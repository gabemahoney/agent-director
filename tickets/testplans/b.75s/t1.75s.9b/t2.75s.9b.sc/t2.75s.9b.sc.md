# Sweep-only recovery with a real owned process

Maps to b.x9o / t3.7tz.1y.cw.4z and t3.7tz.1y.cb.kn. Use the existing Codex stand-in,
owned-pane hook helper and captured errored-turn rollout. No status, get or
list runs between creating the recovery evidence and inspecting its result.

## Setup and steps

1. Spawn a Codex stand-in with a private cwd and CODEX_HOME. Report SessionStart
   and UserPromptSubmit from the owned pane using the captured session/turn.
2. Install the captured completion without a Stop hook. Read SQLite directly
   to establish working. Run sweep once and inspect SQLite directly.
3. End that actual process externally; once its exact identity is gone, run
   sweep once more and inspect SQLite directly.

## Pass criteria

First sweep has the exact empty legacy counters and commits waiting with
codex_turn_ended, no invented last_seen timestamp and the original process
still alive. The pane content remains unchanged across the sweep (no typing).
The second sweep reports exactly the killed id and commits missing. This
synthetic captured fixture does not qualify a new provider version.

```bash
set -euo pipefail
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN OPENAI_API_KEY CODEX_API_KEY AGENT_DIRECTOR_INSTANCE_ID
work="$(mktemp -d /tmp/recovery-shared.XXXXXX)"
id="$(cat /proc/sys/kernel/random/uuid)"
config="$HOME/.config/agent-director/config.toml"
db="$HOME/.config/agent-director/state.db"
mkdir -p "$work/cwd" "$work/codex/sessions/2026/10/23"
cp "$config" "$work/config.original" 2>/dev/null || true
cleanup() {
    if [[ -s "$work/socket" ]]; then tmux -S "$(cat "$work/socket")" kill-session -t "cd-$id" 2>/dev/null || true; fi
    if [[ -f "$work/config.original" ]]; then cp "$work/config.original" "$config"; else rm -f "$config"; fi
    rm -rf "$work"
}
trap cleanup EXIT
printf '[launch]\ncodex_path = "/opt/driver/stand-in-codex/codex"\n' > "$config"
sid=01a16668-480f-7080-adb4-b91464140a96
turn=01a16668-4862-7930-b160-2863caf991d1
capture=/work/source/internal/testsupport/codexfix/testdata/rollout-errored-turn.jsonl
rollout="$work/codex/sessions/2026/10/23/rollout-2026-10-23T01-37-52-$sid.jsonl"
head -n -1 "$capture" > "$rollout"
agent-director spawn --agent codex --instance-id "$id" --tmux-session-name "cd-$id" --cwd "$work/cwd" --extra-env "CODEX_HOME=$work/codex" > "$work/spawn.json"
[[ "$(jq -r .instance_id "$work/spawn.json")" == "$id" ]]
for _ in {1..100}; do [[ -s "/tmp/stand-in-codex/$id/pid" ]] && break; sleep 0.1; done
[[ -s "/tmp/stand-in-codex/$id/pid" ]]
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $1 FROM spawns WHERE instance_id='$id';"; }
col tmux_socket > "$work/socket"
socket="$(cat "$work/socket")"
pane="$(col pane_id)"
fire() {
    local payload
    payload="$(jq -cn --arg event "$1" --arg sid "$sid" --arg turn "$turn" --arg path "$rollout" --arg cwd "$work/cwd" '{hook_event_name:$event,session_id:$sid,turn_id:$turn,transcript_path:$path,cwd:$cwd,source:"startup",prompt:"capture"}')"
    /opt/driver/pane-hook.sh --agent codex "$id" "$payload" > "$work/hook.out"
    [[ ! -s "$work/hook.out" ]]
}
fire SessionStart
fire UserPromptSubmit
[[ "$(col state)" == working ]]
last_seen="$(col last_seen_at)"
pid="$(col pid)"
start="$(col proc_starttime)"
process_start() { python3 -c 'import pathlib,sys; print(pathlib.Path("/proc/"+sys.argv[1]+"/stat").read_text().rsplit(")",1)[1].split()[19])' "$pid"; }
[[ "$(process_start)" == "$start" ]]
cp "$capture" "$rollout"
tmux -S "$socket" capture-pane -p -t "$pane" > "$work/before"
agent-director sweep > "$work/sweep.json"
jq -e --arg id "$id" '(. | del(.checks)) == {count:0,ids:[],unverified:0,unverified_ids:[]} and (.checks | length == 1) and (.checks[0] | .instance_id == $id and .state == "waiting" and .process_liveness == "alive" and .completion == "complete" and .reason == null and (.confirmations | keys == ["displayed_state","process_liveness","state"]) and .confirmations.state.source == "transcript" and .confirmations.process_liveness.source == "process")' "$work/sweep.json"
[[ "$(col state)" == waiting ]]
[[ "$(col codex_turn_ended)" == "$turn" ]]
[[ "$(col last_seen_at)" == "$last_seen" ]]
[[ "$(process_start)" == "$start" ]]
tmux -S "$socket" capture-pane -p -t "$pane" > "$work/after"
cmp "$work/before" "$work/after"
tmux -S "$socket" kill-session -t "cd-$id"
for _ in {1..100}; do [[ ! -r "/proc/$pid/stat" ]] && break; sleep 0.1; done
[[ ! -r "/proc/$pid/stat" ]]
agent-director sweep > "$work/exit.json"
jq -e --arg id "$id" '(. | del(.checks)) == {count:1,ids:[$id],unverified:0,unverified_ids:[]} and (.checks | length == 1) and (.checks[0] | .instance_id == $id and .state == "missing" and .process_liveness == "exited" and .completion == "complete" and .reason == null and .confirmations.process_liveness.source == "process")' "$work/exit.json"
[[ "$(col state)" == missing ]]
[[ -n "$(col ended_at)" ]]
echo 'recov-9 shared sweep passed'
```
