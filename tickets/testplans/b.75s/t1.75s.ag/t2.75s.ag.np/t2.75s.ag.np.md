# agent-11: a codex row that never gets a first prompt stays pending and is noted unreported

Maps to SRD SR-6.1, SR-11.3 and SR-14.5 of Epic 13 (`b.ssz`): a
Codex agent sends `SessionStart` at its first prompt, so a launched codex
row stays `pending` until then. `find-missing` leaves a pending row alone
inside the pending grace period (`[tmux] pending_grace_seconds`), and past
it notes a pending row whose agent process is alive `unreported`
(`liveness_note`): the row stays `pending`, is never listed in `ids`, and
the caller reads its pane before sending anything.

Agent and hooks (collector "Driver mode"): the pane process is the Codex
stand-in `/opt/driver/stand-in-codex/codex`, named by `[launch] codex_path`.
Nothing types a hook into it, so no `SessionStart` ever arrives.

## Setup

Per-case DB-reset. The case writes `~/.config/agent-director/config.toml`
with `[launch]` `codex_path = "/opt/driver/stand-in-codex/codex"` and
`[tmux]` `pending_grace_seconds = 30`, the shortest value the config accepts
at the default timeouts (a lower one is raised to 30 at load). It makes a
cwd, and removes it, the config and the stand-in's record directory on exit.

## Steps

1. Write the config; make the cwd.
2. `agent-director spawn --agent codex --cwd <cwd> --instance-id <id>
   --tmux-session-name cd-ag-np`; wait for the stand-in's record.
3. At once, `agent-director find-missing`, then `get`.
4. Run `find-missing` every 3 s until `get` shows `liveness_note`
   `unreported`, for at most 75 s.
5. `get`, and check the pane's process.

## Pass criteria

- Step 2: spawn exits 0; the row is `pending` with agent `codex`.
- Step 3 (inside the grace period): `count` 0, `ids` and `unverified_ids`
  empty; `get` shows `state` `pending` and no `liveness_note`.
- Step 4: every run has `count` 0 and empty `ids`; no `liveness_note` other
  than `unreported` ever shows; `unreported` shows within 75 s.
- Step 5: `state` is still `pending`, `liveness_unverified_since` is set,
  the recorded pane process is alive, and no `ad.hook.ignored` record names
  the row.

## Script

```bash
set -euo pipefail
# Credential isolation, before any tmux or agent-director call.
unset ANTHROPIC_API_KEY CLAUDE_CODE_OAUTH_TOKEN CODEX_HOME

db="$HOME/.config/agent-director/state.db"
trail="$HOME/.config/agent-director/ad-trail.jsonl"
config="$HOME/.config/agent-director/config.toml"
standin="/opt/driver/stand-in-codex/codex"
rec_root="/tmp/stand-in-codex"
uuid="$(cat /proc/sys/kernel/random/uuid)"
ids=()
codex_home=""
cwd=""
fail() { echo "FAIL: $*"; exit 1; }

# col ID EXPR: one expression of the row's spawns record.
col() { /opt/driver/sql.sh -readonly "$db" "SELECT $2 FROM spawns WHERE instance_id = '$1';"; }

cleanup() {
    local x row sock pane s
    for x in "${ids[@]}"; do
        row="$(col "$x" "IFNULL(tmux_socket,'') || '|' || IFNULL(pane_id,'')" 2>/dev/null || true)"
        sock="${row%%|*}"
        pane="${row#*|}"
        if [ -n "$pane" ]; then
            s="$(tmux ${sock:+-S "$sock"} display-message -p -t "$pane" '#{session_id}' 2>/dev/null || true)"
            [ -z "$s" ] || tmux ${sock:+-S "$sock"} kill-session -t "$s" 2>/dev/null || true
        fi
        rm -rf "$rec_root/$x"
    done
    rm -f "$config"
    rm -rf ${codex_home:+"$codex_home"} ${cwd:+"$cwd"}
    rmdir "$rec_root" 2>/dev/null || true
}
trap cleanup EXIT

# state ID: the row's state (get).
state() { agent-director get --instance-id "$1" | jq -r '.state'; }

# wait_state ID WANT: the row's state is WANT within 10 s.
wait_state() {
    local x="$1" want="$2" end got=""
    end=$(( $(date +%s) + 10 ))
    while :; do
        got="$(state "$x")"
        [ "$got" != "$want" ] || return 0
        [ "$(date +%s)" -lt "$end" ] || fail "row $x state is [$got]; want [$want] within 10 s"
        sleep 0.2
    done
}

# wait_record ID [OLD_PID]: the stand-in's record of ID is complete (its pid
# file is written last) and holds a pid other than OLD_PID.
wait_record() {
    local x="$1" old="${2:-}" end
    end=$(( $(date +%s%N) + 15000000000 ))
    while :; do
        if [ -s "$rec_root/$x/pid" ] && [ "$(cat "$rec_root/$x/pid")" != "$old" ]; then return 0; fi
        [ "$(date +%s%N)" -lt "$end" ] || fail "no new stand-in record for $x in $rec_root within 15 s"
        sleep 0.1
    done
}

# spawn_codex ID NAME [SPAWN_ARGS...]: a codex spawn that exits 0 and leaves a
# pending codex row whose stand-in recorded its launch.
spawn_codex() {
    local x="$1" name="$2" out
    shift 2
    ids+=("$x")
    out="$(agent-director spawn --agent codex --cwd "$cwd" --instance-id "$x" --tmux-session-name "$name" "$@")" \
        || fail "spawn $x exited non-zero"
    [ "$(jq -r '.instance_id' <<<"$out")" = "$x" ] || fail "spawn echoed [$out]; want id $x"
    wait_record "$x"
    [ "$(agent-director get --instance-id "$x" | jq -r '[.agent, .state] | join("|")')" = "codex|pending" ] \
        || fail "row $x is not codex|pending"
}

# wait_unreported ID: find-missing, run every 3 s, ends by noting the pending
# row ID unreported, within 75 s of now; every run lists nothing (a pending row
# is never marked missing for being unreported).
wait_unreported() {
    local x="$1" end res note=""
    end=$(( $(date +%s) + 75 ))
    while :; do
        res="$(agent-director find-missing)" || fail "find-missing exited non-zero"
        [ "$(jq -c '[.count, .ids, (.unverified_ids // [])]' <<<"$res")" = '[0,[],[]]' ] \
            || fail "find-missing result [$res]; want count 0 and no ids (the row is only noted)"
        note="$(agent-director get --instance-id "$x" | jq -r '.liveness_note // ""')"
        [ "$note" != "unreported" ] || return 0
        [ -z "$note" ] || fail "liveness_note is [$note]; want none until unreported"
        [ "$(date +%s)" -lt "$end" ] || fail "row $x was not noted unreported within 75 s (pending_grace_seconds = 30)"
        sleep 3
    done
}

id="ag-np-$uuid"

# Step 1: the config with the shortest pending grace, and the cwd.
[ -x "$standin" ] || fail "$standin is not an executable file"
mkdir -p "$HOME/.config/agent-director"
printf '[launch]\ncodex_path = "%s"\n\n[tmux]\npending_grace_seconds = 30\n' "$standin" > "$config"
cwd="$(mktemp -d /tmp/ag-np-XXXXXX)"
cwd="$(cd -P "$cwd" && pwd)"

# Step 2: spawn; no hook is ever typed.
spawn_codex "$id" cd-ag-np

# Step 3: inside the grace period nothing is judged or noted.
res="$(agent-director find-missing)" || fail "find-missing exited non-zero"
[ "$(jq -c '[.count, .ids, (.unverified_ids // [])]' <<<"$res")" = '[0,[],[]]' ] || fail "find-missing [$res]; want count 0, no ids"
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '.state' <<<"$row")" = "pending" ] || fail "state is [$(jq -r '.state' <<<"$row")] inside the grace period; want pending"
[ -z "$(jq -r '.liveness_note // ""' <<<"$row")" ] || fail "liveness_note is [$(jq -r '.liveness_note' <<<"$row")] inside the grace period; want none"

# Step 4: past the grace period the live pending row is noted unreported.
wait_unreported "$id"

# Step 5: still pending, still alive, no hook ever reached it.
row="$(agent-director get --instance-id "$id")"
[ "$(jq -r '.state' <<<"$row")" = "pending" ] || fail "state is [$(jq -r '.state' <<<"$row")] after the note; want pending"
[ -n "$(jq -r '.liveness_unverified_since // ""' <<<"$row")" ] || fail "liveness_unverified_since is not set after the unreported note: $row"
pane_pid="$(col "$id" pane_pid)"
[ -n "$pane_pid" ] && kill -0 "$pane_pid" 2>/dev/null || fail "the row's pane process [$pane_pid] is not alive"
if [ -f "$trail" ]; then
    n="$(jq -c --arg x "$id" 'select(.event == "ad.hook.ignored" and .instance_id == $x)' "$trail" | wc -l)"
    [ "$n" -eq 0 ] || fail "$n ad.hook.ignored record(s) name $id"
fi

echo "agent-11 ok: codex row with no first prompt: pending and unlisted inside the grace period, noted unreported past it (find-missing count 0), pane process alive"
```
